# Adelphai Dorleyaiou — the Dorley programme. Scope: the new novice.

tm_begin_programme_effect = {
	save_scope_as = tm_novice
	$CAPTOR$ = { save_scope_as = tm_captor }

	# Sponsor: a daughter of the knife at the captor's court if there is one, else the captor.
	if = {
		limit = { scope:tm_captor = { any_courtier = { tm_is_daughter_trigger = yes } } }
		scope:tm_captor = {
			random_courtier = {
				limit = { tm_is_daughter_trigger = yes }
				save_scope_as = tm_sponsor
			}
		}
	}
	else = {
		scope:tm_captor = { save_scope_as = tm_sponsor }
	}

	add_trait = tm_novice
	set_variable = { name = tm_progress value = 0 }
	set_variable = { name = tm_levers value = 0 }
	scope:tm_sponsor = { set_relation_guardian = scope:tm_novice }		# guardian <-> ward, vanilla relation

	scope:tm_captor = {
		trigger_event = {
			id = tm_dorley.0010
			months = 3
		}
	}
}

# After a refuge candidate joins the court, offer to adopt her at once (Daughters Chosen faiths only).
tm_refuge_offer_adoption_effect = {
	$CANDIDATE$ = { save_scope_as = tm_refuge_candidate }
	if = {
		limit = { rite ?= { rite_has_parameter = tm_adoptive_matriline } }
		trigger_event = { id = tm_refuge.0011 days = 1 }
	}
}
