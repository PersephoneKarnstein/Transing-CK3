# Crusader Dolls — how the faiths grow (docs/faith_design.md §14).
# A 119-year observe run (867–986) left the four faiths with 81 living members between them: nothing drew anyone in,
# only rulers could start a rite, and the AI never adopted. These effects are the growth engine.

# A Marked character of another faith hears of the Sisters and may join one of the six. The call travels by contact:
# members at court, members in her county, members within three counties of hers, the faith holding land in the realm,
# a House, a holy site. With none of these it is a rumour (2% a year). A larger faith has more members in more places,
# so it reaches more people; the ceiling is the number of Marked characters alive, since daughters bear no children.
# Widened a little after the 1188 observe run (docs/faith_design.md §16, "Collapse fixes"): the rumour from 1% to 2%,
# and +3 for a member within three counties. Ha-Nehpakhot calls only Rabbanite Jews (tm_rabbanite_jew_trigger).
# Scope: the character. Called yearly from tm_yearly_growth (tm_on_actions.txt).
tm_sisters_call_effect = {
	# Saved scopes outlive one character in the yearly loop: clear them, or the last character's faith carries over.
	clear_saved_scope = tm_call_faith
	clear_saved_scope = tm_call_county
	save_scope_as = tm_call_seeker			# root is the pulse, not her
	location ?= { county ?= { save_scope_as = tm_call_county } }
	# Which faith: one met at court first; then the faith of her own religion or region.
	if = {
		limit = {
			employer ?= {
				OR = {
					AND = { NOT = { this = scope:tm_call_seeker } rite ?= { rite_has_parameter = tm_sisters_faith } }
					any_courtier = { NOT = { this = scope:tm_call_seeker } rite ?= { rite_has_parameter = tm_sisters_faith } }
				}
			}
		}
		employer = {
			if = {
				limit = { NOT = { this = scope:tm_call_seeker } rite ?= { rite_has_parameter = tm_sisters_faith } }
				faith = { save_scope_as = tm_call_faith }
			}
			else = {
				random_courtier = {
					limit = { NOT = { this = scope:tm_call_seeker } rite ?= { rite_has_parameter = tm_sisters_faith } }
					faith = { save_scope_as = tm_call_faith }
				}
			}
		}
	}
	else_if = {	# a sworn sister's faith (round 2, T21): she counts as one met at court
		limit = { any_relation = { type = tm_sister rite ?= { rite_has_parameter = tm_sisters_faith } NOT = { faith = faith:tm_nehpakhot } } }
		random_relation = {
			type = tm_sister
			limit = { rite ?= { rite_has_parameter = tm_sisters_faith } NOT = { faith = faith:tm_nehpakhot } }
			faith = { save_scope_as = tm_call_faith }
		}
	}
	else_if = {	# Rabbanite Jews only: not Samaritans, Karaites, Kabarists or the Beta Israel (tm_fixes_triggers.txt)
		limit = { tm_rabbanite_jew_trigger = yes }
		faith:tm_nehpakhot = { save_scope_as = tm_call_faith }
	}
	else_if = {
		limit = {
			location ?= {
				OR = {
					geographical_region = world_asia_minor
					geographical_region = world_europe_south_east
				}
			}
		}
		faith:transbian_maximalism = { save_scope_as = tm_call_faith }
	}
	else_if = {
		limit = {
			location ?= {
				OR = {
					geographical_region = world_europe_north
					geographical_region = world_europe_west_britannia
					geographical_region = custom_eastern_baltic
				}
			}
		}
		faith:tm_baeddel = { save_scope_as = tm_call_faith }
	}
	else_if = {
		limit = {
			location ?= {
				OR = {
					geographical_region = world_asia_china
					geographical_region = world_steppe_tarim
				}
			}
		}
		faith:tm_yutai = { save_scope_as = tm_call_faith }
	}
	else_if = {
		limit = { location ?= { geographical_region = world_india_deccan } }
		faith:tm_garland = { save_scope_as = tm_call_faith }
	}
	else_if = {
		limit = { location ?= { geographical_region = world_africa_west } }
		faith:tm_badenya = { save_scope_as = tm_call_faith }
	}
	if = {
		limit = { exists = scope:tm_call_faith }
		random = {
			chance = 2									# a rumour only
			modifier = {								# a member at her court, or her lord
				add = 8
				employer ?= {
					OR = {
						AND = { NOT = { this = scope:tm_call_seeker } faith = scope:tm_call_faith }
						any_courtier = { NOT = { this = scope:tm_call_seeker } faith = scope:tm_call_faith }
					}
				}
			}
			modifier = {								# three or more
				add = 4
				employer ?= { any_courtier = { count >= 3 NOT = { this = scope:tm_call_seeker } faith = scope:tm_call_faith } }
			}
			modifier = {								# a member living in her county, at another court
				add = 4
				location ?= {
					county ?= {
						any_county_province = {
							any_character_in_location = {
								NOT = { employer ?= scope:tm_call_seeker.employer }
								faith = scope:tm_call_faith
							}
						}
					}
				}
			}
			modifier = {								# a member living within three counties of hers
				add = 3
				exists = scope:tm_call_county
				scope:tm_call_county = {
					tm_call_member_nearby_trigger = { FAITH = scope:tm_call_faith EXCLUDE = scope:tm_call_county }
				}
			}
			modifier = {								# the faith holds land in her realm
				add = 4
				top_liege ?= { any_realm_county = { faith = scope:tm_call_faith } }
			}
			modifier = {								# a Sisters' House in her county
				add = 5
				location ?= { county ?= { has_variable = tm_house } }
			}
			modifier = {								# a holy site of the faith in her duchy
				add = 3
				location ?= {
					county ?= {
						duchy ?= {
							any_de_jure_county = {
								any_county_province = { barony ?= { is_holy_site_of = scope:tm_call_faith } }
							}
						}
					}
				}
			}
			modifier = {								# a daughter of any faith at her court
				add = 3
				employer ?= { any_courtier = { tm_is_daughter_trigger = yes } }
			}
			modifier = {								# a sworn sister of the faith, wherever she lives (round 2, T21)
				add = 4
				any_relation = { type = tm_sister faith = scope:tm_call_faith }
			}
			modifier = {								# the user's "accepting place" (round 2, T29): sisters sent on from
				add = 3								# a closed House settled in her county in the last ten years
				exists = scope:tm_call_county
				scope:tm_call_county = { has_variable = tm_refugee_carried }
			}
			modifier = {								# she has already declared her name
				add = 4
				has_variable = tm_declared
			}
			modifier = {								# a count turns slowly: her county's faith goes with her
				factor = 0.25
				is_ruler = yes
				highest_held_title_tier <= tier_county
			}
			modifier = {								# a duke, king or emperor more slowly still, and at a cost
				factor = 0.05
				highest_held_title_tier >= tier_duchy
			}
			set_character_faith = scope:tm_call_faith
			add_stress = -20
			tm_track_effect = { FAITH = scope:tm_call_faith WHAT = joins }
			if = {
				limit = { is_ruler = yes }
				tm_track_effect = { FAITH = scope:tm_call_faith WHAT = ruler_joins }
			}
			if = {
				limit = { highest_held_title_tier >= tier_duchy }
				every_vassal = {
					add_opinion = { target = scope:tm_call_seeker modifier = tm_liege_joined_sisters_opinion }
				}
			}
		}
	}
}

# Offer the faith's rite to a Marked character or eunuch who is not a ruler: the AI weighs decisions only for rulers,
# so without this a courtier could never rise. Uses the same event chains as the decisions. Scope: the seeker.
tm_offer_courtier_rite_effect = {
	set_variable = { name = tm_rite_offered value = yes years = 3 }
	tm_track_effect = { FAITH = faith WHAT = offers }
	trigger_event = tm_growth.0001			# so that root is the seeker: the rite effects compare against root
}

# Begins the faith's own chain for the seeker. Scope and root: the seeker (tm_growth.0001).
tm_offer_begin_effect = {
	if = {
		limit = { rite ?= { rite_has_parameter = tm_seidr_gate } }
		tm_seidr_begin_effect = yes
	}
	else_if = {
		limit = { rite ?= { rite_has_parameter = tm_embryo_gate } }
		if = {
			limit = { NOT = { has_variable = tm_embryo_active } }
			trigger_event = tm_embryo.0001
		}
	}
	else_if = {
		limit = { rite ?= { rite_has_parameter = tm_turning_gate } }
		trigger_event = { id = tm_hipukh.0009 days = 1 }	# waits for Adar
	}
	else_if = {
		limit = { rite ?= { rite_has_parameter = tm_garland_gate } }
		trigger_event = { id = tm_garland.0009 days = 1 }	# waits for Mārkaḻi
	}
	else_if = {
		limit = { rite ?= { rite_has_parameter = tm_crossing_gate } }
		tm_crossing_begin_effect = yes
	}
	else_if = {
		limit = { rite ?= { rite_has_parameter = tm_knife_voluntary } }
		trigger_event = tm_knife.0001
	}
}

# The kahal's match. Ha-Nehpakhot holds no land and lives at Christian courts, where no other Jews are; Jewish
# communities arranged their own marriages, and the wider Jewry of Provence is not modelled as characters. An unmarried
# adult of the sect is matched with a Jewish spouse from outside the courts, who joins the sect and the court; their
# children are born into it. Without this the sect died out in every observe run. Scope: the unmarried member.
# Where the game has a Rabbanite Jew at her court or in its capital's pool, the match is made with them, inside the
# lord's lands (docs/wave3/survival.md); they join the sect as a kahal join.
tm_kahal_match_effect = {
	clear_saved_scope = tm_match_seeker		# called in a loop: the last match must not carry over
	clear_saved_scope = tm_match_spouse
	clear_saved_scope = tm_match_capital
	save_scope_as = tm_match_seeker
	employer ?= {
		capital_province ?= { save_scope_as = tm_match_capital }
		if = {
			limit = { any_courtier = { tm_surv_match_candidate_trigger = yes } }
			random_courtier = {
				limit = { tm_surv_match_candidate_trigger = yes }
				save_scope_as = tm_match_spouse
			}
		}
		else_if = {
			limit = {
				exists = scope:tm_match_capital
				any_pool_character = { province = scope:tm_match_capital tm_surv_match_candidate_trigger = yes }
			}
			random_pool_character = {
				province = scope:tm_match_capital
				limit = { tm_surv_match_candidate_trigger = yes }
				save_scope_as = tm_match_spouse
			}
		}
	}
	if = {
		limit = { exists = scope:tm_match_spouse }
		scope:tm_match_spouse = {
			tm_surv_kahal_take_in_effect = { COURT = scope:tm_match_seeker.employer }
		}
	}
	else_if = {
		limit = { exists = employer }
		create_character = {
			opposite_gender = scope:tm_match_seeker
			age = { 18 30 }
			culture = scope:tm_match_seeker.culture
			faith = faith:tm_nehpakhot
			dynasty = none
			employer = scope:tm_match_seeker.employer
			save_scope_as = tm_match_spouse
		}
	}
	else = {
		create_character = {
			opposite_gender = scope:tm_match_seeker
			age = { 18 30 }
			culture = scope:tm_match_seeker.culture
			faith = faith:tm_nehpakhot
			dynasty = none
			location = scope:tm_match_seeker.location
			save_scope_as = tm_match_spouse
		}
	}
	if = {
		limit = { is_female = yes }
		marry_matrilineal = scope:tm_match_spouse	# her house, her line
	}
	else = {
		marry = scope:tm_match_spouse
	}
	tm_track_effect = { FAITH = faith:tm_nehpakhot WHAT = matches }
}

# The pilgrimage to the kahal. The sect exists for its daughters, but a Marked son is born to about one family in a
# hundred: one household cannot raise enough of its own, and in observe runs it went generations with none. So word
# goes along the roads Jews travelled (Provence, Sepharad, Ashkenaz, the Radhanite routes) and a Marked young Jew comes
# to the seat to be taken in. An elder woman of the sect, a daughter first, adopts her into her house; the courtier
# rite (tm_offer_courtier_rite_effect) may offer the Turning in time. A larger kahal is known further, so draws more.
# Members of the sect living away from the kahal may also come home to it. Scope: none (tm_yearly_growth).
tm_kahal_pilgrimage_effect = {
	clear_saved_scope = tm_kahal_court
	clear_saved_scope = tm_pilgrim_home
	clear_saved_scope = tm_pilgrim
	clear_saved_scope = tm_pilgrim_mother
	# The kahal: the court where most of the sect live.
	every_living_character = {
		limit = {
			faith = faith:tm_nehpakhot
			employer ?= { tm_settled_court_trigger = yes }
		}
		employer = { add_to_temporary_list = tm_kahal_courts }
	}
	ordered_in_list = {
		list = tm_kahal_courts
		order_by = {
			value = 0
			every_courtier = {
				limit = { faith = faith:tm_nehpakhot }
				add = 1
			}
		}
		save_scope_as = tm_kahal_court
	}
	if = {
		limit = { exists = scope:tm_kahal_court }
		# A pilgrim from the wider Jewry.
		random = {
			chance = 10
			modifier = { add = 5 scope:tm_kahal_court = { any_courtier = { count >= 10 faith = faith:tm_nehpakhot } } }
			modifier = { add = 5 scope:tm_kahal_court = { any_courtier = { count >= 20 faith = faith:tm_nehpakhot } } }
			modifier = { add = 10 scope:tm_kahal_court = { any_courtier = { count >= 40 faith = faith:tm_nehpakhot } } }
			random_living_character = {
				limit = { tm_rabbanite_jew_trigger = yes }	# the sect is Rabbanite (tm_fixes_triggers.txt)
				save_scope_as = tm_pilgrim_home
			}
			if = {
				limit = { exists = scope:tm_pilgrim_home }
				create_character = {
					gender = male
					age = { 16 26 }
					culture = scope:tm_pilgrim_home.culture
					faith = faith:tm_nehpakhot
					dynasty = none
					trait = tm_transgender
					employer = scope:tm_kahal_court
					save_scope_as = tm_pilgrim
				}
			}
			else = {
				create_character = {
					gender = male
					age = { 16 26 }
					culture = scope:tm_kahal_court.culture
					faith = faith:tm_nehpakhot
					dynasty = none
					trait = tm_transgender
					employer = scope:tm_kahal_court
					save_scope_as = tm_pilgrim
				}
			}
			scope:tm_kahal_court = {
				ordered_courtier = {
					limit = {
						is_female = yes
						faith = faith:tm_nehpakhot
						age >= { value = scope:tm_pilgrim.age add = 16 }
					}
					order_by = {
						value = age
						if = { limit = { tm_is_daughter_trigger = yes } add = 100 }
					}
					save_scope_as = tm_pilgrim_mother
				}
			}
			if = {
				limit = { exists = scope:tm_pilgrim_mother }
				adopt_effect = {
					CHILD = scope:tm_pilgrim
					ADOPTER = scope:tm_pilgrim_mother
				}
			}
			tm_track_effect = { FAITH = faith:tm_nehpakhot WHAT = pilgrims }
		}
		# Members of the sect living away come home.
		every_living_character = {
			limit = {
				faith = faith:tm_nehpakhot
				is_adult = yes
				is_ruler = no
				is_imprisoned = no
				OR = {
					has_trait = tm_transgender
					tm_is_daughter_trigger = yes
				}
				NOT = { employer ?= scope:tm_kahal_court }
				NOT = { this = scope:tm_kahal_court }
			}
			random = {
				chance = 10
				save_scope_as = tm_pilgrim
				scope:tm_kahal_court = { add_courtier = scope:tm_pilgrim }
				tm_track_effect = { FAITH = faith:tm_nehpakhot WHAT = pilgrims }
			}
		}
	}
}

# Two Marked young women of the faith (the game codes them as male) at a seat court, so the courtier rite has seekers from the start.
tm_seed_marked_courtiers_effect = {
	while = {
		count = 2
		create_character = {
			gender = male
			age = { 16 28 }
			culture = $HOST$.culture
			faith = $FAITH$
			employer = $HOST$
			dynasty = none
			trait = tm_transgender
			after_creation = { tm_register_seed_effect = { SEAT = $SEAT$ } }
		}
	}
}

# A seat county at game start. The county takes the faith. If its lord holds other counties and this is not their
# capital, the county goes to a member of the faith as the lord's vassal: a landed foothold whose own chaplain keeps it,
# and a count under a liege cannot march on a kingdom (a converted independent lord's line formed Scandinavia in one
# observe run, and held Mali in another). Otherwise the lord keeps it and the faith lives at their court.
# Saves scope:tm_foothold_host, the court the faith's seed characters should join.
# If the caller saved scope:tm_foothold_preset (a bookmark character), she is the vassal instead of a new woman.
# $COUNTY$ a county title, $FAITH$ a faith, $SEAT$ the seat's name (tm_register_seed_effect).
tm_seed_foothold_effect = {
	clear_saved_scope = tm_foothold_lord		# the effect runs once per seat: earlier seats' scopes must not carry over
	clear_saved_scope = tm_foothold_count
	clear_saved_scope = tm_foothold_host
	$COUNTY$ = {
		if = {
			limit = { NOT = { exists = scope:tm_foothold_courtiers_only } }	# the caller asked that the people keep their faith
			set_county_faith = $FAITH$
		}
		holder ?= { save_scope_as = tm_foothold_lord }
	}
	if = {
		limit = {
			exists = scope:tm_foothold_lord
			scope:tm_foothold_lord = {
				any_held_title = { tier = tier_county count >= 2 }
				NOT = { capital_county ?= $COUNTY$ }
			}
			NOT = { exists = scope:tm_foothold_lord_only }	# the caller asked that the holder keep the county
		}
		# The bookmark's historical countess, if the caller saved her as scope:tm_foothold_preset; else a new woman.
		if = {
			limit = { exists = scope:tm_foothold_preset }
			scope:tm_foothold_preset = {
				save_scope_as = tm_foothold_count
				tm_register_seed_effect = { SEAT = $SEAT$ }
			}
			scope:tm_foothold_lord = { add_courtier = scope:tm_foothold_count }
		}
		else = {
			create_character = {
				gender = female
				age = { 24 34 }
				culture = $COUNTY$.culture
				faith = $FAITH$
				dynasty = generate
				employer = scope:tm_foothold_lord
				trait = education_stewardship_2
				save_scope_as = tm_foothold_count
				after_creation = { tm_register_seed_effect = { SEAT = $SEAT$ } }
			}
		}
		create_title_and_vassal_change = {
			type = granted
			save_scope_as = change
			add_claim_on_loss = no
		}
		$COUNTY$ = {
			change_title_holder = {
				holder = scope:tm_foothold_count
				change = scope:change
				government_base = scope:tm_foothold_lord
			}
			every_in_de_jure_hierarchy = {
				limit = { tier = tier_barony holder ?= scope:tm_foothold_lord }
				change_title_holder = {
					holder = scope:tm_foothold_count
					change = scope:change
				}
			}
		}
		resolve_title_and_vassal_change = scope:change
		scope:tm_foothold_count = { save_scope_as = tm_foothold_host }
		$COUNTY$ = { set_variable = { name = tm_seat_branch value = flag:vassal } }	# read by tools/census_report.py
	}
	else_if = {
		limit = { exists = scope:tm_foothold_lord }
		scope:tm_foothold_lord = { save_scope_as = tm_foothold_host }
		$COUNTY$ = { set_variable = { name = tm_seat_branch value = flag:lord } }
	}
	else = {
		$COUNTY$ = { set_variable = { name = tm_seat_branch value = flag:unheld } }
	}
	clear_saved_scope = tm_foothold_preset
	clear_saved_scope = tm_foothold_lord_only
	clear_saved_scope = tm_foothold_courtiers_only
}

# The northern faith's seat at game start (tm_on_game_start picks the county by date): a völva at the court that holds
# it, and two Marked courtiers. The holder is converted only if already of the faith's own religion (a pagan count),
# never a Christian one: in vanilla 1066 Gotland is a Catholic republic. $COUNTY$ the seat, $SEAT$ its tracking name.
tm_seed_baeddel_seat_effect = {
	tm_seed_foothold_effect = { COUNTY = $COUNTY$ FAITH = faith:tm_baeddel SEAT = $SEAT$ }
	scope:tm_foothold_host ?= { save_scope_as = tm_baeddel_count }
	$COUNTY$.holder ?= {
		if = {
			limit = { faith.religion = faith:tm_baeddel.religion }
			set_character_faith = faith:tm_baeddel
		}
	}
	if = {
		limit = { exists = scope:tm_baeddel_count }
		create_character = {
			after_creation = { tm_register_seed_effect = { SEAT = $SEAT$ } }
			gender = female
			age = { 40 60 }
			culture = scope:tm_baeddel_count.culture
			faith = faith:tm_baeddel
			employer = scope:tm_baeddel_count
			dynasty = none
			trait = tm_daughter_of_seidr
			trait = education_learning_3
		}
		tm_seed_marked_courtiers_effect = { HOST = scope:tm_baeddel_count FAITH = faith:tm_baeddel SEAT = $SEAT$ }
	}
	set_global_variable = { name = tm_baeddel_seat value = $COUNTY$ }	# the homecoming's seat (tm_yearly_growth)
}

# Badenya's seat at game start (tm_on_game_start picks the county by date): a daughter of the House at the court that
# holds it, and two Marked courtiers. After 1040 the holder keeps the county: in vanilla, Kirina's lord is Muslim
# from 1040 (al-Bakri says the king of Malal converted), Ghana's king holds Awkar, and in 1178 Kirina's lord is the
# challenge character Maghan Kon Fatta, whom the mod must leave alone. $COUNTY$ the seat, $SEAT$ its tracking name.
tm_seed_badenya_seat_effect = {
	if = {
		limit = { current_date >= 1040.1.1 }
