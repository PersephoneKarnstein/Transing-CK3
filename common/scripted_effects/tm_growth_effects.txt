# Adelphai Dorleyaiou — how the faiths grow (docs/faith_design.md §14).
# A 119-year observe run (867–986) left the four faiths with 81 living members between them: nothing drew anyone in,
# only rulers could start a rite, and the AI never adopted. These effects are the growth engine.

# A Marked character of another faith hears of the Sisters and may join one of the six. The call travels by contact:
# members at court, members nearby, the faith holding land in the realm, a House, a holy site. With none of these it is
# a rumour (1% a year). A larger faith has more members in more places, so it reaches more people; the ceiling is the
# number of Marked characters alive, since daughters bear no children. Scope: the character. Called yearly from
# tm_yearly_growth (tm_on_actions.txt).
tm_sisters_call_effect = {
	# Saved scopes outlive one character in the yearly loop: clear them, or the last character's faith carries over.
	clear_saved_scope = tm_call_faith
	save_scope_as = tm_call_seeker			# root is the pulse, not her
	# Which faith: one met at court first; then the faith of her own religion or region.
	if = {
		limit = {
			employer ?= {
				OR = {
					AND = { NOT = { this = scope:tm_call_seeker } rite ?= { rite_has_parameter = tm_sisters_faith } }
					any_courtier = { NOT = { this = scope:tm_call_seeker } rite ?= { rite_has_parameter = tm_sisters_faith } }
				}
			}
		}
		employer = {
			if = {
				limit = { NOT = { this = scope:tm_call_seeker } rite ?= { rite_has_parameter = tm_sisters_faith } }
				faith = { save_scope_as = tm_call_faith }
			}
			else = {
				random_courtier = {
					limit = { NOT = { this = scope:tm_call_seeker } rite ?= { rite_has_parameter = tm_sisters_faith } }
					faith = { save_scope_as = tm_call_faith }
				}
			}
		}
	}
	else_if = {
		limit = { rite ?= { rite_has_parameter = jewish_syncretic_recipient } }
		faith:tm_nehpakhot = { save_scope_as = tm_call_faith }
	}
	else_if = {
		limit = {
			location ?= {
				OR = {
					geographical_region = world_asia_minor
					geographical_region = world_europe_south_east
				}
			}
		}
		faith:transbian_maximalism = { save_scope_as = tm_call_faith }
	}
	else_if = {
		limit = {
			location ?= {
				OR = {
					geographical_region = world_europe_north
					geographical_region = world_europe_west_britannia
					geographical_region = custom_eastern_baltic
				}
			}
		}
		faith:tm_baeddel = { save_scope_as = tm_call_faith }
	}
	else_if = {
		limit = {
			location ?= {
				OR = {
					geographical_region = world_asia_china
					geographical_region = world_steppe_tarim
				}
			}
		}
		faith:tm_yutai = { save_scope_as = tm_call_faith }
	}
	else_if = {
		limit = { location ?= { geographical_region = world_india_deccan } }
		faith:tm_garland = { save_scope_as = tm_call_faith }
	}
	else_if = {
		limit = { location ?= { geographical_region = world_africa_west } }
		faith:tm_badenya = { save_scope_as = tm_call_faith }
	}
	if = {
		limit = { exists = scope:tm_call_faith }
		random = {
			chance = 1									# a rumour only
			modifier = {								# a member at her court, or her lord
				add = 8
				employer ?= {
					OR = {
						AND = { NOT = { this = scope:tm_call_seeker } faith = scope:tm_call_faith }
						any_courtier = { NOT = { this = scope:tm_call_seeker } faith = scope:tm_call_faith }
					}
				}
			}
			modifier = {								# three or more
				add = 4
				employer ?= { any_courtier = { count >= 3 NOT = { this = scope:tm_call_seeker } faith = scope:tm_call_faith } }
			}
			modifier = {								# a member living in her county, at another court
				add = 4
				location ?= {
					county ?= {
						any_county_province = {
							any_character_in_location = {
								NOT = { employer ?= scope:tm_call_seeker.employer }
								faith = scope:tm_call_faith
							}
						}
					}
				}
			}
			modifier = {								# the faith holds land in her realm
				add = 4
				top_liege ?= { any_realm_county = { faith = scope:tm_call_faith } }
			}
			modifier = {								# a Sisters' House in her county
				add = 5
				location ?= { county ?= { has_variable = tm_house } }
			}
			modifier = {								# a holy site of the faith in her duchy
				add = 3
				location ?= {
					county ?= {
						duchy ?= {
							any_de_jure_county = {
								any_county_province = { barony ?= { is_holy_site_of = scope:tm_call_faith } }
							}
						}
					}
				}
			}
			modifier = {								# a daughter of any faith at her court
				add = 3
				employer ?= { any_courtier = { tm_is_daughter_trigger = yes } }
			}
			modifier = {								# she has already declared her name
				add = 4
				has_variable = tm_declared
			}
			modifier = {								# a count turns slowly: her county's faith goes with her
				factor = 0.25
				is_ruler = yes
				highest_held_title_tier <= tier_county
			}
			modifier = {								# a duke, king or emperor more slowly still, and at a cost
				factor = 0.05
				highest_held_title_tier >= tier_duchy
			}
			set_character_faith = scope:tm_call_faith
			add_stress = -20
			if = {
				limit = { highest_held_title_tier >= tier_duchy }
				every_vassal = {
					add_opinion = { target = scope:tm_call_seeker modifier = tm_liege_joined_sisters_opinion }
				}
			}
		}
	}
}

# Offer the faith's rite to a Marked character or eunuch who is not a ruler: the AI weighs decisions only for rulers,
# so without this a courtier could never rise. Uses the same event chains as the decisions. Scope: the seeker.
tm_offer_courtier_rite_effect = {
	set_variable = { name = tm_rite_offered value = yes years = 3 }
	if = {
		limit = { rite ?= { rite_has_parameter = tm_seidr_gate } }
		tm_seidr_begin_effect = yes
	}
	else_if = {
		limit = { rite ?= { rite_has_parameter = tm_embryo_gate } }
		if = {
			limit = { NOT = { has_variable = tm_embryo_active } }
			trigger_event = tm_embryo.0001
		}
	}
	else_if = {
		limit = { rite ?= { rite_has_parameter = tm_turning_gate } }
		trigger_event = { id = tm_hipukh.0009 days = 1 }	# waits for Adar
	}
	else_if = {
		limit = { rite ?= { rite_has_parameter = tm_garland_gate } }
		trigger_event = { id = tm_garland.0009 days = 1 }	# waits for Mārkaḻi
	}
	else_if = {
		limit = { rite ?= { rite_has_parameter = tm_crossing_gate } }
		tm_crossing_begin_effect = yes
	}
	else_if = {
		limit = { rite ?= { rite_has_parameter = tm_knife_voluntary } }
		trigger_event = tm_knife.0001
	}
}

# Counts of each faith's living members, daughters and Marked, written to debug.log. Scope: none (global).
tm_census_log_effect = {
	save_scope_value_as = { name = tm_census_dorleyism value = { value = 0 every_living_character = { limit = { faith = faith:transbian_maximalism } add = 1 } } }
	save_scope_value_as = { name = tm_census_baeddel value = { value = 0 every_living_character = { limit = { faith = faith:tm_baeddel } add = 1 } } }
	save_scope_value_as = { name = tm_census_yutai value = { value = 0 every_living_character = { limit = { faith = faith:tm_yutai } add = 1 } } }
	save_scope_value_as = { name = tm_census_nehpakhot value = { value = 0 every_living_character = { limit = { faith = faith:tm_nehpakhot } add = 1 } } }
	save_scope_value_as = { name = tm_census_garland value = { value = 0 every_living_character = { limit = { faith = faith:tm_garland } add = 1 } } }
	save_scope_value_as = { name = tm_census_badenya value = { value = 0 every_living_character = { limit = { faith = faith:tm_badenya } add = 1 } } }
	save_scope_value_as = { name = tm_census_daughters value = { value = 0 every_living_character = { limit = { tm_is_daughter_trigger = yes } add = 1 } } }
	save_scope_value_as = { name = tm_census_marked value = { value = 0 every_living_character = { limit = { has_trait = tm_transgender } add = 1 } } }
	save_scope_value_as = { name = tm_census_counties value = { value = 0 every_county = { limit = { OR = { faith = faith:transbian_maximalism faith = faith:tm_baeddel faith = faith:tm_yutai faith = faith:tm_nehpakhot faith = faith:tm_garland faith = faith:tm_badenya } } add = 1 } } }
	# debug_log writes only when the game runs with -debug_mode, so the counts are also kept in the save (variables={ } near the top).
	set_global_variable = { name = tm_census_dorleyism value = scope:tm_census_dorleyism }
	set_global_variable = { name = tm_census_baeddel value = scope:tm_census_baeddel }
	set_global_variable = { name = tm_census_yutai value = scope:tm_census_yutai }
	set_global_variable = { name = tm_census_nehpakhot value = scope:tm_census_nehpakhot }
	set_global_variable = { name = tm_census_garland value = scope:tm_census_garland }
	set_global_variable = { name = tm_census_badenya value = scope:tm_census_badenya }
	set_global_variable = { name = tm_census_daughters value = scope:tm_census_daughters }
	set_global_variable = { name = tm_census_marked value = scope:tm_census_marked }
	set_global_variable = { name = tm_census_counties value = scope:tm_census_counties }
	set_global_variable = { name = tm_census_year value = current_year }
	debug_log = "tm_census_log"
	debug_log_scopes = yes
}

# Two Marked young men of the faith at a seat court, so the courtier rite has seekers from the start.
tm_seed_marked_courtiers_effect = {
	while = {
		count = 2
		create_character = {
			gender = male
			age = { 16 28 }
			culture = $HOST$.culture
			faith = $FAITH$
			employer = $HOST$
			dynasty = none
			trait = tm_transgender
		}
	}
}

# A seat county at game start. The county takes the faith. If its lord holds other counties and this is not their
# capital, the county goes to a member of the faith as the lord's vassal: a landed foothold whose own chaplain keeps it,
# and a count under a liege cannot march on a kingdom (a converted independent lord's line formed Scandinavia in one
# observe run, and held Mali in another). Otherwise the lord keeps it and the faith lives at their court.
# Saves scope:tm_foothold_host, the court the faith's seed characters should join.
# $COUNTY$ a county title, $FAITH$ a faith.
tm_seed_foothold_effect = {
	clear_saved_scope = tm_foothold_lord		# the effect runs once per seat: earlier seats' scopes must not carry over
	clear_saved_scope = tm_foothold_count
	clear_saved_scope = tm_foothold_host
	$COUNTY$ = {
		set_county_faith = $FAITH$
		holder ?= { save_scope_as = tm_foothold_lord }
	}
	if = {
		limit = {
			exists = scope:tm_foothold_lord
			scope:tm_foothold_lord = {
				any_held_title = { tier = tier_county count >= 2 }
				NOT = { capital_county ?= $COUNTY$ }
			}
		}
		create_character = {
			gender = female
			age = { 24 34 }
			culture = $COUNTY$.culture
			faith = $FAITH$
			dynasty = generate
			employer = scope:tm_foothold_lord
			trait = education_stewardship_2
			save_scope_as = tm_foothold_count
		}
		create_title_and_vassal_change = {
			type = granted
			save_scope_as = change
			add_claim_on_loss = no
		}
		$COUNTY$ = {
			change_title_holder = {
				holder = scope:tm_foothold_count
				change = scope:change
				government_base = scope:tm_foothold_lord
			}
			every_in_de_jure_hierarchy = {
				limit = { tier = tier_barony holder ?= scope:tm_foothold_lord }
				change_title_holder = {
					holder = scope:tm_foothold_count
					change = scope:change
				}
			}
		}
		resolve_title_and_vassal_change = scope:change
		scope:tm_foothold_count = { save_scope_as = tm_foothold_host }
	}
	else_if = {
		limit = { exists = scope:tm_foothold_lord }
		scope:tm_foothold_lord = { save_scope_as = tm_foothold_host }
	}
}
