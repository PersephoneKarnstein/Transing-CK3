# Transbian Maximalism — scripted effects
# DRAFT, untested in-game. Every effect, trigger and event target below was checked against the
# script_docs output for CK3 1.20.1 (docs/script_docs/*.log, generated 2026-10-14):
#   - there is NO effect that changes a living character's sex (only set_sexuality / set_pregnancy_gender), hence the re-embodiment route;
#   - `castrate` is not an effect: vanilla castration is just add_trait = eunuch_1 / beardless_eunuch (07_dlc_ep3_scripted_effects.txt);
#   - remove_character only removes someone from a travel plan, so the old body exits via `death`;
#   - set_sexuality takes a key (heterosexual/homosexual/bisexual/asexual), not a scope, so it is copied by branches;
#   - every_relation needs `type = <relation>`;
#   - set_relation_guardian / set_relation_ward exist for the sponsor mechanic.

# ---------------------------------------------------------------------------------------------
# tm_reembody_as_woman_effect
#
# Scope: the character completing transition (male-bodied in engine terms).
# Creates a female-bodied copy with the same face, moves traits, faith, culture, house, sexuality,
# resources, lifestyle XP and titles across, hands player control over if needed, and retires the
# old body quietly.
#
# Losses accepted by design (see docs/model_swap.md): memories, secrets, hooks, schemes, opinions,
# perks (no perk iterator exists; accepted loss per design lock). Children are NOT re-parented here; the
# faith's adoption mechanic handles that. Marriages ARE carried over (step 8).
# ---------------------------------------------------------------------------------------------
tm_reembody_as_woman_effect = {
	save_scope_as = tm_old_body
	debug_log = "TM reembody: start"
	debug_log_scopes = no

	# Values to carry over (triggers used as values; re-read on the old scope before the new one exists).
	save_scope_value_as = { name = tm_gold value = gold }
	save_scope_value_as = { name = tm_prestige value = prestige }
	save_scope_value_as = { name = tm_piety value = piety }
	save_scope_value_as = { name = tm_age value = age }
	save_scope_value_as = { name = tm_xp_dip value = diplomacy_lifestyle_xp }
	save_scope_value_as = { name = tm_xp_mar value = martial_lifestyle_xp }
	save_scope_value_as = { name = tm_xp_ste value = stewardship_lifestyle_xp }
	save_scope_value_as = { name = tm_xp_int value = intrigue_lifestyle_xp }
	save_scope_value_as = { name = tm_xp_lea value = learning_lifestyle_xp }

	# 1. New body. `gender = female` is the only lever the engine offers for the female mesh.
	#    Skills are copied explicitly; otherwise create_character rolls them.
	create_character = {
		gender = female
		age = scope:tm_age
		culture = scope:tm_old_body.culture
		faith = scope:tm_old_body.faith
		dynasty = none						# house is set afterwards (step 4); an argument from a lowborn old body would fail the creation
		location = scope:tm_old_body.location
		random_traits = no
		fertility = 0						# daughters of the knife are infertile; never let the new body roll female fertility
		save_scope_as = tm_new_body
	}

	debug_log = "TM reembody: body created"
	scope:tm_new_body = {
		debug_log_scopes = no
		# 2. Same face: copies the inheritable genes in the DNA string; hair/clothes regenerate from the female lists.
		copy_inheritable_appearance_from = scope:tm_old_body

		# 3. Traits. every_character_trait scopes to the trait, so add it to the new body via prev.
		scope:tm_old_body = {
			every_character_trait = {
				limit = {
					NOT = { this = trait:eunuch_1 }
					NOT = { this = trait:beardless_eunuch }
				}
				scope:tm_new_body = { add_trait = prev }
			}
		}
		# Always present on the new body, whatever the old one carried: can_have_children = no is what keeps
		# a re-embodied (engine-female) daughter from ever becoming pregnant.
		add_trait = tm_daughter_of_the_knife

		# 4. Identity.
		if = { limit = { exists = scope:tm_old_body.house } set_house = scope:tm_old_body.house }
		# CK3 stores orientation relative to the character's own sex, so it must flip with the body:
		# a man attracted to women (heterosexual) is a woman attracted to women (homosexual), and vice versa.
		if = { limit = { scope:tm_old_body = { has_sexuality = heterosexual } }	set_sexuality = homosexual }
		else_if = { limit = { scope:tm_old_body = { has_sexuality = homosexual } }	set_sexuality = heterosexual }
		else_if = { limit = { scope:tm_old_body = { has_sexuality = bisexual } }	set_sexuality = bisexual }
		else_if = { limit = { scope:tm_old_body = { has_sexuality = asexual } }	set_sexuality = asexual }

		# Parents carry over, so she is her mother's and father's daughter rather than nobody's.
		if = { limit = { exists = scope:tm_old_body.father } set_father = scope:tm_old_body.father }
		if = { limit = { exists = scope:tm_old_body.mother } set_mother = scope:tm_old_body.mother }

		# Skills, copied by value after creation (vanilla pattern: value = scope:actor.diplomacy).
		add_diplomacy_skill = { value = scope:tm_old_body.diplomacy subtract = diplomacy }
		add_martial_skill = { value = scope:tm_old_body.martial subtract = martial }
		add_stewardship_skill = { value = scope:tm_old_body.stewardship subtract = stewardship }
		add_intrigue_skill = { value = scope:tm_old_body.intrigue subtract = intrigue }
		add_learning_skill = { value = scope:tm_old_body.learning subtract = learning }
		add_prowess_skill = { value = scope:tm_old_body.prowess subtract = prowess }

		# 5. Resources and lifestyle experience.
		add_gold = scope:tm_gold
		add_prestige = scope:tm_prestige
		add_piety = scope:tm_piety
		add_diplomacy_lifestyle_xp = scope:tm_xp_dip
		add_martial_lifestyle_xp = scope:tm_xp_mar
		add_stewardship_lifestyle_xp = scope:tm_xp_ste
		add_intrigue_lifestyle_xp = scope:tm_xp_int
		add_learning_lifestyle_xp = scope:tm_xp_lea
		# Perks are not copied (no perk iterator); accepted loss per design lock 2026-10-14.

		# 6. Court placement for the unlanded case; the landed case moves with the titles below.
		if = {
			limit = { scope:tm_old_body = { is_landed = no  exists = employer } }
			set_employer = scope:tm_old_body.employer
		}
	}

	# 7. Titles, vassals and capital (batch change; vassals follow).
	#    Read capital, primary title and liege BEFORE anything moves (after the transfer the old body has none).
	#    Hand over county tier and up only: a county carries its baronies, and moving a capital barony on its
	#    own is an engine error.
	if = {
		limit = { is_landed = yes }
		capital_county ?= { save_scope_as = tm_old_capital }
		primary_title ?= { save_scope_as = tm_old_primary }
		if = {
			limit = {
				exists = liege
				NOT = { liege = scope:tm_old_body }
			}
			liege = { save_scope_as = tm_old_liege }
		}
		create_title_and_vassal_change = {
			type = granted
			save_scope_as = tm_change
			add_claim_on_loss = no
		}
		every_held_title = {
			limit = { tier >= tier_county }
			change_title_holder_include_vassals = {
				holder = scope:tm_new_body
				change = scope:tm_change
			}
		}
		if = {
			limit = { exists = scope:tm_old_liege }
			scope:tm_new_body = {
				change_liege = {
					liege = scope:tm_old_liege
					change = scope:tm_change
				}
			}
		}
		resolve_title_and_vassal_change = scope:tm_change
		scope:tm_new_body = {
			if = {
				limit = { exists = scope:tm_old_capital }
				set_realm_capital = scope:tm_old_capital
			}
			if = {
				limit = { exists = scope:tm_old_primary }
				set_primary_title_to = scope:tm_old_primary
			}
		}
	}

	# 8. Marriages carry over (design lock, user 2026-10-14). The spouse divorces the old register and
	#    marries the new body (vanilla idiom: spouse-side `divorce = <old>`). A husband becomes an
	#    opposite-sex marriage; a wife becomes a same-sex one, which the faith's doctrines accept.
	#    TODO in-game test: confirm `marry` succeeds for the same-sex case.
	scope:tm_old_body = {
		every_spouse = {
			divorce = scope:tm_old_body
			scope:tm_new_body = { marry = prev }
		}
	}
	# Other relations worth keeping.
	scope:tm_old_body = {
		every_relation = { type = friend	scope:tm_new_body = { set_relation_friend = prev } }
		every_relation = { type = best_friend	scope:tm_new_body = { set_relation_best_friend = prev } }
		every_relation = { type = rival	scope:tm_new_body = { set_relation_rival = prev } }
		every_relation = { type = lover	scope:tm_new_body = { set_relation_lover = prev } }
		# Sponsor link for the Dorley chain, if present.
		every_relation = { type = guardian	scope:tm_new_body = { set_relation_ward = prev } }
	}

	# 9. Player hand-over (scope must be the player; target must not be).
	if = {
		limit = { is_ai = no }
		set_player_character = scope:tm_new_body
	}

	debug_log = "TM reembody: transfer done, retiring old body"
	# 10. Close the old register. death_vanished is a vanilla death reason with no corpse flavour.
	# Players among the close family hear that she lives (the engine's death notice cannot be suppressed).
	scope:tm_old_body = {
		every_close_family_member = {
			limit = {
				is_ai = no
				NOT = { this = scope:tm_new_body }
			}
			trigger_event = { id = tm_knife.0003 days = 1 }
		}
		death = { death_reason = death_tm_reborn }
	}
}
