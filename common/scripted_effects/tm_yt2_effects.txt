# Crusader Dolls — the palace faction, stage 2 (tm_yt2_events.txt; design note docs/wave3/yutai.md)

# The standing value: tm_yt2_balance, a number from -3 to +3 kept on the throne's primary title (so it outlives an
# emperor). Above zero, the inner offices (the Northern Office) have the throne's ear; below zero, the scholar-officials
# (the Southern Office) do. The names are the period's own (So 2000, p. 143). After 960 under a celestial throne the
# inner offices cannot rise above +2: Song eunuchs had no army of their own and rose and fell with their patrons
# (So 2000, pp. 577-579).

# Make sure the seat has a balance. Scope: the seat (a landed title).
tm_yt2_seat_init_effect = {
	if = {
		limit = { NOT = { has_variable = tm_yt2_balance } }
		set_variable = { name = tm_yt2_balance value = 0 }
	}
}

# Keep the balance inside its era's bounds. Scope: the seat.
tm_yt2_clamp_effect = {
	if = {
		limit = {
			current_date >= 960.1.1
			holder ?= { has_government = celestial_government }
		}
		clamp_variable = { name = tm_yt2_balance min = -3 max = 2 }
	}
	else = {
		clamp_variable = { name = tm_yt2_balance min = -3 max = 3 }
	}
}

# The inner offices gain ground. Scope: the seat.
tm_yt2_north_effect = {
	custom_tooltip = tm_yt2_north_tt
	hidden_effect = {
		tm_yt2_seat_init_effect = yes
		change_variable = { name = tm_yt2_balance add = $AMOUNT$ }
		tm_yt2_clamp_effect = yes
	}
}

# The scholar-officials gain ground. Scope: the seat.
tm_yt2_south_effect = {
	custom_tooltip = tm_yt2_south_tt
	hidden_effect = {
		tm_yt2_seat_init_effect = yes
		change_variable = { name = tm_yt2_balance add = -1 }
		tm_yt2_clamp_effect = yes
	}
}

# The inner offices spend their standing on the society's behalf. Scope: the seat.
tm_yt2_spend_effect = {
	custom_tooltip = tm_yt2_spend_tt
	hidden_effect = {
		tm_yt2_seat_init_effect = yes
		change_variable = { name = tm_yt2_balance add = -1 }
		tm_yt2_clamp_effect = yes
	}
}

# Once a year for each daughter of the Way under a celestial or meritocratic throne, from tm_palace_yearly.
# Scope: the daughter. Needs scope:tm_palace_throne and scope:tm_palace_daughter; saves scope:tm_palace_seat.
tm_yt2_yearly_effect = {
	scope:tm_palace_throne = { primary_title = { save_scope_as = tm_palace_seat } }
	scope:tm_palace_seat = {
		tm_yt2_seat_init_effect = yes
		# Drift toward even, once a year per seat: an office is the throne's to take back, and the officials'
		# gains fade as well.
		if = {
			limit = { NOT = { has_variable = tm_yt2_drifted } }
			set_variable = { name = tm_yt2_drifted value = yes days = 360 }
			if = {
				limit = { var:tm_yt2_balance > 0 }
				random = { chance = 30 change_variable = { name = tm_yt2_balance add = -1 } }
			}
			else_if = {
				limit = { var:tm_yt2_balance < 0 }
				random = { chance = 30 change_variable = { name = tm_yt2_balance add = 1 } }
			}
			tm_yt2_clamp_effect = yes
		}
	}
	# A daughter in office. A censor of the Way holds the memorials on her desk; otherwise the Southern Office writes.
	if = {
		limit = {
			has_variable = tm_palace_office
			NOT = { has_variable = tm_yt2_memorial_cd }
		}
		if = {
			limit = {
				scope:tm_palace_throne = { tm_yt2_way_censor_trigger = yes }
				title:e_minister_censor.holder ?= {
					NOT = { this = scope:tm_palace_daughter }
					NOT = { has_variable = tm_yt2_censor_cd }
					tm_flavour_ready_trigger = yes
				}
			}
			random = {
				chance = 10
				set_variable = { name = tm_yt2_memorial_cd value = yes years = 5 }
				title:e_minister_censor.holder = {
					set_variable = { name = tm_yt2_censor_cd value = yes years = 10 }
					tm_flavour_spent_effect = yes
					trigger_event = { id = tm_yt2.0003 days = { 1 300 } }
				}
			}
		}
		else_if = {
			limit = {
				tm_flavour_ready_trigger = yes
				NOT = { has_variable = tm_yt2_one_body }
				NOT = { scope:tm_palace_throne = { tm_yt2_way_censor_trigger = yes } }
				scope:tm_palace_seat = {
					OR = {
						var:tm_yt2_balance >= 2
						var:tm_yt2_balance <= -1
					}
				}
			}
			random = {
				chance = 3
				# The scholar-officials press harder the more ground they hold.
				modifier = { add = 3 scope:tm_palace_seat = { var:tm_yt2_balance <= -1 } }
				modifier = { add = 3 scope:tm_palace_seat = { var:tm_yt2_balance <= -2 } }
				modifier = { add = 3 scope:tm_palace_seat = { var:tm_yt2_balance <= -3 } }
				set_variable = { name = tm_yt2_memorial_cd value = yes years = 5 }
				tm_flavour_spent_effect = yes
				trigger_event = { id = tm_palace.0002 days = { 1 300 } }
			}
		}
	}
	# Song: a clerk of the Court of Imperial Ceremonies asks after the society's adopted daughters (from 971; So 2000,
	# p. 200). A daughter who keeps a House, under a celestial throne.
	if = {
		limit = {
			current_date >= 971.1.1
			scope:tm_palace_throne = { has_government = celestial_government }
			tm_holds_sisters_house_trigger = yes
			NOT = { has_variable = tm_yt2_clerk_cd }
			tm_flavour_ready_trigger = yes
		}
		random = {
			chance = 2
			set_variable = { name = tm_yt2_clerk_cd value = yes years = 20 }
			tm_flavour_spent_effect = yes
			trigger_event = { id = tm_yt2.0002 days = { 1 300 } }
		}
	}
	# Wave 4: after the purge of 903 (tm_palace.0003) she is asked once whether to hide the name (tm_yt3.0001,
	# de la Vaissiere 2005, pp. 220-222). The purge flag is set after this effect has run for the year, so the
	# question comes the year after the purge. A chain under way: no flavour budget.
	if = {
		limit = {
			has_character_flag = tm_palace_purge_seen
			NOT = { has_character_flag = tm_yt3_purge_asked }
			current_date < 910.1.1
			tm_yt3_hide_name_eligible_trigger = yes
		}
		add_character_flag = tm_yt3_purge_asked
		trigger_event = { id = tm_yt3.0001 days = { 30 180 } }
	}
}
