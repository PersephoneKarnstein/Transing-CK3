# Crusader Dolls — Ha-Nehpakhot, the bailiff's variants and the households of the street (docs/wave6/ha-nehpakhot.md).

# Run-wide counts for the census (tools/census_report.py): households sent for, falls heard.
tm_hn4_tally_effect = {
	if = {
		limit = { NOT = { has_global_variable = tm_hn4_n_$WHAT$ } }
		set_global_variable = { name = tm_hn4_n_$WHAT$ value = 0 }
	}
	change_global_variable = { name = tm_hn4_n_$WHAT$ add = 1 }
}

# Send for a household of the street. A Jewish household comes to her court with the cost of the road paid: a man, his
# wife and their child, Rabbanite Jews, not members of the sect. The model is the Arles contract of 1422, which paid a
# man to bring his wife and child from Barcelona (Stouff, "Les juifs d'Arles" §16). It is service and settlement, not
# the purchase of protection. At her court the kahal may take them in later (tm_on_actions.txt, the kahal's 4%).
# Where they come from: Barcelona, or Arles for a ruler in the Crown of Aragon; a ruler in the Maghreb or the Sahel
# sends to a Jewish quarter of the coast, and the household is of her own land's culture, as the refugees of wave 3 are.
# Scope: the ruler.
tm_hn4_send_for_household_effect = {
	save_scope_as = tm_hn4_host
	if = {
		limit = { tm_hn3_capital_in_west_trigger = yes }
		capital_county.culture = { save_scope_as = tm_hn4_household_culture }
	}
	else_if = {
		limit = { capital_county ?= { tm_hn4_county_in_aragon_trigger = yes } }
		culture:occitan = { save_scope_as = tm_hn4_household_culture }		# from the street at Arles
	}
	else = {
		culture:sephardi = { save_scope_as = tm_hn4_household_culture }	# from the call at Barcelona
	}
	create_character = {
		gender = male
		age = { 24 40 }
		culture = scope:tm_hn4_household_culture
		faith = faith:rabbinic_faith
		dynasty = generate
		employer = scope:tm_hn4_host
		save_scope_as = tm_hn4_householder
	}
	create_character = {
		gender = female
		age = { 20 34 }
		culture = scope:tm_hn4_household_culture
		faith = faith:rabbinic_faith
		dynasty = none
		employer = scope:tm_hn4_host
		save_scope_as = tm_hn4_householder_wife
	}
	scope:tm_hn4_householder = { marry = scope:tm_hn4_householder_wife }
	create_character = {
		gender_female_chance = 50
		age = { 2 9 }
		culture = scope:tm_hn4_household_culture
		faith = faith:rabbinic_faith
		father = scope:tm_hn4_householder
		mother = scope:tm_hn4_householder_wife
		dynasty = inherit
		employer = scope:tm_hn4_host
		save_scope_as = tm_hn4_householder_child
	}
	scope:tm_hn4_householder = {
		set_variable = tm_hn4_household
		add_opinion = { target = scope:tm_hn4_host modifier = tm_hn4_household_opinion }
	}
	scope:tm_hn4_householder_wife = {
		set_variable = tm_hn4_household
		add_opinion = { target = scope:tm_hn4_host modifier = tm_hn4_household_opinion }
	}
	scope:tm_hn4_householder_child = { set_variable = tm_hn4_household }
	tm_hn4_tally_effect = { WHAT = households }
}

# The lamp relit. When the sect is under 40 and no house of it stands (no ruler of it, and no court where three of it
# live), or it is under 12, women of the lamp come back to the Jewish street of the Venaissin from another town of the sect: an elder, a physician's household
# and a Marked youth. The Comtat Venaissin, a papal land from 1274, kept its Jews when France (1306) and Provence
# (1501) expelled theirs. With Roads to Power the elder keeps a scholars' camp, as Bonadonna did from 867; otherwise
# all of them come to the court of the county's lord, as the seed does at a later start. Called from tm_yearly_growth
# (tm_on_actions.txt) by the hidden event tm_hn4.0010, 4% a year, then not again for 40 years. Scope: the lord of the
# Venaissin.
tm_hn4_relight_effect = {
	set_global_variable = { name = tm_hn4_relit_recently value = yes years = 40 }
	save_scope_as = tm_hn4_street_lord
	create_character = {
		gender = female
		age = { 45 60 }
		culture = title:c_venaissin.culture
		faith = faith:tm_nehpakhot
		dynasty = none
		location = title:c_venaissin.title_province
		trait = tm_daughter_of_the_turning
		trait = education_learning_3
		trait = patient
		learning = 14
		save_scope_as = tm_hn4_elder
	}
	if = {
		limit = {
			has_dlc_feature = roads_to_power
			exists = scope:tm_hn4_street_lord
		}
		scope:tm_hn4_elder = {
			scope:tm_hn4_street_lord = { save_scope_as = host }	# vanilla's camp-naming reads it
			create_landless_adventurer_title_effect = {
				REASON = flag:voluntary
				FLAVOR_CHAR = scope:tm_hn4_street_lord
			}
			if = {
				limit = { is_ruler = yes }
				add_realm_law_skip_effects = camp_purpose_scholars
			}
		}
	}
	if = {
		limit = { scope:tm_hn4_elder = { is_ruler = yes } }
		scope:tm_hn4_elder = { save_scope_as = tm_hn4_house }
	}
	else_if = {
		limit = { exists = scope:tm_hn4_street_lord }
		scope:tm_hn4_street_lord = {
			save_scope_as = tm_hn4_house
			add_courtier = scope:tm_hn4_elder
		}
	}
	if = {
		limit = { exists = scope:tm_hn4_house }
		# A physician's household of the sect; their children are born into it.
		create_character = {
			gender = male
			age = { 26 36 }
			culture = title:c_venaissin.culture
			faith = faith:tm_nehpakhot
			dynasty = generate
			employer = scope:tm_hn4_house
			trait = lifestyle_physician
			trait = education_learning_2
			save_scope_as = tm_hn4_relit_father
		}
		create_character = {
			gender = female
			age = { 20 30 }
			culture = title:c_venaissin.culture
			faith = faith:tm_nehpakhot
			dynasty = none
			employer = scope:tm_hn4_house
			save_scope_as = tm_hn4_relit_mother
		}
		scope:tm_hn4_relit_father = { marry = scope:tm_hn4_relit_mother }
		while = {
			count = 2
			create_character = {
				age = { 3 10 }
				gender_female_chance = 50
				faith = faith:tm_nehpakhot
				father = scope:tm_hn4_relit_father
				mother = scope:tm_hn4_relit_mother
				dynasty = inherit
				employer = scope:tm_hn4_house
				save_temporary_scope_as = tm_hn4_relit_child
			}
			scope:tm_hn4_relit_child = {
				if = {
					limit = { is_female = yes }
					set_relation_guardian = scope:tm_hn4_relit_mother
				}
				else = {
					set_relation_guardian = scope:tm_hn4_relit_father
				}
			}
		}
		# A Marked young man who came with them to be taken in.
		create_character = {
			gender = male
			age = { 16 24 }
			culture = title:c_venaissin.culture
			faith = faith:tm_nehpakhot
			dynasty = none
			employer = scope:tm_hn4_house
			trait = tm_transgender
			trait = education_learning_1
		}
		scope:tm_hn4_street_lord ?= {
			if = {
				limit = { is_ai = no }
				send_interface_toast = {
					type = event_toast_effect_neutral
					title = tm_hn4_relit_toast
					desc = tm_hn4_relit_toast_desc
					left_icon = scope:tm_hn4_elder
				}
			}
		}
	}
	tm_hn4_tally_effect = { WHAT = relit }
}
