# Crusader Dolls — raising a daughter. Scope: the character being raised.
# Every faith's gate ends here; only the trait and the naming event differ.
#   TRAIT  = tm_daughter_of_the_knife / tm_daughter_of_seidr / tm_daughter_of_the_embryo / tm_daughter_of_the_turning
#   RENAME = the faith's naming event (tm_knife.0002 / tm_seidr.0002 / tm_embryo.0002)
tm_rise_daughter_effect = {
	if = {
		limit = { has_trait = tm_novice }
		remove_trait = tm_novice
	}
	# A eunuch who seeks womanhood is a sister (Bhatt, faith_design.md §12): her eunuch trait gives way.
	if = { limit = { has_trait = eunuch_1 } remove_trait = eunuch_1 }
	if = { limit = { has_trait = beardless_eunuch } remove_trait = beardless_eunuch }
	# Her sworn sisters stood beside her at the gate (round 2, T21/T34). Listed now: the relation does not pass to a new
	# body (tm_reembody_as_woman_effect copies friends, not tm_sister).
	every_relation = {
		type = tm_sister
		limit = { is_alive = yes }
		add_to_list = tm_gate_sisters
	}
	add_trait = $TRAIT$
	add_piety = 50
	tm_track_effect = { FAITH = faith WHAT = rises }
	if = {
		limit = {
			OR = {
				has_trait = tm_transgender
				has_variable = tm_declared		# named before witnesses: a sister before the knife (rule 8)
			}
		}
		add_stress = -100				# relief: the rite answers what the Mother marked
	}
	# No one else pays in stress for rising either (rule 22, round 2: no pain as proof).
	# A sister already named before witnesses (tm_declare_decision) keeps that name; the gate changes her
	# body, not her name. Otherwise New Personal History fires on whichever body survives the rite.
	if = {
		limit = { has_character_modifier = tm_declared_modifier }
		remove_character_modifier = tm_declared_modifier
	}
	if = {
		limit = { has_game_rule = tm_reembodiment_on }
		# Read before the old body dies: a dead character holds no variables.
		clear_saved_scope = tm_was_declared
		if = {
			limit = { has_variable = tm_declared }
			save_scope_value_as = { name = tm_was_declared value = yes }
		}
		tm_reembody_as_woman_effect = yes
		if = {
			limit = { exists = scope:tm_was_declared }
			scope:tm_new_body = {
				change_first_name = { template_character = scope:tm_old_body }
				set_variable = tm_declared
			}
		}
		else = {
			scope:tm_new_body = { trigger_event = { id = $RENAME$ days = 1 } }
		}
	}
	else_if = {
		limit = { NOT = { has_variable = tm_declared } }
		trigger_event = { id = $RENAME$ days = 1 }
	}
	# The old likeness and a name of her own come after the rite itself (tm_life.0033, 0034; user 2026-10-21),
	# on whichever body survives it.
	if = {
		limit = {
			has_game_rule = tm_reembodiment_on
			exists = scope:tm_new_body
		}
		scope:tm_new_body = {
			tm_life_after_rite_effect = yes
			tm_after_gate_body_effect = yes
		}
	}
	else = {
		tm_life_after_rite_effect = yes
		tm_after_gate_body_effect = yes
	}
}

# After any gate, on whichever body survives the rite (round 2). Scope: the new daughter.
tm_after_gate_body_effect = {
	# T22, At Home in Her Body (research/book_heaney_feminism_against_cisness.md §5 row 3b): a small lasting reward
	# that is hers, not the faith's.
	if = {
		limit = { NOT = { has_character_modifier = tm_at_home_in_her_body_modifier } }
		add_character_modifier = { modifier = tm_at_home_in_her_body_modifier }
	}
	# T27: newly risen, for Elder in Years, Youngest in the House (tm_life.0050).
	set_variable = { name = tm_newly_risen value = yes years = 2 }
	# T21/T34: a sworn sister counts as her sponsor, and has her own small reward.
	save_scope_as = tm_gate_daughter
	every_in_list = {
		list = tm_gate_sisters
		tm_sponsor_reward_effect = { DAUGHTER = scope:tm_gate_daughter }
		remove_from_list = tm_gate_sisters		# cleared, as vanilla does, in case several rise in one effect
	}
}

# The sponsor's own reward when her daughter rises (round 2, T34; research/book_hornsby_stone_bible_trouble.md row 1):
# her own turning as much as the daughter's. Scope: the sponsor. DAUGHTER = the new daughter.
tm_sponsor_reward_effect = {
	add_piety = 25
	add_opinion = { target = $DAUGHTER$ modifier = tm_through_the_gate_opinion }
	$DAUGHTER$ = { add_opinion = { target = prev modifier = tm_through_the_gate_opinion } }
	send_interface_toast = {
		type = event_toast_effect_good
		title = tm_sponsor_reward_toast
	}
}

# Start the seiðr rite for the seeker (scope). Finds a völva at the host's court, or one comes on the road.
tm_seidr_begin_effect = {
	if = {
		limit = { is_ruler = yes }
		save_scope_as = tm_host
	}
	else_if = {
		limit = { exists = employer }
		employer = { save_scope_as = tm_host }
	}
	else = {
		save_scope_as = tm_host
	}
	# The Völva of the Hall sings it when the court has one (tm_voice_court_positions.txt)
	if = {
		limit = {
			scope:tm_host = {
				any_court_position_holder = {
					type = tm_voice_volva_court_position
					NOT = { this = root }
					is_imprisoned = no
				}
			}
		}
		scope:tm_host = {
			random_court_position_holder = {
				type = tm_voice_volva_court_position
				limit = {
					NOT = { this = root }
					is_imprisoned = no
				}
				save_scope_as = tm_volva
			}
		}
	}
	else_if = {
		limit = {
			scope:tm_host = {
				any_courtier = {
					NOT = { this = root }
					has_trait = tm_daughter_of_seidr
					is_adult = yes
					is_imprisoned = no
				}
			}
		}
		scope:tm_host = {
			random_courtier = {
				limit = {
					NOT = { this = root }
					has_trait = tm_daughter_of_seidr
					is_adult = yes
					is_imprisoned = no
				}
				save_scope_as = tm_volva
			}
		}
	}
	else = {
		tm_create_travelling_volva_effect = { HOST = scope:tm_host }
	}
	trigger_event = tm_seidr.0010
}

# Niddui needs the lord's licence: a Jewish community could ban only by its Christian lord's leave, granted for a
# term (Arles 1405, Shatzmiller p. 87; a rabbi tried for an unlicensed ban, 1310, p. 89). A lord who has taken the
# daughter's people under protection (tm_hipukh.0030) refuses it. Scope: the daughter.
tm_turning_niddui_effect = {
	if = {
		limit = {
			location ?= { county ?= { holder ?= { has_character_modifier = tm_translation_patron_modifier } } }
		}
		send_interface_toast = {
			type = event_toast_effect_good
			title = tm_turning_niddui_refused
		}
	}
	else = {
		add_character_modifier = { modifier = tm_niddui_modifier years = 3 }
	}
}

# Begin the pāvai vow (scope: the seeker). The companions are the women of the host's court, and an elder daughter
# of the Garland keeps the vow with them. tm_garland_events.txt.
tm_pavai_begin_effect = {
	if = {
		limit = { is_ruler = yes }
		save_scope_as = tm_host
	}
	else_if = {
		limit = { exists = employer }
		employer = { save_scope_as = tm_host }
	}
	else = {
		save_scope_as = tm_host
	}
	# The Eldest of the Street keeps the vow when the court has one (tm_voice_court_positions.txt)
	if = {
		limit = {
			scope:tm_host = {
				any_court_position_holder = {
					type = tm_voice_street_eldest_court_position
					NOT = { this = root }
					is_imprisoned = no
				}
			}
		}
		scope:tm_host = {
			random_court_position_holder = {
				type = tm_voice_street_eldest_court_position
				limit = {
					NOT = { this = root }
					is_imprisoned = no
				}
				save_scope_as = tm_garland_elder
			}
		}
	}
	else_if = {
		limit = {
			scope:tm_host = {
				any_courtier = {
					NOT = { this = root }
					has_trait = tm_daughter_of_the_garland
					is_adult = yes
				}
			}
		}
		scope:tm_host = {
			random_courtier = {
				limit = {
					NOT = { this = root }
					has_trait = tm_daughter_of_the_garland
					is_adult = yes
				}
				weight = {
					base = 1
					modifier = { add = 9 has_character_modifier = tm_ml2_staff_modifier }	# one who holds the staff (tm_ml2) is asked first
				}
				save_scope_as = tm_garland_elder
			}
		}
	}
	else = {
		create_character = {
			gender = female
			age = { 40 65 }
			culture = scope:tm_host.culture
			faith = faith:tm_garland
			dynasty = none
			location = scope:tm_host.location		# an unlanded seeker has no capital
			trait = tm_daughter_of_the_garland
			trait = education_diplomacy_3
			save_scope_as = tm_garland_elder
		}
	}
	# Companions: up to five women of the court, plus the elder.
	set_variable = { name = tm_pavai_companions value = 1 }
	scope:tm_host = {
		every_courtier = {
			limit = {
				is_female = yes
				is_adult = yes
				NOT = { this = root }
			}
			if = {
				limit = { root.var:tm_pavai_companions < 6 }
				root = { change_variable = { name = tm_pavai_companions add = 1 } }
			}
		}
	}
	add_character_modifier = { modifier = tm_pavai_vow_modifier days = 60 }
	trigger_event = tm_garland.0010
}

# Scope: the seeker. Ends the vow's temporary state.
tm_pavai_clear_effect = {
	remove_variable = tm_pavai_companions
	remove_variable = tm_pavai_garland_worn
	remove_variable = tm_pavai_saiva
	remove_variable = tm_pavai_vaishnava
	remove_character_modifier = tm_pavai_vow_modifier
}
