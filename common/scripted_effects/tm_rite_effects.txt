# Adelphai Dorleyaiou — raising a daughter. Scope: the character being raised.
# Every faith's gate ends here; only the trait and the naming event differ.
#   TRAIT  = tm_daughter_of_the_knife / tm_daughter_of_seidr / tm_daughter_of_the_embryo
#   RENAME = the faith's naming event (tm_knife.0002 / tm_seidr.0002 / tm_embryo.0002)
tm_rise_daughter_effect = {
	if = {
		limit = { has_trait = tm_novice }
		remove_trait = tm_novice
	}
	# A eunuch who seeks womanhood is a sister (Bhatt, faith_design.md §12): her eunuch trait gives way.
	if = { limit = { has_trait = eunuch_1 } remove_trait = eunuch_1 }
	if = { limit = { has_trait = beardless_eunuch } remove_trait = beardless_eunuch }
	add_trait = $TRAIT$
	add_piety = 50
	if = {
		limit = { has_trait = tm_transgender }
		add_stress = -100				# relief: the rite answers what the Mother marked
	}
	else = {
		add_stress = 50
	}
	# A sister already named before witnesses (tm_declare_decision) keeps that name; the gate changes her
	# body, not her name. Otherwise New Personal History fires on whichever body survives the rite.
	if = {
		limit = { has_character_modifier = tm_declared_modifier }
		remove_character_modifier = tm_declared_modifier
	}
	if = {
		limit = { has_game_rule = tm_reembodiment_on }
		tm_reembody_as_woman_effect = yes
		if = {
			limit = { scope:tm_old_body = { has_variable = tm_declared } }
			scope:tm_new_body = {
				change_first_name = { template_character = scope:tm_old_body }
				set_variable = tm_declared
			}
		}
		else = {
			scope:tm_new_body = { trigger_event = { id = $RENAME$ days = 1 } }
		}
	}
	else_if = {
		limit = { NOT = { has_variable = tm_declared } }
		trigger_event = { id = $RENAME$ days = 1 }
	}
}

# The knife itself (Dorleyism).
tm_take_the_knife_effect = {
	tm_rise_daughter_effect = { TRAIT = tm_daughter_of_the_knife RENAME = tm_knife.0002 }
}

# Raise by whatever gate the root ruler's rite keeps (arrivals). Scope: the newcomer. ROOT_RULER = whose rite.
tm_rise_by_rite_effect = {
	if = {
		limit = { $ROOT_RULER$.rite ?= { rite_has_parameter = tm_seidr_gate } }
		tm_rise_daughter_effect = { TRAIT = tm_daughter_of_seidr RENAME = tm_seidr.0002 }
	}
	else_if = {
		limit = { $ROOT_RULER$.rite ?= { rite_has_parameter = tm_embryo_gate } }
		tm_rise_daughter_effect = { TRAIT = tm_daughter_of_the_embryo RENAME = tm_embryo.0002 }
	}
	else = {
		tm_take_the_knife_effect = yes
	}
}

# A trial of the jade embryo was interrupted from outside. Nothing is lost: the gestation waits and the
# same trial comes round again (Serano: delay, never a reset). Scope: the adept. EVENT = the trial to repeat.
tm_embryo_delayed_effect = {
	add_stress = 25
	remove_character_modifier = tm_gestating_modifier
	add_character_modifier = { modifier = tm_gestating_modifier days = 300 }
	trigger_event = { id = $EVENT$ days = { 60 90 } }
}

# She sets the work down by her own choice. Scope: the adept.
tm_embryo_set_down_effect = {
	remove_variable = tm_embryo_active
	remove_character_modifier = tm_gestating_modifier
}

# The daughter trait of ROOT_RULER's rite, with no ceremony (characters created already risen).
tm_add_rite_daughter_trait_effect = {
	if = {
		limit = { $ROOT_RULER$.rite ?= { rite_has_parameter = tm_seidr_gate } }
		add_trait = tm_daughter_of_seidr
	}
	else_if = {
		limit = { $ROOT_RULER$.rite ?= { rite_has_parameter = tm_embryo_gate } }
		add_trait = tm_daughter_of_the_embryo
	}
	else = {
		add_trait = tm_daughter_of_the_knife
	}
}
