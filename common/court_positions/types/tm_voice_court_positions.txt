# Crusader Dolls — each faith's own office, held only by a daughter of that faith (docs/wave5/offices.md).
# "We're building trans-women religions, not just trans-affirming ones" (user, wave 5): cis women stay members and
# sisters; the faith's own voice belongs to its daughters. Template: vanilla court_gardener_court_position and
# court_physician_court_position (00_court_positions.txt; any ruler's court, landless adventurers included).
# Salary is piety (a religious office; _court_positions.info allows piety), so no gold changes hands, which keeps
# the Elder of the Lamp clear of money (Ha-Nehpakhot guardrails).
# AI: hires whenever a valid daughter is at court. Gold-based vanilla terms (scope:base_value, debt) are left out
# because the office costs no gold. The candidate score follows vanilla (rival -1000; holding another office -150
# when hiring only, so no hire/fire loop).

# Companion of the Mothers: the koinonoi beside the Mother of Mothers (research/byz_trevett_montanism.md). Only at
# her court. A seat here names her a Companion (tm_companion_modifier, the cap of seven kept); the Companions back a
# daughter for the Mother's succession (tm_voice_effects.txt).
tm_voice_companion_court_position = {
	sort_order = 276
	max_available_positions = 3
	skill = learning

	court_position_asset = {
		animation = admiration
		background = "gfx/interface/illustrations/event_scenes/church.dds"
	}

	opinion = {
		value = regular_court_position_opinion
	}

	aptitude_level_breakpoints = { 20 40 60 80 }

	aptitude = {
		value = 1
		add = {
			value = learning
			multiply = 2.5
			max = 50
			desc = court_position_skill_learning
		}
		add = {
			value = diplomacy
			multiply = 1
			max = 20
			desc = court_position_skill_diplomacy
		}
		add = general_aptitude_bonus
	}

	is_shown = {
		faith = faith:transbian_maximalism
		has_title = title:d_tm_mother_house
	}

	valid_position = {
		faith = faith:transbian_maximalism
		has_title = title:d_tm_mother_house
	}

	is_shown_character = {
		scope:employee = {
			court_position_employee_shown_trigger = yes
		}
	}

	valid_character = {
		scope:employee = {
			base_court_position_validity_trigger = { EMPLOYER = scope:liege }
			tm_voice_office_holder_trigger = { FAITH = faith:transbian_maximalism }
		}
	}

	revoke_cost = {
		prestige = {
			value = minor_court_position_prestige_revoke_cost
		}
	}

	salary = {
		piety = {
			value = minor_court_position_piety_salary
		}
		round = no
	}

	base_employer_modifier = {
		monthly_piety_gain_mult = 0.05
	}

	custom_employer_modifier_description = tm_voice_companion_court_position_employer_tt

	modifier = {
		monthly_learning_lifestyle_xp_gain_mult = 0.1
	}

	on_court_position_received = {
		scope:employee = {
			if = {
				limit = {
					NOT = { has_character_modifier = tm_companion_modifier }
					NOT = { any_living_character = { count >= 7 has_character_modifier = tm_companion_modifier } }
				}
				add_character_modifier = { modifier = tm_companion_modifier }
			}
		}
	}

	ai_position_score = {
		value = 100
	}
	ai_candidate_score = {
		value = 200
		add = court_position_candidate_score_base_value
		add = court_position_candidate_aptitude_value
	}
}
