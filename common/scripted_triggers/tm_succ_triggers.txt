# Crusader Dolls — the Sisters Choose taken up by AI rulers (tm_succ_on_actions.txt; docs/wave5/succession.md).

# The title already follows a law of its own (an elective law, the head of faith's, a noble family's, the land's,
# a confederation's or the celestial ministries'): leave it. Scope: a title.
tm_succ_title_has_own_law_trigger = {
	OR = {
		has_title_law_flag = elective_succession_law
		has_title_law = temporal_head_of_faith_succession_law
		has_title_law = noble_family_succession_law
		has_title_law = law_of_the_land_succession_law
		has_title_law = celestial_ministry_appointment_succession_law
		has_title_law = celestial_grand_marshal_appointment_succession_law
	}
}

# A daughter of the ruler's faith whom the Sisters Choose would name a candidate (tm_sisters_elective.txt): her child,
# born or taken, a woman of her council, or a landed vassal. Scope: the ruler.
tm_succ_has_daughter_candidate_trigger = {
	OR = {
		any_child = {
			is_alive = yes
			faith = root.faith
			tm_is_daughter_trigger = yes
		}
		any_councillor = {
			faith = root.faith
			tm_is_daughter_trigger = yes
		}
		any_vassal = {
			faith = root.faith
			tm_is_daughter_trigger = yes
		}
	}
}

# Would take up the Sisters Choose for her primary title. Mirrors the law's can_have and can_pass
# (common/laws/tm_succession_laws.txt). Scope: the ruler.
tm_succ_would_take_sisters_law_trigger = {
	rite ?= { rite_has_parameter = tm_sisters_faith }
	OR = {
		government_has_flag = government_is_feudal
		government_has_flag = government_is_clan
		government_has_flag = government_is_tribal
	}
	highest_held_title_tier >= tier_duchy
	can_change_title_law_trigger = yes
	primary_title = {
		NOT = { tm_succ_title_has_own_law_trigger = yes }
		can_title_have_law_general_trigger = yes
	}
	tm_succ_has_daughter_candidate_trigger = yes
}
