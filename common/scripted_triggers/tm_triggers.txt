# Crusader Dolls — scripted triggers

# Dorley intake (Aunt Bea's House): a man proven violent, or one the Mother already marked
# (the Stefan case: she walked into the basement on purpose). Adults only; no upper age limit.
tm_intake_eligible_trigger = {
	is_male = yes
	age >= 18
	OR = {
		has_trait = sadistic
		has_trait = wrathful
		has_trait = callous
		has_trait = tm_transgender
		any_secret = {
			OR = {
				secret_type = secret_murder
				secret_type = secret_murder_attempt
			}
		}
	}
}

# Anyone the Sisters would go to war to bring home.
tm_captive_daughter_trigger = {
	OR = {
		tm_is_daughter_trigger = yes
		has_trait = tm_novice
		has_trait = tm_transgender
	}
}

# Any daughter, whichever gate she came through (knife, seiðr, embryo, the Turning).
tm_is_daughter_trigger = {
	OR = {
		has_trait = tm_daughter_of_the_knife
		has_trait = tm_daughter_of_seidr
		has_trait = tm_daughter_of_the_embryo
		has_trait = tm_daughter_of_the_turning
		has_trait = tm_daughter_of_the_garland
		has_trait = tm_daughter_of_the_house
	}
}

# This character's rite raises daughters by some gate (used for arrivals and AI).
tm_rite_can_raise_daughters_trigger = {
	rite ?= {
		OR = {
			rite_has_parameter = tm_knife_voluntary
			rite_has_parameter = tm_seidr_gate
			rite_has_parameter = tm_embryo_gate
			rite_has_parameter = tm_turning_gate
			rite_has_parameter = tm_garland_gate
			rite_has_parameter = tm_crossing_gate
		}
	}
}

# Holds at least one county with a Sisters' House (tm_found_house_decision).
tm_holds_sisters_house_trigger = {
	any_held_title = {
		tier = tier_county
		has_variable = tm_house
	}
}

# Belongs to a Sisters' House: of a sisters' faith, and a woman, Marked, a daughter, or a friend of the
# House. The Houses are not separatist (Raha & van der Drift): cis women of the faith are sisters, and men
# of the faith may stay as friends of the House.
tm_house_member_trigger = {
	rite ?= { rite_has_parameter = tm_sisters_faith }
	OR = {
		is_female = yes
		tm_is_daughter_trigger = yes
		has_trait = tm_transgender
		has_character_modifier = tm_house_friend_modifier
	}
}

# Can be chosen Elder Sister of a House: a member who is a woman (friends of the House are guests).
tm_elder_sister_eligible_trigger = {
	is_adult = yes
	is_imprisoned = no
	rite ?= { rite_has_parameter = tm_sisters_faith }
	OR = {
		is_female = yes
		tm_is_daughter_trigger = yes
		has_trait = tm_transgender
	}
}

# Knows the warding songs (varðlokkur): every daughter of the high seat, and anyone taught them.
tm_knows_vardlokkur_trigger = {
	OR = {
		has_trait = tm_daughter_of_seidr
		has_variable = tm_knows_vardlokkur
	}
}

# Could ask a völva to sit for them.
tm_seidr_seeker_trigger = {
	is_male = yes
	is_adult = yes
	is_imprisoned = no
	rite ?= { rite_has_parameter = tm_seidr_gate }
	tm_is_daughter_trigger = no
	OR = {
		has_trait = tm_transgender
		has_trait = eunuch_1
		has_trait = beardless_eunuch
		has_variable = tm_declared
	}
}

# May hold the eunuch-reserved offices of a Dorleyite court: daughters of the knife, and eunuchs.
tm_cut_court_trigger = {
	OR = {
		has_trait = tm_daughter_of_the_knife
		has_trait = eunuch_1
		has_trait = beardless_eunuch
	}
}

# May keep the empress's chamber: any woman, or any daughter.
tm_praiposita_court_trigger = {
	OR = {
		is_female = yes
		tm_is_daughter_trigger = yes
	}
}

# Could keep the fourteenth of Adar and pray to be turned: of Ha-Nehpakhot, or of another Jewish faith and
# Marked (the Turning converts her). Eunuchs may come too: a eunuch who seeks womanhood is a sister.
tm_turning_seeker_trigger = {
	is_male = yes
	is_adult = yes
	is_imprisoned = no
	tm_is_daughter_trigger = no
	OR = {
		rite ?= { rite_has_parameter = tm_turning_gate }
		AND = {
			rite ?= { rite_has_parameter = jewish_syncretic_recipient }
			OR = {
				has_trait = tm_transgender
				has_variable = tm_declared
			}
		}
	}
}

# Purim falls in Adar, which the game's calendar puts in February or March.
tm_is_adar_trigger = {
	OR = {
		current_month = 2
		current_month = 3
	}
}

# Mārkaḻi, the Tamil month of the pāvai vow, from mid-December to mid-January.
tm_is_markali_trigger = {
	OR = {
		current_month = 12
		current_month = 1
	}
}

# A court a faith can make its home: not a mercenary company or a holy order. Companies raised in a county of the
# faith take its faith, and in an observe run (867–1005) the largest Ha-Nehpakhot "court" became a mercenary band,
# so the kahal's pilgrims went to the camp. Adventurers' camps still count. Scope: the court's ruler.
tm_settled_court_trigger = {
	NOR = {
		government_has_flag = government_is_mercenary
		government_has_flag = government_is_holy_order
	}
}
