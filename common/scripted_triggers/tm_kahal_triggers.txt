# Crusader Dolls — Ha-Nehpakhot's family and women's synagogue (events/tm_kahal_events.txt).

# A kinsman or kinswoman who may send a daughter of the Turning a prayer book. Their choice is their own; this only
# says who is asked. Scope: the kin. Root: the daughter.
tm_kahal_book_giver_trigger = {
	is_alive = yes
	is_adult = yes
	is_imprisoned = no
	faith.religion = religion:judaism_religion
	NOT = { has_variable = tm_kahal_book_declined }
	opinion = { target = root value >= -25 }
}

# A woman of a Jewish faith at the daughter's court, other than herself. Scope: the woman. Root: the daughter.
tm_kahal_door_woman_trigger = {
	is_female = yes
	is_adult = yes
	is_imprisoned = no
	faith.religion = religion:judaism_religion
	NOT = { this = root }
}

# A daughter of the Turning at a court with Jewish women to stand with. Scope: the daughter (root).
tm_kahal_door_trigger = {
	has_trait = tm_daughter_of_the_turning
	is_adult = yes
	OR = {
		AND = {
			is_ruler = yes
			any_courtier = { tm_kahal_door_woman_trigger = yes }
		}
		AND = {
			is_ruler = no
			employer ?= {
				OR = {
					tm_kahal_door_woman_trigger = yes
					any_courtier = { tm_kahal_door_woman_trigger = yes }
				}
			}
		}
	}
}
