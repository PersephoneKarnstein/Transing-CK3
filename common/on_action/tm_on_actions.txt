# Adelphai Dorleyaiou — on_action hooks.
# Vanilla's on_game_start_after_lobby has its own `effect` (which a second definition would replace) and
# no character scope. So we only append our own on_action to its `on_actions` list (lists merge) and do
# the work in tm_on_game_start_after_lobby, iterating players explicitly.

on_game_start_after_lobby = {
	on_actions = {
		tm_on_game_start_after_lobby
	}
}

tm_on_game_start_after_lobby = {
	effect = {
		# The first human player of the faith chooses its House of the Knife.
		random_player = {
			limit = { faith = faith:transbian_maximalism }
			trigger_event = {
				id = tm_knife.0100
				days = 5
			}
		}
	}
}

# Starting presence (design lock: open question 1). Done by script rather than by overriding vanilla
# history files, so it survives game updates and works for every bookmark.
on_game_start = {
	on_actions = {
		tm_on_game_start
	}
}

tm_on_game_start = {
	effect = {
		if = {
			limit = { NOT = { has_global_variable = tm_setup_done } }
			set_global_variable = { name = tm_setup_done value = yes }

			# Phrygia (Dorylaion, Pessinus) keeps the old faith; its count's house converts with it.
			title:c_phrygia = {
				set_county_culture = culture:kybeline
				set_county_faith = faith:transbian_maximalism
				holder ?= {
					set_culture = culture:kybeline
					set_character_faith = faith:transbian_maximalism
				}
			}

			# Theia Beatrike, Mother of Mothers: "Aunt Bea", an elder daughter of the knife.
			create_character = {
				name = tm_name_theia_beatrike
				gender = female
				age = 54
				culture = culture:kybeline
				faith = faith:transbian_maximalism
				dynasty = none
				location = title:c_phrygia.title_province
				random_traits = no
				trait = tm_daughter_of_the_knife
				trait = education_learning_4
				trait = compassionate
				trait = patient
				trait = just
				learning = 18
				diplomacy = 12
				stewardship = 9
				intrigue = 11
				martial = 4
				save_scope_as = tm_beatrike
			}
			create_title_and_vassal_change = {
				type = created
				save_scope_as = tm_mother_house_change
				add_claim_on_loss = no
			}
			title:d_tm_mother_house = {
				change_title_holder = {
					holder = scope:tm_beatrike
					change = scope:tm_mother_house_change
				}
			}
			resolve_title_and_vassal_change = scope:tm_mother_house_change

			# Elder Sisters at the Mother House, so the Dorley path has sponsors and the finder has daughters.
			while = {
				count = 3
				create_character = {
					gender = female
					age = { 30 55 }
					culture = culture:kybeline
					faith = faith:transbian_maximalism
					employer = scope:tm_beatrike
					dynasty = none
					trait = tm_daughter_of_the_knife
					trait = education_learning_3
				}
			}

			# The northern faith keeps Gotland; a völva sits at its count's court.
			title:c_gutland = {
				set_county_faith = faith:tm_baeddel
				holder ?= {
					set_character_faith = faith:tm_baeddel
					save_scope_as = tm_gutland_count
				}
			}
			if = {
				limit = { exists = scope:tm_gutland_count }
				create_character = {
					gender = female
					age = { 40 60 }
					culture = scope:tm_gutland_count.culture
					faith = faith:tm_baeddel
					employer = scope:tm_gutland_count
					dynasty = none
					trait = tm_daughter_of_seidr
					trait = education_learning_3
				}
			}

			# The Way of the Jade Embryo keeps Dunhuang (Shazhou, with the Mogao caves); a nüguan at court.
			title:c_shazhou = {
				set_county_faith = faith:tm_yutai
				holder ?= {
					set_character_faith = faith:tm_yutai
					save_scope_as = tm_shazhou_count
				}
			}
			if = {
				limit = { exists = scope:tm_shazhou_count }
				create_character = {
					gender = female
					age = { 40 60 }
					culture = scope:tm_shazhou_count.culture
					faith = faith:tm_yutai
					employer = scope:tm_shazhou_count
					dynasty = none
					trait = tm_daughter_of_the_embryo
					trait = education_learning_3
				}
			}

			# Historical characters are not "randomly created", so the 1% seed never rolls for them.
			# Mark ~1% of living adult men at the start, as the trait's random_creation does for later ones.
			every_living_character = {
				limit = {
					is_male = yes
					is_adult = yes
					NOT = { has_trait = tm_transgender }
				}
				random = {
					chance = 1
					add_trait = tm_transgender
				}
			}

			# Theia Beatrike also takes Phrygia, the seat, as a vassal of the same liege; the old count
			# becomes a daughter of the knife (cosmetic: no re-embodiment at game start). In 867 he also
			# holds Prusa and stays landed there; if he held nothing else he joins Beatrike's court.
			title:c_phrygia = {
				holder ?= {
					save_scope_as = tm_old_count
					# top liege, not liege: Beatrike heads a duchy-tier title, so she cannot be a duke's vassal
					top_liege ?= { save_scope_as = tm_old_liege }
				}
			}
			if = {
				limit = { exists = scope:tm_old_count }
				create_title_and_vassal_change = {
					type = granted
					save_scope_as = tm_phrygia_change
					add_claim_on_loss = no
				}
				title:c_phrygia = {
					change_title_holder = {
						holder = scope:tm_beatrike
						change = scope:tm_phrygia_change
						government_base = scope:tm_old_count
					}
				}
				if = {
					limit = {
						exists = scope:tm_old_liege
						NOT = { scope:tm_old_liege = scope:tm_old_count }
					}
					scope:tm_beatrike = {
						change_liege = {
							liege = scope:tm_old_liege
							change = scope:tm_phrygia_change
						}
					}
				}
				resolve_title_and_vassal_change = scope:tm_phrygia_change

				scope:tm_old_count = {
					add_trait = tm_daughter_of_the_knife
					if = {
						limit = { is_landed = no }
						scope:tm_beatrike = { add_courtier = scope:tm_old_count }
					}
				}
			}
		}
	}
}

# Arrivals: each year a ruler of the faith has a small chance that a Marked woman finds her way to the court
# (mechanical counterpart to Rescue the Daughters; the 'flight to a sister's house' of the belief list, E9).
yearly_playable_pulse = {
	on_actions = {
		tm_yearly_playable
	}
}

tm_yearly_playable = {
	trigger = {
		is_landed = yes
		rite ?= { rite_has_parameter = tm_sisters_faith }
	}
	effect = {
		random = {
			chance = 6
			modifier = {								# the Mother House draws them
				add = 6
				any_held_title = { this = title:c_phrygia }
			}
			modifier = {
				add = 3
				any_courtier = { tm_is_daughter_trigger = yes }
			}
			modifier = {								# a declared refuge (tm_declare_refuge_decision)
				add = 15
				house ?= { has_variable = tm_refuge }
			}
			random_list = {
				70 = { trigger_event = tm_arrival.0001 }	# a Marked traveller
				30 = { trigger_event = tm_arrival.0003 }	# a woman turned out for barrenness
			}
		}
	}
}
