# Crusader Dolls — on_action hooks.
# Vanilla's on_game_start_after_lobby has its own `effect` (which a second definition would replace) and
# no character scope. So we only append our own on_action to its `on_actions` list (lists merge) and do
# the work in tm_on_game_start_after_lobby, iterating players explicitly.

on_game_start_after_lobby = {
	on_actions = {
		tm_on_game_start_after_lobby
	}
}

tm_on_game_start_after_lobby = {
	effect = {
		# The first human player of the faith chooses its House of the Knife.
		random_player = {
			limit = { faith = faith:transbian_maximalism }
			trigger_event = {
				id = tm_knife.0100
				days = 5
			}
		}
	}
}

# Starting presence (design lock: open question 1). Done by script rather than by overriding vanilla
# history files, so it survives game updates and works for every bookmark.
on_game_start = {
	on_actions = {
		tm_on_game_start
	}
}

tm_on_game_start = {
	effect = {
		if = {
			limit = { NOT = { has_global_variable = tm_setup_done } }
			set_global_variable = { name = tm_setup_done value = yes }

			# Phrygia (Dorylaion, Pessinus) keeps the old faith; its count's house converts with it. Only before 1080:
			# after Manzikert Dorylaion lay empty for about a century, and Manuel I rebuilt it only in 1175 (Kinnamos;
			# docs/research/later_starts/adelphai_1178_house_in_exile.md). In vanilla 1178 Phrygia's count is John Doukas,
			# a historical figure the mod must not convert or make a daughter. From 1080 the Mother House is in exile.
			if = {
				limit = { current_date < 1080.1.1 }
				title:c_phrygia = {
					set_county_culture = culture:kybeline
					set_county_faith = faith:transbian_maximalism
					holder ?= {
						set_culture = culture:kybeline
						set_variable = { name = tm_conv_ours value = yes days = 1 }
						set_character_faith = faith:transbian_maximalism
					}
				}
			}

			# Theia Beatrike, Mother of Mothers: "Aunt Bea", an elder daughter of the knife. At 867 she is the
			# historical character of the bookmark (history/characters/tm_characters.txt); at a later start, a stand-in.
			character:tm_theia_beatrike ?= {
				if = {
					limit = { is_alive = yes }
					save_scope_as = tm_beatrike
					tm_register_seed_effect = { SEAT = phrygia }
				}
			}
			if = {
				limit = { NOT = { exists = scope:tm_beatrike } }
				create_character = {
					after_creation = { tm_register_seed_effect = { SEAT = phrygia } }
					name = tm_name_theia_beatrike
					gender = female
					age = 44
					culture = culture:kybeline
					faith = faith:transbian_maximalism
					dynasty = none
					location = title:c_phrygia.title_province
					random_traits = no
					trait = tm_daughter_of_the_knife
					trait = education_learning_4
					trait = compassionate
					trait = patient
					trait = just
					learning = 18
					diplomacy = 12
					stewardship = 9
					intrigue = 11
					martial = 4
					save_scope_as = tm_beatrike
				}
			}
			create_title_and_vassal_change = {
				type = created
				save_scope_as = tm_mother_house_change
				add_claim_on_loss = no
			}
			title:d_tm_mother_house = {
				change_title_holder = {
					holder = scope:tm_beatrike
					change = scope:tm_mother_house_change
				}
			}
			resolve_title_and_vassal_change = scope:tm_mother_house_change

			# Elder Sisters at the Mother House, so the Dorley path has sponsors and the finder has daughters.
			while = {
				count = 3
				create_character = {
					after_creation = { tm_register_seed_effect = { SEAT = phrygia } }
					gender = female
					age = { 30 55 }
					culture = culture:kybeline
					faith = faith:transbian_maximalism
					employer = scope:tm_beatrike
					dynasty = none
					trait = tm_daughter_of_the_knife
					trait = education_learning_3
				}
			}

			# The northern faith's seat, by date (docs/later_starts.md). Gotland until the island turns Christian, by
			# c. 1050 in Guta saga (in vanilla 1066 it is a Catholic republic). Then Åland, whose vanilla count Olaf is
			# Norse pagan, under the Svealand of Erik "the Heathen", while the Uppsala cult lasts (Adam of Bremen, c. 1075;
			# vanilla's Uppland is pagan to 1090). Later the faith holds no land. At 867 Gotland's count is converted,
			# as Shazhou's is (user, 2026-10-20); in the second observe run that line went on to create Scandinavia.
			if = {
				limit = { current_date < 1050.1.1 }
				# At 867 Gotland is Steinunn's (the bookmark's), the count Steinn's elder sister: she takes his county
				# and he stays at her court, of her faith, as her heir (user, 2026-10-23).
				clear_saved_scope = tm_steinunn
				character:tm_baeddel_steinunn ?= {
					if = {
						limit = {
							is_alive = yes
							title:c_gutland.holder ?= { this = character:41267 }
						}
						save_scope_as = tm_steinunn
					}
				}
				if = {
					limit = { exists = scope:tm_steinunn }
					character:41267 = { save_scope_as = tm_steinn }
					tm_hand_over_realm_effect = { FROM = scope:tm_steinn TO = scope:tm_steinunn }
					scope:tm_steinn = { set_variable = { name = tm_conv_ours value = yes days = 1 } set_character_faith = faith:tm_baeddel }
					scope:tm_steinunn = { add_courtier = scope:tm_steinn }
				}
				tm_seed_baeddel_seat_effect = { COUNTY = title:c_gutland SEAT = gotland }
			}
			else_if = {
				limit = { current_date < 1090.1.1 }
				tm_seed_baeddel_seat_effect = { COUNTY = title:c_aland SEAT = aland }
			}

			# The Way of the Jade Embryo keeps Dunhuang (Shazhou, with the Mogao caves); a nüguan at court.
			# At 867 the bookmark's Huijing takes Guasha from the vanilla duke Xun, a historical man the mod does not
			# convert: he dies before the game begins, and she takes his son Chengxun (han_40016, aged 7) as her ward,
			# adopted child and heir, raised in the Way (user, 2026-10-23). Suo's noble-family estate goes to Xun's heirs.
			clear_saved_scope = tm_huijing
			character:tm_yutai_huijing ?= {
				if = {
					limit = {
						is_alive = yes
						title:c_shazhou.holder ?= { this = character:304191 }
					}
					save_scope_as = tm_huijing
				}
			}
			# Guasha's arms are the tiger only for the bookmark's selector (tm_coas.txt): in play, vanilla's seal.
			title:d_guasha = { set_coa = tm_d_guasha_vanilla }
			if = {
				limit = { exists = scope:tm_huijing }
				character:304191 = { save_scope_as = tm_xun }
				tm_hand_over_realm_effect = { FROM = scope:tm_xun TO = scope:tm_huijing }
				character:han_40016 ?= {
					if = {
						limit = { is_alive = yes }
						save_scope_as = tm_chengxun
					}
				}
				scope:tm_xun = { death = { death_reason = death_natural_causes } }
				if = {
					limit = { exists = scope:tm_chengxun }
					scope:tm_huijing = {
						add_courtier = scope:tm_chengxun
						adopt = scope:tm_chengxun
						set_relation_ward = scope:tm_chengxun
						set_designated_heir = scope:tm_chengxun
					}
					scope:tm_chengxun = { set_variable = { name = tm_conv_ours value = yes days = 1 } set_character_faith = faith:tm_yutai }
				}
			}
			title:c_shazhou = {
				set_county_faith = faith:tm_yutai
				holder ?= {
					set_variable = { name = tm_conv_ours value = yes days = 1 }
					set_character_faith = faith:tm_yutai
					save_scope_as = tm_shazhou_count
				}
			}
			if = {
				limit = { exists = scope:tm_shazhou_count }
				create_character = {
					after_creation = { tm_register_seed_effect = { SEAT = shazhou } }
					gender = female
					age = { 40 60 }
					culture = scope:tm_shazhou_count.culture
					faith = faith:tm_yutai
					employer = scope:tm_shazhou_count
					dynasty = none
					trait = tm_daughter_of_the_embryo
					trait = education_learning_3
				}
			}

			if = {
				limit = { exists = scope:tm_shazhou_count }
				tm_seed_marked_courtiers_effect = { HOST = scope:tm_shazhou_count FAITH = faith:tm_yutai SEAT = shazhou }
			}

			# Ha-Nehpakhot hold no land: the women of the lamp in the Jewish street of Arles, under its lord. Kalonymus
			# (born 1287) is claimed in the faith's own voice only; no character stands for him.
			title:c_venaissin = {
				set_variable = tm_kahal			# the women of the lamp's own street: the Turning is likelier here (tm_hipukh.0001)
				holder ?= { save_scope_as = tm_arles_lord }
			}
			# At 867 the elder is Bonadonna, who keeps the women of the lamp's camp (Roads to Power; history/titles/
			# tm_laamp_titles.txt) and stands on the bookmark. At a later start she is dead and the script makes an elder
			# at the lord's court instead.
			character:tm_nehpakhot_bonadonna ?= {
				if = {
					limit = {
						is_alive = yes
						has_title = title:d_laamp_tm_lamp
					}
					save_scope_as = tm_arles_elder
					save_scope_as = tm_arles_camp
					tm_register_seed_effect = { SEAT = arles }
				}
			}
			if = {
				limit = { NOT = { exists = scope:tm_arles_elder } }
				create_character = {
					after_creation = { tm_register_seed_effect = { SEAT = arles } }
					gender = female
					age = { 50 65 }
					culture = title:c_venaissin.culture
					faith = faith:tm_nehpakhot
					dynasty = none
					location = title:c_venaissin.title_province
					save_scope_as = tm_arles_elder
					trait = tm_daughter_of_the_turning
					trait = education_learning_4
					trait = patient
					learning = 16
				}
			}
			while = {
				count = 2
				create_character = {
					after_creation = { tm_register_seed_effect = { SEAT = arles } }
					gender = male
					age = { 18 30 }
					culture = title:c_venaissin.culture
					faith = faith:tm_nehpakhot
					dynasty = none
					location = title:c_venaissin.title_province
					trait = tm_transgender
					trait = education_learning_2
					save_temporary_scope_as = tm_arles_youth
				}
				# In Bonadonna's camp; else at the lord's court, as Jews of Arles served as its physicians: pool
				# characters drift and die out.
				if = {
					limit = { exists = scope:tm_arles_camp }
					scope:tm_arles_camp = { add_courtier = scope:tm_arles_youth }
				}
				else_if = {
					limit = { exists = scope:tm_arles_lord }
					scope:tm_arles_lord = { add_courtier = scope:tm_arles_youth }
				}
			}
			if = {
				limit = {
					NOT = { exists = scope:tm_arles_camp }
					exists = scope:tm_arles_lord
				}
				scope:tm_arles_lord = { add_courtier = scope:tm_arles_elder }
			}

			# A household of the sect: a physician, his wife and their children. Daughters bear no children and the Marked
			# who join are few, so without families the sect dies with its first generation (it was extinct by 1042 in an
			# observe run). Their children are born into it.
			create_character = {
				after_creation = { tm_register_seed_effect = { SEAT = arles } }
				gender = male
				age = { 30 40 }
				culture = title:c_venaissin.culture
				faith = faith:tm_nehpakhot
				dynasty = generate
				location = title:c_venaissin.title_province
				trait = lifestyle_physician
				trait = education_learning_3
				save_scope_as = tm_arles_father
			}
			create_character = {
				after_creation = { tm_register_seed_effect = { SEAT = arles } }
				gender = female
				age = { 24 32 }
				culture = title:c_venaissin.culture
				faith = faith:tm_nehpakhot
				dynasty = none
				location = title:c_venaissin.title_province
				save_scope_as = tm_arles_mother
			}
			scope:tm_arles_father = { marry = scope:tm_arles_mother }
			while = {
				count = 3
				create_character = {
					after_creation = { tm_register_seed_effect = { SEAT = arles } }
					age = { 6 12 }						# old enough for a guardian
					gender_female_chance = 50
					father = scope:tm_arles_father
					mother = scope:tm_arles_mother
					dynasty = inherit
					location = title:c_venaissin.title_province
					save_temporary_scope_as = tm_arles_child
				}
				# Raised by their own parents, or they grow up in the faith of the lord's court: in an observe run all
				# three were Christian by 901.
				scope:tm_arles_child = {
					if = {
						limit = { is_female = yes }
						set_relation_guardian = scope:tm_arles_mother
					}
					else = {
						set_relation_guardian = scope:tm_arles_father
					}
				}
				if = {
					limit = { exists = scope:tm_arles_lord }
					scope:tm_arles_lord = { add_courtier = scope:tm_arles_child }
				}
			}
			if = {
				limit = { exists = scope:tm_arles_lord }
				scope:tm_arles_lord = {
					add_courtier = scope:tm_arles_father
					add_courtier = scope:tm_arles_mother
				}
			}

			# The Worn Garland: the temple street of Kanchipuram (tm_seed_foothold_effect).
			tm_seed_foothold_effect = { COUNTY = title:c_kanchipuram FAITH = faith:tm_garland SEAT = kanchipuram }
			scope:tm_foothold_host ?= { save_scope_as = tm_kanchi_lord }
			if = {
				limit = { exists = scope:tm_kanchi_lord }
				create_character = {
					after_creation = { tm_register_seed_effect = { SEAT = kanchipuram } }
					gender = female
					age = { 40 60 }
					culture = title:c_kanchipuram.culture
					faith = faith:tm_garland
					employer = scope:tm_kanchi_lord
					dynasty = none
					trait = tm_daughter_of_the_garland
					trait = education_diplomacy_3
				}
			}
			# Kanchipuram's lord keeps it: at 867 it is the Pallava capital; in vanilla 1066 a Telugu Yadava count holds
			# it, and in 1178 the Chola king. A patron of the temple street holds Candragiri, in the same duchy, as his
			# vassal: a landed court of the faith (user, 2026-10-20). The Marked courtiers are hers.
			character:tm_garland_cantiradevi ?= {		# the bookmark's countess (history/characters/tm_characters.txt)
				if = {
					limit = { is_alive = yes is_landed = no }
					save_scope_as = tm_foothold_preset
				}
			}
			tm_seed_foothold_effect = { COUNTY = title:c_candragiri FAITH = faith:tm_garland SEAT = kanchipuram }
			if = {
				limit = { scope:tm_foothold_host ?= { faith = faith:tm_garland } }
				tm_seed_marked_courtiers_effect = { HOST = scope:tm_foothold_host FAITH = faith:tm_garland SEAT = kanchipuram }
			}
			else_if = {
				limit = { exists = scope:tm_kanchi_lord }
				tm_seed_marked_courtiers_effect = { HOST = scope:tm_kanchi_lord FAITH = faith:tm_garland SEAT = kanchipuram }
			}

			# Badenya's seat, by date (docs/later_starts.md). Kirina at 867. From 1040 Kirina's court is Muslim (al-Bakri,
			# 1068: the king of Malal converted and broke the idols), while Ghana keeps its royal grove-town, al-Ghaba
			# (Awkar), so the faith sits there until Ghana's people convert (al-Zuhri: 469/1076-77, not a conquest;
			# Conrad and Fisher 1982-83). From 1076 it is back among the villages of Kirina, under a Muslim lord.
			character:tm_badenya_djeneba ?= {		# the bookmark's countess (history/characters/tm_characters.txt)
				if = {
					limit = { is_alive = yes is_landed = no }
					save_scope_as = tm_foothold_preset
				}
			}
			if = {
				limit = { current_date >= 1040.1.1 current_date < 1076.1.1 }
				clear_saved_scope = tm_foothold_preset
				# Ghana's royal county keeps its own faith: the faith lives among the women at the king's court, not as
				# the religion of Ghana's people (docs/research/later_starts/badenya_ghana_1066.md).
				save_scope_value_as = { name = tm_foothold_courtiers_only value = yes }
				tm_seed_badenya_seat_effect = { COUNTY = title:c_awkar SEAT = awkar }
			}
			else = {
				tm_seed_badenya_seat_effect = { COUNTY = title:c_sibi_mali SEAT = kirina }
			}

			# Historical characters are not "randomly created", so the 1% seed never rolls for them.
			# Mark ~1.5% of living adult men at the start, as the trait's random_creation does for later ones.
			every_living_character = {
				limit = {
					is_male = yes
					is_adult = yes
					NOT = { has_trait = tm_transgender }
				}
				random = {
					chance = 1.5
					add_trait = tm_transgender
				}
			}

			# Theia Beatrike also takes Phrygia, the seat, as a vassal of the same liege; the old count
			# becomes a daughter of the knife (cosmetic: no re-embodiment at game start). In 867 he also
			# holds Prusa and stays landed there; if he held nothing else he joins Beatrike's court.
			if = {
				limit = { current_date < 1080.1.1 }	# see the note on Phrygia above
				title:c_phrygia = {
					holder ?= {
						save_scope_as = tm_old_count
						# top liege, not liege: Beatrike heads a duchy-tier title, so she cannot be a duke's vassal
						top_liege ?= { save_scope_as = tm_old_liege }
					}
				}
			}
			if = {
				limit = { exists = scope:tm_old_count }
				create_title_and_vassal_change = {
					type = granted
					save_scope_as = tm_phrygia_change
					add_claim_on_loss = no
				}
				title:c_phrygia = {
					change_title_holder = {
						holder = scope:tm_beatrike
						change = scope:tm_phrygia_change
						government_base = scope:tm_old_count
					}
				}
				if = {
					limit = {
						exists = scope:tm_old_liege
						NOT = { scope:tm_old_liege = scope:tm_old_count }
					}
					scope:tm_beatrike = {
						change_liege = {
							liege = scope:tm_old_liege
							change = scope:tm_phrygia_change
						}
					}
				}
				resolve_title_and_vassal_change = scope:tm_phrygia_change

				scope:tm_old_count = {
					add_trait = tm_daughter_of_the_knife
					if = {
						limit = { is_landed = no }
						scope:tm_beatrike = { add_courtier = scope:tm_old_count }
					}
				}
			}
			# The Mother of Mothers rules as a theocracy: in the 867 run Byzantium's administrative appointments
			# revoked Phrygia and gave it to an Orthodox governor. Untested whether this alone holds it (TO-TEST G4).
			scope:tm_beatrike = {
				change_government = theocracy_government
			}
			tm_seed_marked_courtiers_effect = { HOST = scope:tm_beatrike FAITH = faith:transbian_maximalism SEAT = phrygia }
		}
	}
}

# Arrivals: each year a ruler of the faith has a small chance that a Marked woman finds her way to the court
# (mechanical counterpart to Rescue the Daughters; the 'flight to a sister's house' of the belief list, E9).
yearly_playable_pulse = {
	on_actions = {
		tm_yearly_playable
		tm_yearly_houses
		tm_yearly_islamic_courts
		tm_yearly_turning
		tm_yearly_south
	}
}

# The Worn Garland and Badenya (tm_garland_events.txt, tm_badenya_events.txt).
tm_yearly_south = {
	effect = {
		# A name in stone: a ruler of the Garland endows the temple. A ruler who is herself a daughter gives in her own
		# name instead (tm_ml2.0010), so no ruler sees both gift events (docs/stage2/worn-garland.md, change 1).
		if = {
			limit = {
				rite ?= { rite_has_parameter = tm_garland_gate }
				NOT = { tm_ml2_donor_trigger = yes }
			}
			random = {
				chance = 15
				trigger_event = { id = tm_garland.0020 days = { 1 300 } }
			}
		}
		# A pandit's objection: a Hindu lord not of the Garland with a daughter at court.
		if = {
			limit = {
				faith ?= { religion = religion:hinduism_religion }
				NOT = { rite ?= { rite_has_parameter = tm_garland_gate } }
				any_courtier = { tm_is_daughter_trigger = yes is_adult = yes }
			}
			random = {
				chance = 5
				random_courtier = {
					limit = { tm_is_daughter_trigger = yes is_adult = yes }
					save_scope_as = tm_garland_daughter
				}
				trigger_event = { id = tm_garland.0030 days = { 1 300 } }
			}
		}
		# A naming feast for a daughter of the House.
		if = {
			limit = { has_trait = tm_daughter_of_the_house }
			random = {
				chance = 30
				trigger_event = { id = tm_badenya.0020 days = { 1 300 } }
			}
		}
	}
}

# Ha-Nehpakhot (tm_hipukh_events.txt): the three lamps for a daughter of the Turning; a lord of any faith
# offered a translation; a Jewish lord asked to rule on a daughter at his court.
tm_yearly_turning = {
	effect = {
		if = {
			limit = {
				has_trait = tm_daughter_of_the_turning
				faith = { has_doctrine_parameter = tm_three_lamps }	# the tenet (a reformed sect without it does not keep them)
			}
			random = {
				chance = 25
				trigger_event = { id = tm_hipukh.0020 days = { 1 300 } }
			}
		}
		if = {
			limit = {
				any_courtier = { has_trait = tm_daughter_of_the_turning is_adult = yes }
			}
			random = {
				chance = 5
				random_list = {
					50 = {
						random_courtier = {
							limit = { has_trait = tm_daughter_of_the_turning is_adult = yes }
							save_scope_as = tm_translator
						}
						trigger_event = { id = tm_hipukh.0030 days = { 1 300 } }
					}
					50 = {									# women physicians are attested; women translators are not
						trigger = {
							any_courtier = {
								has_trait = tm_daughter_of_the_turning is_adult = yes
								NOT = { has_character_modifier = tm_fisica_licensed_modifier }
							}
						}

# Raised in the faith (user 2026-10-21): a child born to a member of the six, and kept at that parent's court, is
# brought up in the parent's faith, as children of an ethnoreligion are. In the 1135 observe run the faiths that grew
# (Yutai Dao, Badenya) were mostly born into; the rest were converts whose land passed to heirs of other faiths
# (Gotland went to the count's Finnish pagan daughter in 895). The mother is asked first. Root = the child.
on_birth_child = {
	on_actions = {
		tm_on_birth_raised_in_faith
	}
}

tm_on_birth_raised_in_faith = {
	trigger = {
		NOT = { rite ?= { rite_has_parameter = tm_sisters_faith } }
	}
	effect = {
		if = {
			limit = {
				exists = scope:mother
				scope:mother = {
					rite ?= { rite_has_parameter = tm_sisters_faith }
					OR = {
						NOT = { exists = root.employer }
						root.employer = this
						AND = { exists = employer root.employer = employer }
					}
				}
			}
			set_variable = { name = tm_conv_ours value = yes days = 1 }
			set_character_faith = scope:mother.faith
		}
		else_if = {
			limit = {
				exists = scope:father
				scope:father = {
					rite ?= { rite_has_parameter = tm_sisters_faith }
					OR = {
						root.employer ?= this
						AND = { exists = employer root.employer ?= employer }
					}
				}
			}
			set_variable = { name = tm_conv_ours value = yes days = 1 }
			set_character_faith = scope:father.faith
		}
	}
}

# Tracking: a member of the six dies (not a body left behind by re-embodiment); a seed's death is kept on a list.
on_death = {
	on_actions = {
		tm_on_death_track
	}
}

tm_on_death_track = {
	trigger = {
		NOT = { has_variable = tm_reborn }
		OR = {
			rite ?= { rite_has_parameter = tm_sisters_faith }
			has_variable = tm_seed
		}
	}
	effect = {
		# The game makes and clears characters by the thousand ("disappearance"): pool characters in a faith's counties,
		# and in China exam candidates and the like. Count them apart, or they swamp the deaths of people who lived.
		# A death counts if she had a court or was a ruler. (death_reason can't be read here: in on_death the character
		# is still alive, and error.log says so for every death.)
		# And only if she was seen at a census (tm_surv_seen, set just before each census): in the fourth observe run Yutai
		# Dao counted 462 deaths in the 900s with 84 members, and up to 2215 a decade later with about 1000. Characters
		# made and cleared between two censuses (in China, the game makes them by the hundred) go to pool_deaths.
		if = {
			limit = {
				rite ?= { rite_has_parameter = tm_sisters_faith }
				has_variable = tm_surv_seen
				is_pool_character = no
				OR = {
					is_ruler = yes
					exists = employer
				}
			}
			tm_track_effect = { FAITH = faith WHAT = deaths }
		}
		else_if = {
			limit = { rite ?= { rite_has_parameter = tm_sisters_faith } }
			tm_track_effect = { FAITH = faith WHAT = pool_deaths }
		}
		if = {
			limit = { has_variable = tm_seed }
			add_to_global_variable_list = { name = tm_seeds_dead target = this }
		}
	}
}

				is_ruler = no
				is_imprisoned = no

			}

		every_living_character = {

				is_ruler = no
				is_imprisoned = no
				is_married = no
				is_betrothed = no
				tm_is_daughter_trigger = no			# a daughter bears no children; she keeps her own life

				NOT = { has_trait = celibate }
				fertility > 0
			}
			random = {
				chance = 15
				tm_kahal_match_effect = yes
			}
		}
		# The pilgrimage to the kahal (tm_kahal_pilgrimage_effect): Marked young Jews come to the seat to be taken in.
		tm_kahal_pilgrimage_effect = yes
		# The faiths keep their children (tm_surv_keeps_children_effect): a guardian of another faith may not convert them.
		tm_surv_keeps_children_effect = yes
		# The gathering (tm_surv_gather_effect; it replaces the way home): members of the six with no court who are
		# daughters, Marked, declared or of a family of standing go to the nearest court of their faith.
		tm_surv_gather_effect = yes
		# The kahal: a Rabbanite Jew at a court where the sect is two or more, or at a court or in a county of a Jewish lord's
		# lands, may join it. A sect within Judaism keeps its families, and their children are born into it; this is how it
		# renews itself, not by the Marked alone. The game makes almost no Rabbanite Jews (three in the world in 1200 in the
		# fourth observe run), so the kahal of a landed ruler of the sect also draws from the wider Jewry

		every_living_character = {

					employer ?= {

					}
					AND = {
						NOT = { exists = employer }

				tm_track_effect = { FAITH = faith:tm_nehpakhot WHAT = kahal_joins }
			}
		}
		every_living_character = {
			limit = {
				faith = faith:tm_nehpakhot
				is_landed = yes
				is_imprisoned = no
				tm_settled_court_trigger = yes
			}
			tm_surv_kahal_draw_effect = yes
		}
		# The Mother House never forgets Phrygia: if the county leaves the faith, the Mother of Mothers holds a claim.
		if = {
			limit = {
				title:c_phrygia.holder ?= { NOT = { faith = faith:transbian_maximalism } }
				title:d_tm_mother_house.holder ?= { NOT = { has_claim_on = title:c_phrygia } }
			}
			title:d_tm_mother_house.holder = { add_pressed_claim = title:c_phrygia }
		}
		# Census for tuning: every 10 years, kept on each faith by decade (tm_tracking_effects.txt; tools/census_report.py reads a save).
		if = {
			limit = {
				OR = {
					NOT = { has_global_variable = tm_next_census }

				}
			}
			set_global_variable = { name = tm_next_census value = { value = current_year add = 10 } }

			every_living_character = {
				limit = {
					rite ?= { rite_has_parameter = tm_sisters_faith }
					NOT = { has_variable = tm_surv_seen }
				}
				set_variable = tm_surv_seen
			}
			tm_census_log_effect = yes
		}
	}
