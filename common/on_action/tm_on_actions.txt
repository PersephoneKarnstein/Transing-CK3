# Adelphai Dorleyaiou — on_action hooks.
# Vanilla's on_game_start_after_lobby has its own `effect` (which a second definition would replace) and
# no character scope. So we only append our own on_action to its `on_actions` list (lists merge) and do
# the work in tm_on_game_start_after_lobby, iterating players explicitly.

on_game_start_after_lobby = {
	on_actions = {
		tm_on_game_start_after_lobby
	}
}

tm_on_game_start_after_lobby = {
	effect = {
		# The first human player of the faith chooses its House of the Knife.
		random_player = {
			limit = { faith = faith:transbian_maximalism }
			trigger_event = {
				id = tm_knife.0100
				days = 5
			}
		}
	}
}

# Starting presence (design lock: open question 1). Done by script rather than by overriding vanilla
# history files, so it survives game updates and works for every bookmark.
on_game_start = {
	on_actions = {
		tm_on_game_start
	}
}

tm_on_game_start = {
	effect = {
		if = {
			limit = { NOT = { has_global_variable = tm_setup_done } }
			set_global_variable = { name = tm_setup_done value = yes }

			# Phrygia (Dorylaion, Pessinus) keeps the old faith; its count's house converts with it.
			title:c_phrygia = {
				set_county_culture = culture:kybeline
				set_county_faith = faith:transbian_maximalism
				holder ?= {
					set_culture = culture:kybeline
					set_character_faith = faith:transbian_maximalism
				}
			}

			# Theia Beatrike, Mother of Mothers: "Aunt Bea", an elder daughter of the knife.
			create_character = {
				after_creation = { tm_register_seed_effect = { SEAT = phrygia } }
				name = tm_name_theia_beatrike
				gender = female
				age = 54
				culture = culture:kybeline
				faith = faith:transbian_maximalism
				dynasty = none
				location = title:c_phrygia.title_province
				random_traits = no
				trait = tm_daughter_of_the_knife
				trait = education_learning_4
				trait = compassionate
				trait = patient
				trait = just
				learning = 18
				diplomacy = 12
				stewardship = 9
				intrigue = 11
				martial = 4
				save_scope_as = tm_beatrike
			}
			create_title_and_vassal_change = {
				type = created
				save_scope_as = tm_mother_house_change
				add_claim_on_loss = no
			}
			title:d_tm_mother_house = {
				change_title_holder = {
					holder = scope:tm_beatrike
					change = scope:tm_mother_house_change
				}
			}
			resolve_title_and_vassal_change = scope:tm_mother_house_change

			# Elder Sisters at the Mother House, so the Dorley path has sponsors and the finder has daughters.
			while = {
				count = 3
				create_character = {
					after_creation = { tm_register_seed_effect = { SEAT = phrygia } }
					gender = female
					age = { 30 55 }
					culture = culture:kybeline
					faith = faith:transbian_maximalism
					employer = scope:tm_beatrike
					dynasty = none
					trait = tm_daughter_of_the_knife
					trait = education_learning_3
				}
			}

			# The northern faith keeps Gotland; a völva sits at the court that holds it (tm_seed_foothold_effect).
			tm_seed_foothold_effect = { COUNTY = title:c_gutland FAITH = faith:tm_baeddel SEAT = gotland }
			scope:tm_foothold_host ?= { save_scope_as = tm_gutland_count }
			if = {
				limit = { exists = scope:tm_gutland_count }
				create_character = {
					after_creation = { tm_register_seed_effect = { SEAT = gotland } }
					gender = female
					age = { 40 60 }
					culture = scope:tm_gutland_count.culture
					faith = faith:tm_baeddel
					employer = scope:tm_gutland_count
					dynasty = none
					trait = tm_daughter_of_seidr
					trait = education_learning_3
				}
			}

			if = {
				limit = { exists = scope:tm_gutland_count }
				tm_seed_marked_courtiers_effect = { HOST = scope:tm_gutland_count FAITH = faith:tm_baeddel SEAT = gotland }
			}

			# The Way of the Jade Embryo keeps Dunhuang (Shazhou, with the Mogao caves); a nüguan at court.
			title:c_shazhou = {
				set_county_faith = faith:tm_yutai
				holder ?= {
					set_character_faith = faith:tm_yutai
					save_scope_as = tm_shazhou_count
				}
			}
			if = {
				limit = { exists = scope:tm_shazhou_count }
				create_character = {
					after_creation = { tm_register_seed_effect = { SEAT = shazhou } }
					gender = female
					age = { 40 60 }
					culture = scope:tm_shazhou_count.culture
					faith = faith:tm_yutai
					employer = scope:tm_shazhou_count
					dynasty = none
					trait = tm_daughter_of_the_embryo
					trait = education_learning_3
				}
			}

			if = {
				limit = { exists = scope:tm_shazhou_count }
				tm_seed_marked_courtiers_effect = { HOST = scope:tm_shazhou_count FAITH = faith:tm_yutai SEAT = shazhou }
			}

			# Ha-Nehpakhot hold no land: the women of the lamp in the Jewish street of Arles, under its lord. Kalonymus
			# (born 1287) is claimed in the faith's own voice only; no character stands for him.
			title:c_venaissin = {
				set_variable = tm_kahal			# the women of the lamp's own street: the Turning is likelier here (tm_hipukh.0001)
				holder ?= { save_scope_as = tm_arles_lord }
			}
			create_character = {
				after_creation = { tm_register_seed_effect = { SEAT = arles } }
				gender = female
				age = { 50 65 }
				culture = title:c_venaissin.culture
				faith = faith:tm_nehpakhot
				dynasty = none
				location = title:c_venaissin.title_province
				save_scope_as = tm_arles_elder
				trait = tm_daughter_of_the_turning
				trait = education_learning_4
				trait = patient
				learning = 16
			}
			while = {
				count = 2
				create_character = {
					after_creation = { tm_register_seed_effect = { SEAT = arles } }
					gender = male
					age = { 18 30 }
					culture = title:c_venaissin.culture
					faith = faith:tm_nehpakhot
					dynasty = none
					location = title:c_venaissin.title_province
					trait = tm_transgender
					trait = education_learning_2
					save_temporary_scope_as = tm_arles_youth
				}
				# At the lord's court, as Jews of Arles served as its physicians: pool characters drift and die out.
				if = {
					limit = { exists = scope:tm_arles_lord }
					scope:tm_arles_lord = { add_courtier = scope:tm_arles_youth }
				}
			}
			if = {
				limit = { exists = scope:tm_arles_lord }
				scope:tm_arles_lord = { add_courtier = scope:tm_arles_elder }
			}

			# A household of the sect: a physician, his wife and their children. Daughters bear no children and the Marked
			# who join are few, so without families the sect dies with its first generation (it was extinct by 1042 in an
			# observe run). Their children are born into it.
			create_character = {
				after_creation = { tm_register_seed_effect = { SEAT = arles } }
				gender = male
				age = { 30 40 }
				culture = title:c_venaissin.culture
				faith = faith:tm_nehpakhot
				dynasty = generate
				location = title:c_venaissin.title_province
				trait = lifestyle_physician
				trait = education_learning_3
				save_scope_as = tm_arles_father
			}
			create_character = {
				after_creation = { tm_register_seed_effect = { SEAT = arles } }
				gender = female
				age = { 24 32 }
				culture = title:c_venaissin.culture
				faith = faith:tm_nehpakhot
				dynasty = none
				location = title:c_venaissin.title_province
				save_scope_as = tm_arles_mother
			}
			scope:tm_arles_father = { marry = scope:tm_arles_mother }
			while = {
				count = 3
				create_character = {
					after_creation = { tm_register_seed_effect = { SEAT = arles } }
					age = { 6 12 }						# old enough for a guardian
					gender_female_chance = 50
					father = scope:tm_arles_father
					mother = scope:tm_arles_mother
					dynasty = inherit
					location = title:c_venaissin.title_province
					save_temporary_scope_as = tm_arles_child
				}
				# Raised by their own parents, or they grow up in the faith of the lord's court: in an observe run all
				# three were Christian by 901.
				scope:tm_arles_child = {
					if = {
						limit = { is_female = yes }
						set_relation_guardian = scope:tm_arles_mother
					}
					else = {
						set_relation_guardian = scope:tm_arles_father
					}
				}
				if = {
					limit = { exists = scope:tm_arles_lord }
					scope:tm_arles_lord = { add_courtier = scope:tm_arles_child }
				}
			}
			if = {
				limit = { exists = scope:tm_arles_lord }
				scope:tm_arles_lord = {
					add_courtier = scope:tm_arles_father
					add_courtier = scope:tm_arles_mother
				}
			}

			# The Worn Garland: the temple street of Kanchipuram (tm_seed_foothold_effect).
			tm_seed_foothold_effect = { COUNTY = title:c_kanchipuram FAITH = faith:tm_garland SEAT = kanchipuram }
			scope:tm_foothold_host ?= { save_scope_as = tm_kanchi_lord }
			if = {
				limit = { exists = scope:tm_kanchi_lord }
				create_character = {
					after_creation = { tm_register_seed_effect = { SEAT = kanchipuram } }
					gender = female
					age = { 40 60 }
					culture = title:c_kanchipuram.culture
					faith = faith:tm_garland
					employer = scope:tm_kanchi_lord
					dynasty = none
					trait = tm_daughter_of_the_garland
					trait = education_diplomacy_3
				}
				tm_seed_marked_courtiers_effect = { HOST = scope:tm_kanchi_lord FAITH = faith:tm_garland SEAT = kanchipuram }
			}

			# Badenya keeps Kirina (Sibi); a jelimuso who crossed sits at the court that holds it (tm_seed_foothold_effect).
			tm_seed_foothold_effect = { COUNTY = title:c_sibi_mali FAITH = faith:tm_badenya SEAT = kirina }
			scope:tm_foothold_host ?= { save_scope_as = tm_kirina_lord }
			if = {
				limit = { exists = scope:tm_kirina_lord }
				create_character = {
					after_creation = { tm_register_seed_effect = { SEAT = kirina } }
					gender = female
					age = { 40 60 }
					culture = scope:tm_kirina_lord.culture
					faith = faith:tm_badenya
					employer = scope:tm_kirina_lord
					dynasty = none
					trait = tm_daughter_of_the_house
					trait = education_diplomacy_3
				}
				tm_seed_marked_courtiers_effect = { HOST = scope:tm_kirina_lord FAITH = faith:tm_badenya SEAT = kirina }
			}

			# Historical characters are not "randomly created", so the 1% seed never rolls for them.
			# Mark ~1% of living adult men at the start, as the trait's random_creation does for later ones.
			every_living_character = {
				limit = {
					is_male = yes
					is_adult = yes
					NOT = { has_trait = tm_transgender }
				}
				random = {
					chance = 1
					add_trait = tm_transgender
				}
			}

			# Theia Beatrike also takes Phrygia, the seat, as a vassal of the same liege; the old count
			# becomes a daughter of the knife (cosmetic: no re-embodiment at game start). In 867 he also
			# holds Prusa and stays landed there; if he held nothing else he joins Beatrike's court.
			title:c_phrygia = {
				holder ?= {
					save_scope_as = tm_old_count
					# top liege, not liege: Beatrike heads a duchy-tier title, so she cannot be a duke's vassal
					top_liege ?= { save_scope_as = tm_old_liege }
				}
			}
			if = {
				limit = { exists = scope:tm_old_count }
				create_title_and_vassal_change = {
					type = granted
					save_scope_as = tm_phrygia_change
					add_claim_on_loss = no
				}
				title:c_phrygia = {
					change_title_holder = {
						holder = scope:tm_beatrike
						change = scope:tm_phrygia_change
						government_base = scope:tm_old_count
					}
				}
				if = {
					limit = {
						exists = scope:tm_old_liege
						NOT = { scope:tm_old_liege = scope:tm_old_count }
					}
					scope:tm_beatrike = {
						change_liege = {
							liege = scope:tm_old_liege
							change = scope:tm_phrygia_change
						}
					}
				}
				resolve_title_and_vassal_change = scope:tm_phrygia_change

				scope:tm_old_count = {
					add_trait = tm_daughter_of_the_knife
					if = {
						limit = { is_landed = no }
						scope:tm_beatrike = { add_courtier = scope:tm_old_count }
					}
				}
			}
			# The Mother of Mothers rules as a theocracy: in the 867 run Byzantium's administrative appointments
			# revoked Phrygia and gave it to an Orthodox governor. Untested whether this alone holds it (TO-TEST G4).
			scope:tm_beatrike = {
				change_government = theocracy_government
			}
			tm_seed_marked_courtiers_effect = { HOST = scope:tm_beatrike FAITH = faith:transbian_maximalism SEAT = phrygia }
		}
	}
}

# Arrivals: each year a ruler of the faith has a small chance that a Marked woman finds her way to the court
# (mechanical counterpart to Rescue the Daughters; the 'flight to a sister's house' of the belief list, E9).
yearly_playable_pulse = {
	on_actions = {
		tm_yearly_playable
		tm_yearly_houses
		tm_yearly_islamic_courts
		tm_yearly_turning
		tm_yearly_south
	}
}

# The Worn Garland and Badenya (tm_garland_events.txt, tm_badenya_events.txt).
tm_yearly_south = {
	effect = {
		# A name in stone: a ruler of the Garland endows the temple.
		if = {
			limit = { rite ?= { rite_has_parameter = tm_garland_gate } }
			random = {
				chance = 15
				trigger_event = { id = tm_garland.0020 days = { 1 300 } }
			}
		}
		# A pandit's objection: a Hindu lord not of the Garland with a daughter at court.
		if = {
			limit = {
				faith ?= { religion = religion:hinduism_religion }
				NOT = { rite ?= { rite_has_parameter = tm_garland_gate } }
				any_courtier = { tm_is_daughter_trigger = yes is_adult = yes }
			}
			random = {
				chance = 5
				random_courtier = {
					limit = { tm_is_daughter_trigger = yes is_adult = yes }
					save_scope_as = tm_garland_daughter
				}
				trigger_event = { id = tm_garland.0030 days = { 1 300 } }
			}
		}
		# A naming feast for a daughter of the House.
		if = {
			limit = { has_trait = tm_daughter_of_the_house }
			random = {
				chance = 30
				trigger_event = { id = tm_badenya.0020 days = { 1 300 } }
			}
		}
	}
}

# Ha-Nehpakhot (tm_hipukh_events.txt): the three lamps for a daughter of the Turning; a lord of any faith
# offered a translation; a Jewish lord asked to rule on a daughter at his court.
tm_yearly_turning = {
	effect = {
		if = {
			limit = { has_trait = tm_daughter_of_the_turning }
			random = {
				chance = 25
				trigger_event = { id = tm_hipukh.0020 days = { 1 300 } }
			}
		}
		if = {
			limit = {
				any_courtier = { has_trait = tm_daughter_of_the_turning is_adult = yes }
			}
			random = {
				chance = 5
				random_list = {
					50 = {
						random_courtier = {
							limit = { has_trait = tm_daughter_of_the_turning is_adult = yes }
							save_scope_as = tm_translator
						}
						trigger_event = { id = tm_hipukh.0030 days = { 1 300 } }
					}
					50 = {									# women physicians are attested; women translators are not
						trigger = {
							any_courtier = {
								has_trait = tm_daughter_of_the_turning is_adult = yes
								NOT = { has_character_modifier = tm_fisica_licensed_modifier }
							}
						}
						random_courtier = {
							limit = {
								has_trait = tm_daughter_of_the_turning is_adult = yes
								NOT = { has_character_modifier = tm_fisica_licensed_modifier }
							}
							save_scope_as = tm_fisica
						}
						trigger_event = { id = tm_hipukh.0031 days = { 1 300 } }
					}
				}
			}
		}
		if = {
			limit = {
				rite ?= { rite_has_parameter = jewish_syncretic_recipient }
				NOT = { rite ?= { rite_has_parameter = tm_turning_gate } }
				any_courtier = { tm_is_daughter_trigger = yes is_adult = yes }
			}
			random = {
				chance = 5
				random_courtier = {
					limit = { tm_is_daughter_trigger = yes is_adult = yes }
					save_scope_as = tm_jewish_daughter
				}
				trigger_event = { id = tm_hipukh.0040 days = { 1 300 } }
			}
		}
	}
}

# Every lord who holds a county with a Sisters' House, of any faith (tm_house_effects.txt).
tm_yearly_houses = {
	effect = {
		save_scope_as = tm_house_holder
		every_held_title = {
			limit = {
				tier = tier_county
				has_variable = tm_house
			}
			tm_house_yearly_effect = yes
		}
	}
}

tm_yearly_playable = {
	trigger = {
		is_landed = yes
		rite ?= { rite_has_parameter = tm_sisters_faith }
	}
	effect = {
		# Saves from before 0.2 kept the refuge on the house; it now belongs to the capital county.
		if = {
			limit = { house ?= { has_variable = tm_refuge } }
			house = { remove_variable = tm_refuge }
			if = {
				limit = { capital_county ?= { NOT = { has_variable = tm_house } } }
				capital_county = { tm_found_house_effect = yes }
			}
		}
		# Dorleyism in the Byzantine world (tm_byzantine_events.txt).
		if = {
			limit = { rite ?= { rite_has_parameter = tm_day_of_blood_active } }
			random = {
				chance = 4
				trigger_event = { id = tm_byz.0001 days = { 1 300 } }
			}
			# The heresy laws, rung by rung, under a liege of another Christian faith.
			if = {
				limit = {
					liege ?= {
						NOT = { faith = root.faith }
						faith = { religion_tag = christianity_religion }
					}
					NOT = { var:tm_suppression ?= 3 }
				}
				random = {
					chance = 3
					modifier = { add = 5 current_date >= 1081.1.1 }	# the Komnenian turn
					if = {
						limit = { NOT = { has_variable = tm_suppression } }
						trigger_event = { id = tm_byz.0010 days = { 1 300 } }
					}
					else_if = {
						limit = { var:tm_suppression = 1 }
						trigger_event = { id = tm_byz.0011 days = { 1 300 } }
					}
					# The Wills is Byzantine law; Slavic princes left sexual morality to the church
					# (research/cmp_levin_orthodox_slavs.md), so other lieges stop at the second rung.
					else_if = {
						limit = { liege ?= { culture ?= { has_cultural_pillar = heritage_byzantine } } }
						trigger_event = { id = tm_byz.0012 days = { 1 300 } }
					}
				}
			}
		}
		# A daughter who rules, laughed at by an Islamic neighbour (tm_isl.0002).
		if = {
			limit = {
				tm_is_daughter_trigger = yes
				any_neighboring_top_liege_realm_owner = { rite ?= { rite_has_parameter = islamic_syncretic_recipient } }
			}
			random = {
				chance = 4
				random_neighboring_top_liege_realm_owner = {
					limit = { rite ?= { rite_has_parameter = islamic_syncretic_recipient } }
					save_scope_as = tm_isl_court
				}
				trigger_event = { id = tm_isl.0002 days = { 1 300 } }
			}
		}
		# The Way of the Jade Embryo (tm_yutai_events.txt).
		if = {
			limit = { rite ?= { rite_has_parameter = tm_embryo_gate } }
			random = {
				chance = 3
				trigger_event = { id = tm_yutai.0001 days = { 1 300 } }
			}
			if = {
				limit = {
					exists = liege
					any_held_title = { tier = tier_county has_variable = tm_house NOT = { has_variable = tm_house_named } }
				}
				random = {
					chance = 1
					modifier = { add = 2 current_date >= 1264.1.1 }	# Yuan: heirless estates to the state
					modifier = { add = 1 current_date >= 1369.1.1 }	# Ming lineage law
					trigger_event = { id = tm_yutai.0010 days = { 1 300 } }
				}
			}
			if = {
				limit = {
					current_date >= 882.1.1
					current_date < 900.1.1
					NOT = { has_character_flag = tm_huang_lingwei_seen }
				}
				add_character_flag = tm_huang_lingwei_seen
				trigger_event = { id = tm_yutai.0020 days = { 1 120 } }
			}
		}
		# A völva on the winter road (tm_seidr.0020).
		if = {
			limit = {
				rite ?= { rite_has_parameter = tm_seidr_gate }
			}
			random = {
				chance = 12
				trigger_event = { id = tm_seidr.0020 days = { 1 60 } }
			}
		}
		random = {
			chance = 6
			modifier = {								# the Mother House draws them
				add = 6
				any_held_title = { this = title:c_phrygia }
			}
			modifier = {
				add = 3
				any_courtier = { tm_is_daughter_trigger = yes }
			}
			modifier = {								# a Sisters' House (tm_found_house_decision)
				add = 15
				tm_holds_sisters_house_trigger = yes
			}
			modifier = {								# the Gift Alone: word travels that no one is cut unasked
				add = 3
				rite ?= { rite_has_doctrine = tm_knife_gift_alone }
			}
			random_list = {
				60 = { trigger_event = tm_arrival.0001 }	# a Marked traveller
				25 = { trigger_event = tm_arrival.0003 }	# a woman turned out for barrenness
				15 = {										# a friend of the House
					trigger = { tm_holds_sisters_house_trigger = yes }
					trigger_event = tm_arrival.0004
				}
			}
		}
	}

# Tracking: a member of the six dies (not a body left behind by re-embodiment); a seed's death is kept on a list.
on_death = {
	on_actions = {
		tm_on_death_track
	}
}

tm_on_death_track = {
	trigger = {
		NOT = { has_variable = tm_reborn }
		OR = {
			rite ?= { rite_has_parameter = tm_sisters_faith }
			has_variable = tm_seed
		}
	}
	effect = {
		if = {
			limit = { rite ?= { rite_has_parameter = tm_sisters_faith } }
			tm_track_effect = { FAITH = faith WHAT = deaths }
		}
		if = {
			limit = { has_variable = tm_seed }
			add_to_global_variable_list = { name = tm_seeds_dead target = this }
		}
	}
}

tm_on_death_badenya = {
	trigger = {
		has_trait = tm_daughter_of_the_house
		NOT = { has_variable = tm_reborn }
		OR = {
			exists = liege
			exists = employer
		}
	}
	effect = {
		save_scope_as = tm_dead_daughter
		if = {
			limit = { exists = employer }
			employer = { trigger_event = tm_badenya.0030 }
		}
		else = {
			liege = { trigger_event = tm_badenya.0030 }
		}
	}
}

	}
}

# Growth (docs/faith_design.md §14). Once a year, for the whole world.
yearly_global_pulse = {
	on_actions = {
		tm_yearly_growth
	}
}

tm_yearly_growth = {
	effect = {
		# The call: Marked characters of other faiths may join the nearest of the six.
		every_living_character = {
			limit = {
				has_trait = tm_transgender
				is_adult = yes
				is_imprisoned = no
				NOT = { rite ?= { rite_has_parameter = tm_sisters_faith } }
			}
			tm_sisters_call_effect = yes
		}
		# The rite offered to those who are not rulers (the AI weighs decisions only for rulers).
		every_living_character = {
			limit = {
				is_male = yes
				is_adult = yes
				is_ruler = no
				is_imprisoned = no
				rite ?= { rite_has_parameter = tm_sisters_faith }
				tm_is_daughter_trigger = no
				NOT = { has_variable = tm_rite_offered }
				OR = {
					has_trait = tm_transgender
					has_trait = eunuch_1
					has_trait = beardless_eunuch
					has_variable = tm_declared
				}
			}
			random = {
				chance = 8
				modifier = { add = 17 has_trait = tm_transgender }
				modifier = { add = 10 has_variable = tm_declared }
				tm_offer_courtier_rite_effect = yes
			}
		}
		# The kahal: a Jew at a court where the sect is two or more may join it. A sect within Judaism keeps its families,
		# and their children are born into it; this is how it renews itself, not by the Marked alone.
		every_living_character = {
			limit = {
				is_adult = yes
				is_ruler = no
				is_imprisoned = no
				faith.religion = religion:judaism_religion
				NOT = { faith = faith:tm_nehpakhot }
				employer ?= {
					OR = {
						any_courtier = { count >= 2 faith = faith:tm_nehpakhot }
						AND = { faith = faith:tm_nehpakhot any_courtier = { faith = faith:tm_nehpakhot } }
					}
				}
			}
			random = {
				chance = 4
				set_character_faith = faith:tm_nehpakhot
				tm_track_effect = { FAITH = faith:tm_nehpakhot WHAT = kahal_joins }
			}
		}
		# The Mother House never forgets Phrygia: if the county leaves the faith, the Mother of Mothers holds a claim.
		if = {
			limit = {
				title:c_phrygia.holder ?= { NOT = { faith = faith:transbian_maximalism } }
				title:d_tm_mother_house.holder ?= { NOT = { has_claim_on = title:c_phrygia } }
			}
			title:d_tm_mother_house.holder = { add_pressed_claim = title:c_phrygia }
		}
		# Census for tuning: every 10 years, kept on each faith by decade (tm_tracking_effects.txt; tools/census_report.py reads a save).
		if = {
			limit = {
				OR = {
					NOT = { has_global_variable = tm_next_census }
					current_year >= global_var:tm_next_census
				}
			}
			set_global_variable = { name = tm_next_census value = { value = current_year add = 10 } }
			tm_census_log_effect = yes
		}
	}
}
