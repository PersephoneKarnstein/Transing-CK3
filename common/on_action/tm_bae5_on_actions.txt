# Crusader Dolls — Bædling: the níð-pole, the cup and the Wolf's Sermon (events/tm_bae5_events.txt). Appended to vanilla
# on_actions; the effects are our own on_actions, so vanilla's are untouched.
yearly_global_pulse = {
	on_actions = { tm_bae5_yearly }
}

on_title_gain = {
	on_actions = { tm_bae5_on_title_gain }
}

tm_bae5_yearly = {
	effect = {
		every_living_character = {
			limit = {
				is_adult = yes
				is_imprisoned = no
				rite ?= { rite_has_parameter = tm_seidr_gate }
				tm_flavour_ready_trigger = yes		# the flavour budget (tm_budget_triggers.txt)
			}
			save_scope_as = tm_bae5_me			# root is no character in the global pulse
			clear_saved_scope = tm_nid_raiser	# saved scopes outlive one character of the loop
			clear_saved_scope = tm_cup_lady
			clear_saved_scope = tm_wolf_reader
			# The Níð-Pole: a lord of the north who hates a Bædling ruler raises one against her hall. Once in ten years.
			if = {
				limit = {
					is_ruler = yes
					NOT = { has_variable = tm_bae5_nid_pole_cd }
					any_neighboring_and_across_water_top_liege_realm_owner = { tm_bae5_nid_raiser_trigger = { TARGET = scope:tm_bae5_me } }
				}
				random = {
					chance = 3
					random_neighboring_and_across_water_top_liege_realm_owner = {
						limit = { tm_bae5_nid_raiser_trigger = { TARGET = scope:tm_bae5_me } }
						weight = {
							base = 1
							modifier = { add = 4 has_relation_rival = scope:tm_bae5_me }
						}
						save_scope_as = tm_nid_raiser
					}
					set_variable = { name = tm_bae5_nid_pole_cd value = yes years = 10 }
					tm_flavour_spent_effect = yes
					trigger_event = { id = tm_bae5.0001 days = { 1 120 } }
				}
			}
			# The First Cup: the lady of the hall gives the lord the first cup, at most once in five years.
			else_if = {
				limit = {
					is_ruler = yes
					NOT = { has_variable = tm_bae5_first_cup_cd }
				}
				random = {
					chance = 4
					if = {
						limit = {
							primary_spouse ?= {
								is_female = yes
								is_adult = yes
								is_imprisoned = no
								NOT = { this = scope:tm_bae5_me }
							}
						}
						primary_spouse = { save_scope_as = tm_cup_lady }
					}
					else = {
						random_courtier = {
							limit = {
								is_female = yes
								is_adult = yes
								is_imprisoned = no
								rite ?= { rite_has_parameter = tm_seidr_gate }
							}
							weight = {
								base = 1
								modifier = { add = 4 has_relation_tm_sister = scope:tm_bae5_me }
							}
							save_scope_as = tm_cup_lady
						}
					}
					if = {
						limit = { exists = scope:tm_cup_lady }
						set_variable = { name = tm_bae5_first_cup_cd value = yes years = 5 }
						tm_flavour_spent_effect = yes
						trigger_event = { id = tm_bae5.0010 days = { 1 300 } }
					}
				}
			}
			# The Wolf's Sermon: a Bædling in an English county after 1014. Once in a life.
			else_if = {
				limit = {
					current_year >= 1014
					NOT = { has_variable = tm_bae5_wolf_heard }
					location ?= {
						county ?= {
							culture ?= {
								OR = {
									this = culture:anglo_saxon
									this = culture:english
								}
							}
							holder ?= { religion = religion:christianity_religion }
						}
					}
				}
				random = {
					chance = 3
					location.county.holder = {
						if = {
							limit = {
								cp:councillor_court_chaplain ?= { religion = religion:christianity_religion }
							}
							cp:councillor_court_chaplain = { save_scope_as = tm_wolf_reader }
						}
						else_if = {
							limit = { religion = religion:christianity_religion }
							save_scope_as = tm_wolf_reader
						}
					}
					if = {
						limit = {
							exists = scope:tm_wolf_reader
							NOT = { scope:tm_wolf_reader = scope:tm_bae5_me }
						}
						set_variable = tm_bae5_wolf_heard
						tm_flavour_spent_effect = yes
						trigger_event = { id = tm_bae5.0030 days = { 1 300 } }
					}
				}
			}
		}
	}
}

# The Bragarfull: a Bædling ruler comes into her primary title by inheritance. Root = the heir; scope:title,
# scope:previous_holder, scope:transfer_type (vanilla, title_on_actions.txt). The cup is handed by her mother, born or
# taken, else a sworn sister, else her wife; with none she lifts it herself.
tm_bae5_on_title_gain = {
	trigger = {
		scope:transfer_type ?= flag:inheritance
		scope:title = root.primary_title
		rite ?= { rite_has_parameter = tm_seidr_gate }
		NOT = { has_variable = tm_bae5_bragarfull }
	}
	effect = {
		set_variable = tm_bae5_bragarfull
		if = {
			limit = {
				any_parent = {
					is_female = yes
					is_alive = yes
					NOT = { this = scope:previous_holder }
				}
			}
			random_parent = {
				limit = {
					is_female = yes
					is_alive = yes
					NOT = { this = scope:previous_holder }
				}
				save_scope_as = tm_cup_bearer
			}
		}
		else_if = {
			limit = { any_relation = { type = tm_sister is_alive = yes } }
			random_relation = {
				type = tm_sister
				limit = { is_alive = yes }
				save_scope_as = tm_cup_bearer
			}
		}
		else_if = {
			limit = { primary_spouse ?= { is_female = yes } }
			primary_spouse = { save_scope_as = tm_cup_bearer }
		}
		trigger_event = { id = tm_bae5.0020 days = 3 }
	}
}
