# Crusader Dolls — the faiths' own offices (docs/wave5/offices.md).

yearly_global_pulse = {
	on_actions = { tm_voice_yearly_global }
}

# The Companions back a daughter for the Mother House (tm_voice_effects.txt) only once the Mother of Mothers is 60 or
# more or ill: as with a pope, the office passes when its holder grows old or dies, and no one is groomed under a
# Mother in her strength (user decision, wave 5). The effect re-targets if the backed daughter dies or leaves.
tm_voice_yearly_global = {
	trigger = {
		title:d_tm_mother_house.holder ?= { faith = faith:transbian_maximalism }
	}
	effect = {
		title:d_tm_mother_house.holder = { save_scope_as = tm_voice_mother }
		title:d_tm_mother_house = {
			if = {
				limit = {
					scope:tm_voice_mother = {
						OR = {
							age >= 60
							has_trait = ill
							has_trait = pneumonic
							has_trait = infirm
							has_trait = incapable
						}
					}
				}
				tm_voice_companions_back_daughter_effect = yes
			}
		}
	}
}
