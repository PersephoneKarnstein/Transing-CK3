# Adelphai Dorleyaiou — character interactions

# The Dorley path (Aunt Bea's House): take a violent prisoner into the programme.
# Template: vanilla castrate_interaction (00_prison_interactions.txt). Docs: docs/dorley_precedent.md.
tm_take_into_house_interaction = {
	interface_priority = 31
	icon = eunuch
	category = interaction_category_prison
	desc = tm_take_into_house_interaction_desc

	is_shown = {
		scope:actor = {
			is_adult = yes
			rite ?= { rite_has_parameter = tm_knife_penal_intake_only }
		}
		scope:recipient = {
			is_imprisoned_by = scope:actor
			is_male = yes
			is_adult = yes
			is_eunuch_trigger = no
			NOR = {
				has_trait = tm_novice
				has_trait = tm_washout
				tm_is_daughter_trigger = yes	# vanilla is_eunuch_trigger does not see daughters
			}
		}
	}

	is_valid_showing_failures_only = {
		scope:recipient = {
			custom_description = {
				text = tm_intake_not_eligible
				tm_intake_eligible_trigger = yes
			}
			custom_description = {
				text = "currently_being_tortured"
				NOT = { has_character_flag = is_being_tortured }
			}
		}
	}

	auto_accept = yes

	on_accept = {
		scope:recipient = {
			tm_begin_programme_effect = { CAPTOR = scope:actor }
		}
	}

	ai_targets = { ai_recipients = prisoners }
	ai_frequency = 24
	is_available = {
		rite ?= { rite_has_parameter = tm_knife_penal_intake_only }
	}
	ai_will_do = {
		base = 25
		modifier = { add = 25 has_trait = just }
		modifier = { add = -50 has_trait = compassionate }
	}
}

# Sister-Making: two daughters bind themselves by the old rite.
tm_sister_making_interaction = {
	interface_priority = 60
	icon = icon_scheme_befriend
	category = interaction_category_friendly
	desc = tm_sister_making_interaction_desc

	is_shown = {
		NOT = { scope:actor = scope:recipient }
		scope:actor = {
			tm_is_daughter_trigger = yes
			rite ?= { rite_has_parameter = tm_sister_making_allowed }
		}
		scope:recipient = {
			tm_is_daughter_trigger = yes
			faith = scope:actor.faith
		}
	}

	is_valid_showing_failures_only = {
		scope:actor = { NOT = { has_relation_tm_sister = scope:recipient } }
		scope:recipient = { is_imprisoned = no }
		scope:actor = { is_imprisoned = no }
	}

	cost = { piety = 50 }

	on_accept = {
		scope:actor = {
			set_relation_tm_sister = scope:recipient
			add_piety = 100
			add_prestige = 50
		}
		scope:recipient = {
			add_piety = 100
			add_prestige = 50
		}
	}

	ai_accept = {
		base = 0
		opinion_modifier = {
			who = scope:recipient
			opinion_target = scope:actor
			multiplier = 1
		}
		modifier = { add = 25 scope:recipient = { has_trait = trusting } }
	}

	ai_targets = { ai_recipients = courtiers }
	ai_targets = { ai_recipients = guests }
	ai_frequency = 60
	is_available = { tm_is_daughter_trigger = yes }
	ai_will_do = {
		base = 10
		opinion_modifier = {
			opinion_target = scope:recipient
			multiplier = 0.5
		}
	}
}

# DEBUG (only with -debug_mode): put any male prisoner through the whole programme at once.
tm_debug_force_programme_interaction = {
	interface_priority = 1
	icon = eunuch
	category = interaction_category_prison
	desc = tm_debug_force_programme_interaction_desc
	is_shown = {
		debug_only = yes
		scope:recipient = {
			is_imprisoned_by = scope:actor
			is_male = yes
			NOT = { tm_is_daughter_trigger = yes }
		}
	}
	auto_accept = yes
	on_accept = {
		scope:recipient = {
			release_from_prison = yes
			if = {
				limit = { has_trait = tm_novice }
				remove_trait = tm_novice
			}
			tm_take_the_knife_effect = yes
		}
	}
}

# Take as Daughter: the Sisters' own adoption (Daughters Chosen tenet). Unlike vanilla adopt_interaction
# (00_adoption.txt) it takes adults as well as children, and drops vanilla's "no ruling relatives" and
# "liege shares her dynasty" rules, which make it nearly impossible to keep a matriline going.
# The adoptee joins the adopter's house (vanilla adopt_effect), so she inherits under the realm's law.
tm_adopt_daughter_interaction = {
	interface_priority = 61
	icon = icon_dynasty
	category = interaction_category_diplomacy
	desc = tm_adopt_daughter_interaction_desc
	use_diplomatic_range = yes
	cooldown = { years = 1 }

	is_shown = {
		NOT = { scope:actor = scope:recipient }
		scope:actor = {
			is_adult = yes
			rite ?= { rite_has_parameter = tm_adoptive_matriline }
		}
		scope:recipient = {
			OR = {
				is_female = yes
				has_trait = tm_transgender
				tm_is_daughter_trigger = yes
			}
			NOR = {
				is_child_of = scope:actor
				is_parent_of = scope:actor
				is_spouse_of = scope:actor
				is_consort_of = scope:actor
			}
		}
	}

	is_valid_showing_failures_only = {
		scope:recipient = {
			is_ruler = no				# a landed adoptee would drag her own realm into the succession
			is_imprisoned = no
			NOT = { any_ancestor = { this = scope:actor } }
		}
		scope:actor = { is_imprisoned = no }
	}

	cost = {
		prestige = {
			value = 100
			if = {
				limit = { scope:recipient = { OR = { is_courtier_of = scope:actor has_relation_guardian = scope:actor } } }
				multiply = 0.5
			}
			if = {	# already fostered by the actor: this is only the written clause naming her heir
				limit = { scope:recipient = { has_variable = tm_fostered_by var:tm_fostered_by = scope:actor } }
				multiply = 0.5
			}
		}
	}

	# Your own courtiers and wards come willingly; others are asked.
	auto_accept = {
		scope:recipient = {
			OR = {
				is_courtier_of = scope:actor
				has_relation_guardian = scope:actor
			}
		}
	}

	on_accept = {
		scope:recipient = {
			if = {
				limit = { has_character_modifier = tm_fostered_modifier }
				remove_character_modifier = tm_fostered_modifier
				remove_variable = tm_fostered_by
			}
		}
		adopt_effect = {
			CHILD = scope:recipient
			ADOPTER = scope:actor
		}
		scope:recipient = {
			add_opinion = { target = scope:actor modifier = grateful_opinion opinion = 30 }
		}
		scope:actor = {
			add_piety = 25
		}
	}

	ai_accept = {
		base = 0
		modifier = {
			add = 40
			desc = tm_adopt_ai_marked
			scope:recipient = { OR = { has_trait = tm_transgender tm_is_daughter_trigger = yes } }
		}
		modifier = {
			add = 20
			desc = tm_adopt_ai_same_faith
			scope:recipient.faith = scope:actor.faith
		}
		modifier = {
			add = 25
			desc = tm_adopt_ai_lowborn
			scope:recipient = { NOT = { exists = house } }
		}
		modifier = {
			add = 30
			desc = tm_adopt_ai_cast_out
			scope:recipient = { has_trait = infertile is_female = yes }
		}
		modifier = {
			add = -50
			desc = tm_adopt_ai_has_mother
			scope:recipient = { exists = mother mother = { is_alive = yes } }
		}
		opinion_modifier = {
			who = scope:recipient
			opinion_target = scope:actor
			multiplier = 1
		}
	}

	ai_targets = { ai_recipients = courtiers }
	ai_targets = { ai_recipients = guests }
	ai_frequency = 36
	is_available = {
		is_adult = yes
		rite ?= { rite_has_parameter = tm_adoptive_matriline }
	}
	ai_will_do = {
		base = 0
		modifier = {	# no living daughter to carry the line
			add = 60
			scope:actor = { NOT = { any_child = { is_alive = yes is_female = yes } } }
		}
		modifier = {
			add = 20
			scope:recipient = { tm_is_daughter_trigger = yes }
		}
		modifier = {	# already plenty of children
			add = -100
			scope:actor = { any_child = { count >= 3 is_alive = yes } }
		}
	}
}

# Teach the Warding Songs: the varðlokkur pass from foster-mother to foster-daughter, as Halldis taught
# Guðríðr (Eiríks saga rauða ch. 4). A singer who knows them makes the seiðr rite far likelier to succeed.
tm_teach_vardlokkur_interaction = {
	interface_priority = 55
	icon = icon_scheme_befriend
	category = interaction_category_friendly
	desc = tm_teach_vardlokkur_interaction_desc

	is_shown = {
		NOT = { scope:actor = scope:recipient }
		scope:actor = {
			tm_knows_vardlokkur_trigger = yes
			rite ?= { rite_has_parameter = tm_seidr_gate }
		}
		scope:recipient = {
			NOT = { tm_knows_vardlokkur_trigger = yes }
			age >= 8
			OR = {
				is_female = yes
				has_trait = tm_transgender
			}
			OR = {
				is_child_of = scope:actor
				has_relation_guardian = scope:actor
			}
		}
	}

	is_valid_showing_failures_only = {
		scope:recipient = { is_imprisoned = no }
		scope:actor = { is_imprisoned = no }
	}

	auto_accept = yes

	on_accept = {
		scope:recipient = {
			set_variable = tm_knows_vardlokkur
			add_learning_lifestyle_xp = 100
		}
		scope:actor = { add_piety = 25 }
	}

	ai_targets = { ai_recipients = family }
	ai_targets = { ai_recipients = courtiers }
	ai_frequency = 36
	ai_will_do = { base = 50 }
}

				has_trait = tm_transgender
				tm_is_daughter_trigger = yes
			}
			NOT = { has_character_modifier = tm_fostered_modifier }
			NOR = {
				is_child_of = scope:actor
				is_parent_of = scope:actor
				is_spouse_of = scope:actor
				is_consort_of = scope:actor

	ai_frequency = 36
	ai_will_do = {
		base = 10
		modifier = { add = 30 scope:recipient = { tm_is_daughter_trigger = yes } }
	}
}

# Name a Companion: the Montanist koinonoi stood between the patriarch and the bishops (research/byz_trevett_montanism.md).
# The Mother of Mothers names up to seven Companions among the daughters of the faith.
tm_name_companion_interaction = {
	interface_priority = 58
	icon = icon_scheme_befriend
	category = interaction_category_religion
	desc = tm_name_companion_interaction_desc
	use_diplomatic_range = no

	is_shown = {
		NOT = { scope:actor = scope:recipient }
		scope:actor = {
			has_title = title:d_tm_mother_house
		}
		scope:recipient = {
			faith = scope:actor.faith
			is_adult = yes
			tm_is_daughter_trigger = yes
			NOT = { has_character_modifier = tm_companion_modifier }
		}
	}

	is_valid_showing_failures_only = {
		scope:recipient = { is_imprisoned = no }
		custom_description = {
			text = tm_companions_full
			NOT = { any_living_character = { count >= 7 has_character_modifier = tm_companion_modifier } }
		}
	}

	cost = { piety = 100 }
	auto_accept = yes

	on_accept = {
		scope:recipient = {
			add_character_modifier = { modifier = tm_companion_modifier }
			add_opinion = { target = scope:actor modifier = grateful_opinion opinion = 25 }
			add_piety = 50
		}
	}

	ai_targets = { ai_recipients = courtiers }
	ai_targets = { ai_recipients = vassals }
	ai_frequency = 60
	ai_will_do = { base = 25 }
}
