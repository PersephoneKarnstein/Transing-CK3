# Crusader Dolls — title succession laws.
# The Sisters Choose: an elective law for the Sisters' faiths. CK3 needs an heir, and adoption stays the
# succession engine; this law lets the House's women and the realm's vassals choose among the daughters
# instead (Raha & van der Drift, research/book_raha_vdd_trans_femme_futures.md row 3).
# Voting logic: common/succession_election/tm_sisters_elective.txt.
tm_sisters_elective_succession_law = {
	law_group_type = title_succession_laws
	index = 30

	can_have = {
		OR = {
			government_has_flag = government_is_feudal
			government_has_flag = government_is_clan
			government_has_flag = government_is_tribal
		}
		highest_held_title_tier >= tier_duchy
	}
	can_pass = {
		can_change_title_law_trigger = yes
		rite ?= { rite_has_parameter = tm_sisters_faith }
	}
	can_title_have = {
		can_title_have_law_general_trigger = yes
	}
	succession = {
		order_of_succession = election
		election_type = tm_sisters_elective
	}
	flag = elective_succession_law
	title_allegiance_opinion = 5
	revoke_cost = {
		prestige = change_title_succession_law_prestige_cost
	}
	pass_cost = {
		prestige = change_title_succession_law_prestige_cost
	}
	# Vanilla AI may not weigh title laws at all; tm_succ_on_actions.txt takes the law up for AI rulers either way.
	ai_will_do = {
		value = 0
		if = {
			limit = { tm_succ_has_daughter_candidate_trigger = yes }
			add = 10
		}
	}
}
