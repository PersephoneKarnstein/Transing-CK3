# The seed: "Marked by the Mother". Sisters are born of any womb. birth/random_creation require genetic = yes,
# so the mark is technically genetic (recessive inheritance); at 1.5% the bloodline effect is negligible.
# Raised from 1% to 1.5% (user, 2026-10-21): Marked and daughters stayed in single figures per faith in every run.
# Visible on purpose: the vanilla Character Finder's trait filter can then find marked characters.
tm_transgender = {
	genetic = yes
	valid_sex = male					# trans women only; trans-masculine content is out of scope (design lock §open questions)
	birth = 1.5
	random_creation = 1.5

	same_opinion = 10					# t4t: the marked recognise one another
	shown_in_ruler_designer = yes
	ruler_designer_cost = 0

	ai_boldness = 5
}

# The result of the Rite of the Knife. Separate from vanilla castration: vanilla castrate_interaction
# still gives eunuch_1/beardless_eunuch and never makes a daughter. Note vanilla is_eunuch_trigger checks
# only those two traits, so faith script must test tm_daughter_of_the_knife explicitly. Unlike eunuch_1 it blocks neither marriage nor inheritance, and it is valid for either sex
# (a re-embodied daughter is female to the engine).
tm_daughter_of_the_knife = {
	icon = beardless_eunuch.dds	# placeholder

	category = health
	group_equivalence = eunuch
	level = 3							# eunuch_1 is level 1, beardless_eunuch level 2
	can_have_children = no
	minimum_age = 16
	opposites = {
		eunuch_1
		beardless_eunuch
		tm_daughter_of_seidr
		tm_daughter_of_the_embryo
		tm_daughter_of_the_turning
			tm_daughter_of_the_garland
		tm_daughter_of_the_house
	}

	same_opinion = 15
	stewardship = 1
	learning = 1

	culture_modifier = {
		parameter = eunuch_trait_bonuses		# Court Eunuchs tradition: the vanilla bonuses also apply
		liege_opinion = 10
		stewardship = 1
		intrigue = 1
	}

	# Islamic courts value her as a servant and despise her as a ruler (tm_islamic_events.txt).
	triggered_opinion = {
		parameter = islamic_syncretic_recipient
		opinion_modifier = tm_islamic_daughter_opinion
	}
	# Indic law reads the knife's daughter as a castrated man; Buddhists as a made paṇḍaka (research/sea_gyatso_cabezon_buddhist_law.md).
	triggered_opinion = {
		parameter = eastern_syncretic_recipient
		opinion_modifier = tm_eastern_knife_opinion
	}
	# Jewish courts: "what came forth male is male" (research/jud_neis_raven.md); Ha-Nehpakhot answers it.
	triggered_opinion = {
		parameter = jewish_syncretic_recipient
		opinion_modifier = tm_jewish_daughter_opinion
	}
	triggered_opinion = {
		parameter = tm_turning_gate
		opinion_modifier = tm_turned_sister_opinion
	}
	triggered_opinion = {
		parameter = tm_garland_gate
		opinion_modifier = tm_garland_sister_opinion
	}

	shown_in_ruler_designer = yes
	ruler_designer_cost = 0
}
