# The seed: "Marked by the Mother". Sisters are born of any womb. birth/random_creation require genetic = yes,
# so the mark is technically genetic (recessive inheritance); at 1% the bloodline effect is negligible.
# Visible on purpose: the vanilla Character Finder's trait filter can then find marked characters.
tm_transgender = {
	genetic = yes
	valid_sex = male					# trans women only; trans-masculine content is out of scope (design lock §open questions)
	birth = 1
	random_creation = 1

	same_opinion = 10					# t4t: the marked recognise one another
	shown_in_ruler_designer = yes
	ruler_designer_cost = 0

	ai_boldness = 5
}

# The result of the Rite of the Knife. Separate from vanilla castration: vanilla castrate_interaction
# still gives eunuch_1/beardless_eunuch and never makes a daughter. Note vanilla is_eunuch_trigger checks
# only those two traits, so faith script must test tm_daughter_of_the_knife explicitly. Unlike eunuch_1 it blocks neither marriage nor inheritance, and it is valid for either sex
# (a re-embodied daughter is female to the engine).
tm_daughter_of_the_knife = {
	icon = beardless_eunuch.dds	# placeholder

	category = health
	group_equivalence = eunuch
	can_have_children = no
	minimum_age = 16
	opposites = {
		eunuch_1
		beardless_eunuch
	}

	same_opinion = 15
	stewardship = 1
	learning = 1

	culture_modifier = {
		parameter = eunuch_trait_bonuses		# Court Eunuchs tradition: the vanilla bonuses also apply
		liege_opinion = 10
		stewardship = 1
		intrigue = 1
	}

	shown_in_ruler_designer = yes
	ruler_designer_cost = 0
}
