# Crusader Dolls — Rescue the Daughters (design lock §0; user 2026-10-14). Single county.
# Spec: game/common/casus_belli_types/_casus_belli.info. Template: minor_religious_war (00_religious_war.txt),
# pared down.
# Two causes (user 2026-10-21: captives alone were too rare for the war ever to be declared):
#   - a county where daughters, novices or Marked women of the attacker's faith live under a lord of another faith
#     (tm_county_shelters_daughters_trigger). Daughters cluster at a few courts, so this is not a war on everyone;
#   - the defender holds such a woman prisoner (any neighbouring county of the defender's).
# Victory takes the county; the women of the attacker's faith living there, and every captive the defender holds,
# come to the attacker's court.
tm_rescue_daughters_cb = {
	icon = minor_religious_war
	group = religious
	combine_into_one = yes
	should_show_war_goal_subview = yes
	mutually_exclusive_titles = { always = yes }

	allowed_for_character = {
		scope:attacker.rite ?= { rite_has_parameter = tm_rescue_daughters_cb }
		NOR = {
			government_has_flag = government_is_landless_adventurer
			government_has_flag = government_is_nomadic
		}
	}

	allowed_against_character = {
		NOT = { scope:defender.faith = scope:attacker.faith }
		scope:defender = {
			OR = {
				any_prisoner = { tm_captive_daughter_trigger = yes }
				any_realm_county = { tm_county_shelters_daughters_trigger = { FAITH = scope:attacker.faith } }
			}
		}
	}

	target_titles = neighbor_land_or_water
	target_title_tier = county
	ignore_effect = change_title_holder

	valid_to_start = {
		scope:target = {
			tier = tier_county
			is_landless_type_title = no
			holder = {
				OR = {
					this = scope:defender
					target_is_liege_or_above = scope:defender
				}
			}
		}
		OR = {
			scope:target = { tm_county_shelters_daughters_trigger = { FAITH = scope:attacker.faith } }
			scope:defender = { any_prisoner = { tm_captive_daughter_trigger = yes } }
		}
	}

	on_invalidated_desc = { desc = tm_rescue_daughters_cb_invalidated_desc }
	on_invalidated = { }
	# Only when nobody is left to rescue: one woman still in the county is enough to carry on.
	should_invalidate = {
		NOT = { scope:defender = { any_prisoner = { tm_captive_daughter_trigger = yes } } }
		NOT = {
			any_in_list = {
				list = target_titles
				any_county_province = {
					any_character_in_location = {
						faith = scope:attacker.faith
						is_ruler = no
						tm_captive_daughter_trigger = yes
					}
				}
			}
		}
	}

	cost = {
		piety = 100
	}

	on_victory_desc = { desc = tm_rescue_daughters_cb_victory_desc }
	on_victory = {
		# Gather the women of the attacker's faith in the county before the county changes hands.
		every_in_list = {
			list = target_titles
			every_county_province = {
				every_character_in_location = {
					limit = {
						faith = scope:attacker.faith
						is_ruler = no
						tm_captive_daughter_trigger = yes
					}
					add_to_list = tm_rescued
				}
			}
		}
		create_title_and_vassal_change = {
			type = conquest_holy_war
			save_scope_as = change
			add_claim_on_loss = yes
		}
		every_in_list = {
			list = target_titles
			change_title_holder = {
				holder = scope:attacker
				change = scope:change
				take_baronies = yes
			}
		}
		resolve_title_and_vassal_change = scope:change

		scope:defender = {
			every_prisoner = {
				limit = { tm_captive_daughter_trigger = yes }
				add_to_list = tm_rescued
			}
		}
		every_in_list = {
			list = tm_rescued
			if = {
				limit = { is_imprisoned = yes }
				release_from_prison = yes
			}
			scope:attacker = { add_courtier = prev }
		}
		scope:attacker = {
			add_piety = 200
			add_prestige = 100
		}
	}

	on_white_peace_desc = { desc = tm_rescue_daughters_cb_white_peace_desc }
	on_white_peace = {
		scope:attacker = { add_prestige = -50 }
	}

	on_defeat_desc = { desc = tm_rescue_daughters_cb_defeat_desc }
	on_defeat = {
		scope:attacker = {
			add_prestige = -150
			add_piety = -100
		}
	}

	on_primary_attacker_death = inherit
	on_primary_defender_death = inherit
	transfer_behavior = transfer
	attacker_allies_inherit = yes
	defender_allies_inherit = yes

	war_name = "TM_RESCUE_DAUGHTERS_WAR_NAME"
	cb_name = "TM_RESCUE_DAUGHTERS_CB_NAME"
	is_holy_war = yes
	interface_priority = 80
	max_ai_diplo_distance_to_title = 500
}
