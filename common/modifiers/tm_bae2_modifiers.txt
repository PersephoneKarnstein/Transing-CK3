# Crusader Dolls — Bædling stage 2 modifiers (docs/wave3/baedling.md).

# Lesser outlawry (fjörbaugsgarðr): three winters abroad, then "as free from penalty as if that outlawry had never
# befallen" her (Grágás K §§51-53, research/norse_gragas_laws.md §1.3). Given for 3 years; when it lapses she is free.
tm_bae2_lesser_outlaw_modifier = {
	icon = prestige_negative
	monthly_prestige = -0.2
	stress_gain_mult = 0.1
}

# Full outlawry (skóggangr): no standing at law, killable by anyone (Grágás; Ahola pp. 85-86). Lasts until a reprieve:
# a settlement (Miller pp. 262-77), or the high seat (Ahola p. 401). The woods make hard people (Ahola p. 256).
tm_bae2_full_outlaw_modifier = {
	icon = prestige_negative
	monthly_prestige = -0.5
	prowess = 2
	stress_gain_mult = 0.15
}

# A band of the six faiths has taken land: the first years in the hall (docs/wave3/baedling.md §3). Character
# modifier, 10 years. Vanilla gives a new adventurer realm nothing of the kind, only the ennobled-adventurer malus.
tm_bae2_settling_modifier = {
	icon = martial_positive
	defender_advantage = 4
	garrison_size = 0.25
	levy_reinforcement_rate = 0.25
	short_reign_duration_mult = -0.5
}

# The people of this county kept the faith (or its House, or it is the seat) and their own came home. County

tm_bae2_hall_returned_county_modifier = {
	icon = county_modifier_opinion_positive
	county_opinion_add = 15
	levy_size = 0.2
	monthly_county_control_growth_add = 0.5
}
