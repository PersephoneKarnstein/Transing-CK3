tm_take_the_knife_decision = {
	picture = {
		reference = "gfx/interface/illustrations/decisions/decision_misc.dds"		# placeholder; TODO event art
	}
	sort_order = 90

	is_shown = {
		is_male = yes
		is_adult = yes
		rite ?= { rite_has_parameter = tm_knife_voluntary }
		tm_is_daughter_trigger = no		# eunuchs may come too: the rite recognises what the court's knife did not
	}

	is_valid_showing_failures_only = {
		is_imprisoned = no
		is_at_war = no
	}

	cost = {
		piety = {
			value = 100
			if = {
				limit = { has_personal_tenet_flag = tm_personal_knife_half_cost }
				multiply = 0.5
			}
		}
	}

	effect = {
		trigger_event = tm_knife.0001
	}

	ai_check_interval = 120
	ai_potential = {
		OR = { has_trait = tm_transgender has_trait = eunuch_1 has_trait = beardless_eunuch }
	}
	ai_will_do = {
		base = 50
		modifier = { add = -35 NOT = { has_trait = tm_transgender } }
		modifier = { add = 25 has_trait = brave }
		modifier = { add = -40 has_trait = craven }
	}
}

# DEBUG (only with -debug_mode): count living Marked and daughters worldwide and name one of each.
tm_debug_census_decision = {
	picture = { reference = "gfx/interface/illustrations/decisions/decision_misc.dds" }
	sort_order = 2
	is_shown = { debug_only = yes }
	effect = {
		set_variable = { name = tm_census_marked value = 0 }
		set_variable = { name = tm_census_daughters value = 0 }
		every_living_character = {
			limit = { has_trait = tm_transgender }
			root = { change_variable = { name = tm_census_marked add = 1 } }
		}
		every_living_character = {
			limit = { tm_is_daughter_trigger = yes }
			root = { change_variable = { name = tm_census_daughters add = 1 } }
		}
		random_living_character = {
			limit = { has_trait = tm_transgender }
			save_scope_as = tm_census_example_marked
		}
		random_living_character = {
			limit = { tm_is_daughter_trigger = yes }
			save_scope_as = tm_census_example_daughter
		}
		trigger_event = tm_knife.0900
	}
	ai_check_interval = 0
}

# Found a Sisters' House in your capital county. The House belongs to the county, not to the ruler's
# family (Raha & van der Drift: shelter granted by a lord can be taken away, so let it outlive the lord).
# Whoever holds the county later chooses to tolerate it or close it (tm_house_events.txt). The House draws
# Marked travellers, unlocks the seek decision below, and runs itself: an Elder Sister on a five-year term,
# strain, and splitting into a daughter House (tm_house_events.txt).
tm_found_house_decision = {
	picture = {
		reference = "gfx/interface/illustrations/decisions/decision_misc.dds"		# placeholder; TODO event art
	}
	sort_order = 85

	is_shown = {
		is_landed = yes
		rite ?= { rite_has_parameter = tm_sisters_faith }
		exists = capital_county
		NOT = { capital_county = { has_variable = tm_house } }
	}

	is_valid_showing_failures_only = {
		is_imprisoned = no
	}

	cost = {
		piety = 150
		prestige = 100
	}

	effect = {
		capital_county = { save_scope_as = tm_house_county }
		scope:tm_house_county = { tm_found_house_effect = yes }
		custom_tooltip = tm_found_house_decision_effect_tt
		trigger_event = tm_refuge.0001
	}

	ai_check_interval = 60
	ai_will_do = {
		base = 25
		modifier = { add = 50 NOT = { any_child = { is_alive = yes is_female = yes } } }
		modifier = { add = 25 rite ?= { rite_has_doctrine = tm_knife_gift_alone } }
	}
}
