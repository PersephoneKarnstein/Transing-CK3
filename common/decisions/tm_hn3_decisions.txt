# Crusader Dolls — Ha-Nehpakhot in the Maghreb and the Sahel (events/tm_hn3_events.txt; docs/wave3/ha-nehpakhot-maghreb.md).

# The Decree of the Two Parties: a ruler of the sect in the Maghreb or the Sahel decrees that neither party of the
# Jews shall ban the other (after al-Zahir's decree of 1030: Rustow ch. 8; Stillman p. 198). The same choice comes to
# her by event when a daughter is shunned (tm_hn3.0020). It lasts her lifetime.
tm_hn3_two_parties_decision = {
	picture = {
		reference = "gfx/interface/illustrations/decisions/decision_misc.dds"		# placeholder; TODO event art
	}
	sort_order = 80

	is_shown = {
		faith = faith:tm_nehpakhot
		is_landed = yes
		tm_hn3_capital_in_west_trigger = yes
		NOT = { has_variable = tm_hn3_two_parties }
	}

	is_valid_showing_failures_only = {
		is_imprisoned = no
	}

	is_valid = {
		custom_description = {
			text = tm_hn3_two_parties_decision_needs_jews
			OR = {
				any_courtier = {
					faith.religion = religion:judaism_religion
					NOT = { faith = faith:tm_nehpakhot }
				}
				any_vassal = {
					faith.religion = religion:judaism_religion
					NOT = { faith = faith:tm_nehpakhot }
				}
			}
		}
	}

	effect = {
		custom_tooltip = tm_hn3_two_parties_decision_effect_tt
		tm_hn3_two_parties_effect = yes
	}

	ai_check_interval = 120
	ai_potential = {
		any_courtier = { has_character_modifier = tm_niddui_modifier }
	}
	ai_will_do = {
		base = 50
		modifier = { add = 20 has_trait = just }
		modifier = { add = -30 has_trait = zealous }
	}
}
