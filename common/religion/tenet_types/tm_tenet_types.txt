# Adelphai Dorleyaiou — core tenets
# Spec: game/common/religion/tenet_types/_tenet_types.info
# Exclusive to the Sisters and their offshoots: shown only to faiths already carrying tm_day_of_blood_active.
# Icons reuse vanilla tenet icons until custom art exists.

# The rite of the knife; the cut become daughters. Parameter tm_day_of_blood_active gates the decision,
# the prisoner interaction and the spring festival (all in the faith's own script).
tenet_tm_day_of_blood = {
	icon = tenet_mortification_of_the_flesh

	is_shown = {
		this = faith:transbian_maximalism		# 1.20 doctrines live on the rite, so a faith-level parameter check is always false
	}

	piety_cost = {
		value = faith_tenet_cost_high
		if = {
			limit = { rite_has_tenet = tenet_tm_day_of_blood }
			multiply = faith_unchanged_doctrine_cost_mult
		}
	}

	parameters = {
		tm_day_of_blood_active
	}

	character_modifier = {
		monthly_piety_gain_mult = 0.05
	}

	# Personal tenet (1.20 game-start picker and later choice). Reuses a vanilla flag that is checked by name.
	can_pick_as_personal_tenet = {
		faith = faith:transbian_maximalism
	}
	personal_tenet_modifier = {
		stress_loss_mult = 0.15
		health = 0.15
	}
	personal_tenet_parameters = {
		tm_personal_knife_half_cost					# tm_decisions.txt: the Rite of the Knife costs half piety
		tenet_mortification_suffering_sanctifies	# vanilla 20_health_effects.txt: piety and stress loss on wounds
	}

	traits = {
		virtues = {
			eunuch_1 = { weight = 20 scale = 1 }
			beardless_eunuch = { weight = 20 scale = 1 }
			tm_daughter_of_the_knife = { weight = 30 scale = 1 }
			brave = { weight = 10 scale = 1 }
		}
		sins = {
			craven = { weight = 10 scale = 1 }
		}
	}
}

		}
