namespace = tm_embryo

# The guarantor's question. Root = the adept.
tm_embryo.0001 = {
	type = character_event
	title = tm_embryo.0001.t
	desc = tm_embryo.0001.desc
	theme = learning
	left_portrait = { character = root animation = personality_rational }

	option = {	# "I will not repent."
		name = tm_embryo.0001.a
		set_variable = tm_embryo_active
		add_character_modifier = { modifier = tm_gestating_modifier days = 300 }
		custom_tooltip = tm_embryo.0001.a.tt
		trigger_event = { id = tm_embryo.0010 days = { 80 100 } }
		random = {				# a physician hears of it (research/cn_cheng_divine_demonic_disordered.md)
			chance = 35
			trigger_event = { id = tm_embryo.0013 days = { 120 160 } }
		}
	}
	option = {	# Not yet
		name = tm_embryo.0001.b
		# Rule 23 (round 2): refund what the decision cost; a courtier offered the rite paid nothing.
		if = {
			limit = { NOT = { has_variable = tm_rite_offered } }
			if = {
				limit = { has_personal_tenet_flag = tm_personal_embryo_easier }
				add_piety = 25
			}
			else = { add_piety = 50 }
		}
	}
}

# Third month: the first bathing, and the censor at the gate. The danger is outside her: Tang censors
# checked the clergy roll for unregistered "private" nuns (research/cn_hinsch_women_tang.md); struck off it,
# she would be counted among the palace's men. A failure delays the gestation; it never resets it
# (Serano: the gate recognises, it does not test). Root = the adept.
tm_embryo.0010 = {
	type = character_event
	title = tm_embryo.0010.t
	desc = tm_embryo.0010.desc
	theme = learning
	left_portrait = { character = root animation = personality_rational }
	trigger = { has_variable = tm_embryo_active }

	option = {	# Hold the stillness; let the censor wait at the gate
		name = tm_embryo.0010.a
		random_list = {
			70 = {
				modifier = { add = 15 any_relation = { type = tm_sister tm_is_daughter_trigger = yes } }	# her sworn sister stands as her sponsor (round 2, T21)
				modifier = { add = { value = learning multiply = 2 } }
				modifier = { add = 20 has_personal_tenet_flag = tm_personal_embryo_easier }
				modifier = { add = 10 tm_voice_my_court_has_office_trigger = { POS = tm_voice_society_head_court_position } }	# the society stands by her (tm_voice_court_positions.txt)
				modifier = { add = 20 has_character_modifier = tm_taught_answers_modifier }	# Teach Her the Answers (tm_mech_interactions.txt)
				desc = tm_embryo.trial_kept
				trigger_event = { id = tm_embryo.0011 days = { 80 100 } }
			}
			30 = {
				desc = tm_embryo.trial_delayed
				tm_embryo_delayed_effect = { EVENT = tm_embryo.0010 STRESS = 10 }
			}
		}
	}
	option = {	# The sisters of the guan stand at the gate for me
		name = tm_embryo.0010.b
		trigger = {
			any_courtier = {
				NOT = { this = root }
				OR = { tm_is_daughter_trigger = yes has_trait = tm_transgender }
				faith = root.faith
			}
		}
		trigger_event = { id = tm_embryo.0011 days = { 80 100 } }
	}
	option = {	# A friend in the inner offices has my name taken off the inquiry (palace faction stage 2, tm_yt2.0001 option e:
				# the inner offices hold licences at this throne; eunuch Commissioners of Merit oversaw both clergies,
				# Rideout 1949, p. 56)
		name = tm_embryo.0010.e
		trigger = {
			top_liege ?= {
				NOT = { this = root }
				primary_title = { has_variable = tm_yt2_licences }
			}
		}
		trigger_event = { id = tm_embryo.0011 days = { 80 100 } }
	}
	option = {	# Pay the censor to forget the road to my door
		name = tm_embryo.0010.c
		trigger = { gold >= 30 }
		remove_short_term_gold = 30
		trigger_event = { id = tm_embryo.0011 days = { 80 100 } }
	}
	option = {	# Set the work down for now
		name = tm_embryo.0010.d
		tm_embryo_set_down_effect = yes
	}
}

# Sixth month: the White Tiger, and a letter from home. "Red dragon" is a later (1310, Qing) term; the in-period
# image for the work in a male-born body is the White Tiger (research/cn_valussi_*.md, cn_wang_mu_internal_alchemy.md).
# Her family summons her back to marry and keep the ancestral rites. Root = the adept.

	type = character_event
	title = tm_embryo.0011.t
	desc = tm_embryo.0011.desc
	theme = learning
	left_portrait = { character = root animation = personality_zealous }
	trigger = { has_variable = tm_embryo_active }

	option = {	# Turn the work, and leave the letter unanswered
		name = tm_embryo.0011.a
		random_list = {
			75 = {
				modifier = { add = { value = learning multiply = 2 } }
				modifier = { add = 10 tm_voice_my_court_has_office_trigger = { POS = tm_voice_society_head_court_position } }	# the society stands by her (tm_voice_court_positions.txt)
				desc = tm_embryo.trial_kept
				trigger_event = { id = tm_embryo.0012 days = { 110 130 } }
			}
			25 = {
				desc = tm_embryo.trial_delayed
				tm_embryo_delayed_effect = { EVENT = tm_embryo.0011 STRESS = 10 }
			}
		}
	}
	option = {	# "Not yet! Don't rush me!" (Xue Xuantong, in Du Guangting)
		name = tm_embryo.0011.d
		add_stress = 10
		add_prestige = 25
		trigger_event = { id = tm_embryo.0012 days = { 110 130 } }
	}
	option = {	# I will nurse my parents first, then return to the work
		name = tm_embryo.0011.e
		trigger = { any_parent = { is_alive = yes } }
		add_piety = 50
		every_parent = {
			limit = { is_alive = yes }
			add_opinion = { target = root modifier = grateful_opinion opinion = 20 }
		}
		tm_embryo_delayed_effect = { EVENT = tm_embryo.0011 STRESS = 0 }	# her own choice: no stress
	}
	option = {	# My guarantor answers for the house that shelters me
		name = tm_embryo.0011.b
		trigger = { piety >= 50 }
		add_piety = -50
		trigger_event = { id = tm_embryo.0012 days = { 110 130 } }
	}
	option = {	# Set the work down for now
		name = tm_embryo.0011.c
		tm_embryo_set_down_effect = yes
	}
}

# Tenth month: birth from the crown of the head. Root = the adept.
tm_embryo.0012 = {
	type = character_event
	title = tm_embryo.0012.t
	desc = tm_embryo.0012.desc
	theme = learning
	left_portrait = { character = root animation = happiness }
	trigger = { has_variable = tm_embryo_active }

	option = {
		name = tm_embryo.0012.a
		remove_variable = tm_embryo_active
		remove_character_modifier = tm_gestating_modifier
		# The society's dues at a birth (P.3489; research/cn_ning_hao_sheyi_wenshu.md): her House celebrates.
		if = {
			limit = { tm_holds_sisters_house_trigger = yes }
			save_scope_as = tm_society_host
		}
		else_if = {
			limit = { liege ?= { tm_holds_sisters_house_trigger = yes } }
			liege = { save_scope_as = tm_society_host }
		}
		if = {
			limit = { exists = scope:tm_society_host }
			scope:tm_society_host = {
				add_piety = 15
				every_held_title = {
					limit = { tier = tier_county has_variable = tm_house }
					change_variable = { name = tm_house_strain add = -1 }
				}
			}
		}
		tm_rise_daughter_effect = { TRAIT = tm_daughter_of_the_embryo RENAME = tm_embryo.0002 }
	}
}
