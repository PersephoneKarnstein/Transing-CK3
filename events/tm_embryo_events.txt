namespace = tm_embryo

# The guarantor's question. Root = the adept.
tm_embryo.0001 = {
	type = character_event
	title = tm_embryo.0001.t
	desc = tm_embryo.0001.desc
	theme = learning
	left_portrait = { character = root animation = personality_rational }

	option = {	# "I will not repent."
		name = tm_embryo.0001.a
		set_variable = tm_embryo_active
		add_character_modifier = { modifier = tm_gestating_modifier days = 300 }
		custom_tooltip = tm_embryo.0001.a.tt
		trigger_event = { id = tm_embryo.0010 days = { 80 100 } }
	}
	option = {	# Not yet
		name = tm_embryo.0001.b
	}
}

# Third month: stilling the essence, and the palace censor at the gate. The danger is outside her: the
# censor wants her entered in the eunuchs' register. A failure delays the gestation; it never resets it
# (Serano: the gate recognises, it does not test). Root = the adept.
tm_embryo.0010 = {
	type = character_event
	title = tm_embryo.0010.t
	desc = tm_embryo.0010.desc
	theme = learning
	left_portrait = { character = root animation = personality_rational }
	trigger = { has_variable = tm_embryo_active }

	option = {	# Hold the stillness; let the censor wait at the gate
		name = tm_embryo.0010.a
		random_list = {
			70 = {
				modifier = { add = { value = learning multiply = 2 } }
				modifier = { add = 20 has_personal_tenet_flag = tm_personal_embryo_easier }
				desc = tm_embryo.trial_kept
				trigger_event = { id = tm_embryo.0011 days = { 80 100 } }
			}
			30 = {
				desc = tm_embryo.trial_delayed
				tm_embryo_delayed_effect = { EVENT = tm_embryo.0010 }
			}
		}
	}
	option = {	# The sisters of the guan stand at the gate for me
		name = tm_embryo.0010.b
		trigger = {
			any_courtier = {
				NOT = { this = root }
				OR = { tm_is_daughter_trigger = yes has_trait = tm_transgender }
				faith = root.faith
			}
		}
		trigger_event = { id = tm_embryo.0011 days = { 80 100 } }
	}
	option = {	# Pay the censor to forget the road to my door
		name = tm_embryo.0010.c
		trigger = { gold >= 30 }
		remove_short_term_gold = 30
		trigger_event = { id = tm_embryo.0011 days = { 80 100 } }
	}
	option = {	# Set the work down for now
		name = tm_embryo.0010.d
		tm_embryo_set_down_effect = yes
	}
}
