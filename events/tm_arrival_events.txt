namespace = tm_arrival

# A Marked traveller arrives asking for the Sisters. Root = the ruler.
tm_arrival.0001 = {
	type = character_event
	title = tm_arrival.0001.t
	desc = tm_arrival.0001.desc
	theme = faith
	left_portrait = { character = scope:tm_arrival animation = worry }

	trigger = {
		exists = capital_province
	}

	immediate = {
		create_character = {
			gender = male
			age = { 16 35 }
			culture = root.capital_county.culture
			faith = root.faith
			location = root.capital_province
			trait = tm_transgender
			save_scope_as = tm_arrival
		}
		# She says what she is: she arrives already named (Serano: recognise, do not diagnose).
		scope:tm_arrival = {
			tm_give_daughter_name_effect = yes
			tm_declare_effect = yes
		}
	}

	option = {	# Shelter her
		name = tm_arrival.0001.a
		add_courtier = scope:tm_arrival
		add_piety = 25
	}
	option = {	# Shelter her and take her to the Elder Sisters
		name = tm_arrival.0001.b
		trigger = { tm_rite_can_raise_daughters_trigger = yes }
		add_courtier = scope:tm_arrival
		add_piety = -50
		scope:tm_arrival = { tm_rise_by_rite_effect = { ROOT_RULER = root } }
		# With re-embodiment the daughter is a new character; otherwise she is the same one.
		if = {
			limit = { exists = scope:tm_new_body }
			scope:tm_new_body = { save_scope_as = tm_new_daughter }
		}
		else = {
			scope:tm_arrival = { save_scope_as = tm_new_daughter }
		}
		trigger_event = { id = tm_arrival.0002 days = 2 }	# after her rename (tm_knife.0002 fires at 1 day)
	}
	option = {	# Turn her away: she goes on to another House, or to the roads
		name = tm_arrival.0001.c
		add_piety = -25
		scope:tm_arrival = { tm_send_on_effect = { EXCLUDE = root } }
	}
}

# Second page: the traveller has risen a daughter. Root = the ruler; scope:tm_new_daughter = her.
tm_arrival.0002 = {
	type = character_event
	title = tm_arrival.0002.t
	desc = tm_arrival.0002.desc
	theme = faith
	left_portrait = { character = scope:tm_new_daughter animation = happiness }

	trigger = {
		exists = scope:tm_new_daughter
		scope:tm_new_daughter = { is_alive = yes }
	}

	option = {	# Stay at my court
		name = tm_arrival.0002.a
		if = {
			limit = { NOT = { scope:tm_new_daughter = { is_courtier_of = root } } }
			add_courtier = scope:tm_new_daughter
		}
		scope:tm_new_daughter = {
			add_opinion = { target = root modifier = grateful_opinion opinion = 30 }
		}
	}
	option = {	# Stay as my daughter
		name = tm_arrival.0002.c
		trigger = {
			rite ?= { rite_has_parameter = tm_adoptive_matriline }
			age > scope:tm_new_daughter.age		# a parent can't be younger than her child
		}
		if = {
			limit = { NOT = { scope:tm_new_daughter = { is_courtier_of = root } } }
			add_courtier = scope:tm_new_daughter
		}
		adopt_effect = {
			CHILD = scope:tm_new_daughter
			ADOPTER = root
		}
		scope:tm_new_daughter = {
			add_opinion = { target = root modifier = grateful_opinion opinion = 30 }
		}
	}
	option = {	# Go to the Mother House
		name = tm_arrival.0002.b
		trigger = {
			rite ?= { rite_has_parameter = tm_day_of_blood_active }
			exists = title:d_tm_mother_house.holder
			NOT = { title:d_tm_mother_house.holder = root }
		}
		title:d_tm_mother_house.holder = { add_courtier = scope:tm_new_daughter }
		add_piety = 50
	}
}

# A friend of the House: a man or a woman of the faith asks to stay and serve as the House's guest. The Houses
# are not separatist (research/book_raha_vdd_trans_femme_futures.md row 19). Root = the ruler.
tm_arrival.0004 = {
	type = character_event
	title = tm_arrival.0004.t
	desc = tm_arrival.0004.desc
	theme = faith
	left_portrait = { character = scope:tm_friend animation = personality_rational }

	trigger = {
		exists = capital_province
	}

	immediate = {
		random_list = {
			60 = {
				create_character = {
					gender = male
					age = { 20 50 }
					culture = root.capital_county.culture
					faith = root.faith
					location = root.capital_province
					save_scope_as = tm_friend
				}
			}
			40 = {
				create_character = {
					gender = female
					age = { 20 50 }
					culture = root.capital_county.culture
					faith = root.faith
					location = root.capital_province
					save_scope_as = tm_friend
				}
			}
		}
	}

	option = {	# Welcome, friend
		name = tm_arrival.0004.a
		add_courtier = scope:tm_friend
		scope:tm_friend = {
			add_character_modifier = { modifier = tm_house_friend_modifier }
			add_opinion = { target = root modifier = grateful_opinion opinion = 20 }
		}
		add_piety = 10
	}
	option = {	# Send them on to another House
		name = tm_arrival.0004.b
		scope:tm_friend = { tm_send_on_effect = { EXCLUDE = root } }
	}
}
