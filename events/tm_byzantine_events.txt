# The suppression ladder: the codes against heretics, rung by rung (research/byz_trevett_montanism.md). It climbs
# faster after the Komnenian turn of 1081 (research/byz_sideris_eunuques_societe.md). Root = a Dorleyite ruler
# whose liege keeps another Christian faith. var:tm_suppression counts the rungs reached.

# Rung 1: the books.
tm_byz.0010 = {
	type = character_event
	title = tm_byz.0010.t
	desc = tm_byz.0010.desc
	theme = faith
	left_portrait = { character = root animation = worry }

	# Scapegoat pressure (Transpsychoanalytics row 1): a hostile faith that is secure lets this pass some years
	# (tm_mech_scapegoat_effect, common/scripted_effects/tm_mech_effects.txt).
	trigger = { NOT = { has_variable = tm_scapegoat_calm } }

	immediate = { set_variable = { name = tm_suppression value = 1 } }

	option = {	# Copy them first
		name = tm_byz.0010.a
		trigger = { gold >= 50 }
		remove_short_term_gold = 50
	}
	option = {	# Hide them
		name = tm_byz.0010.b
		random_list = {
			60 = {
				modifier = { add = { value = intrigue multiply = 2 } }
				desc = tm_byz.0010.b.hidden
			}
			40 = {
				desc = tm_byz.0010.b.found
				liege ?= { add_opinion = { target = root modifier = angry_opinion opinion = -20 } }
			}
		}
	}
	option = {	# Hand them over
		name = tm_byz.0010.c
		add_piety = -50
		add_stress = 20
	}
	option = {	# Galenic wing (wave 3, docs/wave3/dorley.md): show them the Emperor's own count. Leo VI's Novel 98 counts
				# three kinds (Sidéris 2010, p. 89); the wing's books agree with it, so the bishop's men leave them.
				# The cost is the name: the Houses' own register goes with the rest.
		name = tm_byz.0010.e
		trigger = { rite ?= { rite_has_parameter = tm_castrates_third_kind } }
		add_piety = -25
		liege ?= { add_opinion = { target = root modifier = respect_opinion opinion = 10 } }
	}
}

# Rung 3: the wills. A heretic's estate goes to orthodox kin, which would undo every adopted daughter.
tm_byz.0012 = {
	type = character_event
	title = tm_byz.0012.t
	desc = tm_byz.0012.desc
	theme = faith
	left_portrait = { character = root animation = anger }

	# Scapegoat pressure (Transpsychoanalytics row 1): a hostile faith that is secure lets this pass some years
	# (tm_mech_scapegoat_effect, common/scripted_effects/tm_mech_effects.txt).
	trigger = { NOT = { has_variable = tm_scapegoat_calm } }

	immediate = { set_variable = { name = tm_suppression value = 3 } }

	option = {	# Register the adoptions with the Emperor's court
		name = tm_byz.0012.a
		trigger = { gold >= 100 }
		remove_short_term_gold = 100
		set_variable = tm_wills_registered
	}
	option = {	# Conform outwardly
		name = tm_byz.0012.b
		add_piety = -100
		add_character_modifier = { modifier = tm_outward_conformity_modifier years = 10 }
	}
	option = {	# Defy them
		name = tm_byz.0012.c
		add_prestige = 50
		liege ?= { add_opinion = { target = root modifier = angry_opinion opinion = -40 } }
		every_child = {
			limit = { is_alive = yes tm_is_daughter_trigger = yes }
			add_character_modifier = { modifier = tm_contested_inheritance_modifier years = 10 }
		}
	}
	option = {	# Galenic wing (wave 3): register the adoptions under Leo VI's adoption law (Novels 26-27; Macrides 1990,
				# p. 111), entering each daughter as a eunuch, the kind Novel 98 counts (Sidéris 2010, p. 89). The court
				# raises no objection to the wing that counts as it counts, so it costs no gold. It costs the
				# daughters their name in the register.
		name = tm_byz.0012.e
		trigger = { rite ?= { rite_has_parameter = tm_castrates_third_kind } }
		set_variable = tm_wills_registered
		every_child = {
			limit = { is_alive = yes tm_is_daughter_trigger = yes }
			add_opinion = { target = root modifier = tm_dl2_counted_eunuch_opinion }
		}
	}
}
