# The House Grows Respectable. Root = the holder; scope:tm_house_county = the House.
tm_life.0001 = {
	type = character_event
	title = tm_life.0001.t
	desc = tm_life.0001.desc
	theme = faith
	left_portrait = { character = scope:tm_life_patron animation = personality_rational }
	right_portrait = { character = scope:tm_life_street animation = worry }

	trigger = {
		exists = scope:tm_house_county
		scope:tm_house_county = { has_variable = tm_house }
		any_courtier = { tm_is_daughter_trigger = yes is_adult = yes NOT = { exists = house } }
		any_courtier = { tm_house_member_trigger = yes is_female = yes is_adult = yes exists = house }
	}

	immediate = {
		random_courtier = {
			limit = { tm_house_member_trigger = yes is_female = yes is_adult = yes exists = house }
			weight = {
				base = 1
				modifier = { add = 3 has_trait = arrogant }
				modifier = { add = 2 highest_held_title_tier >= tier_barony }
			}
			save_scope_as = tm_life_highborn
		}
		random_courtier = {
			limit = { tm_is_daughter_trigger = yes is_adult = yes NOT = { exists = house } }
			save_scope_as = tm_life_street
		}
		# The patron: the House's Protectress if she is someone else, then the holder's liege, then the high-born sister.
		scope:tm_house_county = {
			if = {
				limit = {
					has_variable = tm_house_protectress
					var:tm_house_protectress = {
						is_alive = yes
						NOT = { this = root }
					}
				}
				var:tm_house_protectress = { save_scope_as = tm_life_patron }
			}
		}
		if = {
			limit = {
				NOT = { exists = scope:tm_life_patron }
				liege ?= { NOT = { this = root } }		# an independent ruler is her own liege (run 27)
			}
			liege = { save_scope_as = tm_life_patron }
		}
		if = {
			limit = { NOT = { exists = scope:tm_life_patron } }
			scope:tm_life_highborn = { save_scope_as = tm_life_patron }
		}
	}

	option = {	# The door stays open
		name = tm_life.0001.a
		scope:tm_life_patron = {
			add_opinion = { target = root modifier = tm_life_door_kept_open_opinion }
		}
		add_piety = 25
		add_character_modifier = { modifier = tm_life_open_door_modifier years = 5 }
		every_courtier = {
			limit = { tm_is_daughter_trigger = yes NOT = { exists = house } }
			add_opinion = { target = root modifier = respect_opinion opinion = 10 }
		}
		ai_chance = {
			base = 70
			modifier = { add = 20 has_trait = compassionate }
			modifier = { add = 10 has_trait = just }
			modifier = { add = -20 has_trait = arrogant }
		}
	}
	option = {	# Shut it to the street, as they ask
		name = tm_life.0001.b
		scope:tm_life_patron = {
			add_opinion = { target = root modifier = tm_life_door_shut_for_me_opinion }
		}
		scope:tm_house_county = { change_variable = { name = tm_house_strain add = 2 } }
		every_courtier = {
			limit = { tm_is_daughter_trigger = yes NOT = { exists = house } }
			add_opinion = { target = root modifier = tm_life_door_shut_opinion }
		}
		ai_chance = {
			base = 30
			modifier = { add = 20 has_trait = arrogant }
			modifier = { add = 10 has_trait = ambitious }
		}
	}
}
