# A night when the cell could be opened. Root = the plotter; scope:tm_bae2_scheme = the scheme.
tm_bae2.0001 = {
	type = character_event
	title = tm_bae2.0001.t
	desc = tm_bae2.0001.desc
	theme = intrigue
	override_background = { reference = dungeon }
	left_portrait = { character = root animation = scheme }
	right_portrait = { character = scope:tm_bae2_jailer animation = idle }

	trigger = {
		exists = scope:tm_bae2_scheme
		scope:tm_bae2_scheme = {
			scheme_target_character = { tm_bae2_cell_jailer_trigger = { PLOTTER = root } }
		}
	}

	immediate = {
		save_scope_as = tm_bae2_plotter
		scope:tm_bae2_scheme = {
			scheme_target_character = { save_scope_as = tm_bae2_jailer }
		}
		scope:tm_bae2_jailer = {
			random_prisoner = {
				limit = { tm_bae2_cell_prisoner_trigger = yes }
				save_scope_as = tm_bae2_freed
			}
		}
	}

	option = {	# Open the cell tonight
		name = tm_bae2.0001.a
		custom_tooltip = tm_bae2.0001.a.tt
		hidden_effect = {
			random = {
				chance = scope:tm_bae2_scheme.scheme_success_chance
				save_scope_value_as = { name = tm_bae2_success value = yes }
			}
			random = {
				chance = {
					value = 100
					subtract = scope:tm_bae2_scheme.scheme_secrecy
				}
				save_scope_value_as = { name = tm_bae2_discovered value = yes }
			}
			if = {
				limit = { exists = scope:tm_bae2_success }
				trigger_event = tm_bae2.0010
			}
			else = {
				trigger_event = tm_bae2.0020
			}
		}
		ai_chance = {
			base = 50
			modifier = {
				add = 100
				scope:tm_bae2_scheme = { scheme_success_chance >= 60 }
			}
			modifier = {
				add = -40
				scope:tm_bae2_scheme = { scheme_success_chance < 30 }
			}
		}
	}
	option = {	# Wait for a better night
		name = tm_bae2.0001.b
		custom_tooltip = tm_bae2.0001.b.tt
		ai_chance = {
			base = 30
			modifier = { add = 30 has_trait = patient }
			modifier = { add = -20 has_trait = impatient }
		}
	}
}

# The suit is brought at the lord's court. Root = the lord (a Christian lord of Norse heritage, from 1100).
# Wave 7 (docs/wave7/baedling.md §1): the law of the lord's court decides the suit (tm_bae3_triggers.txt).
#   Iceland, the sorcery charge: the one who hired her answers (scope:tm_bae2_seeress = the seeress, scope:tm_bae2_outlaw
#   = her client); with no client known, scope:tm_bae2_no_client and the suit falls.
#   A relapse where the law is capital (scope:tm_bae2_relapse): the secular arm, not the sentence.
tm_bae2.0100 = {
	type = character_event
	title = tm_bae2.0100.t
	desc = {
		first_valid = {
			triggered_desc = {
				trigger = { exists = scope:tm_bae2_no_client }
				desc = tm_bae2.0100.desc.iceland_none
			}
			triggered_desc = {
				trigger = { exists = scope:tm_bae2_seeress }
				desc = tm_bae2.0100.desc.iceland_client
			}
			triggered_desc = {
				trigger = { scope:tm_bae2_charge = flag:seidr }
				desc = tm_bae2.0100.desc.seidr
			}
			desc = tm_bae2.0100.desc.dress
		}
		first_valid = {
			triggered_desc = {
				trigger = { exists = scope:tm_bae2_no_client }
				desc = tm_bae2.0100.desc.iceland_none_end
			}
			triggered_desc = {
				trigger = { exists = scope:tm_bae2_relapse }
				desc = tm_bae2.0100.desc.relapse
			}
			desc = tm_bae2.0100.desc.end
		}
	}
	theme = faith
	override_background = { reference = terrain }
	left_portrait = { character = scope:tm_bae2_accuser animation = disapproval }
	right_portrait = { character = scope:tm_bae2_outlaw animation = personality_rational }
	lower_right_portrait = scope:tm_bae2_seeress

	trigger = {
		scope:tm_bae2_outlaw = {
			is_alive = yes
			is_imprisoned = no
			trigger_if = {
				limit = { NOT = { exists = scope:tm_bae2_seeress } }
				tm_bae2_outlawry_candidate_trigger = yes
			}
		}
		scope:tm_bae2_accuser = { is_alive = yes }
	}

	option = {	# Let the law take its course
		name = tm_bae2.0100.a
		trigger = {
			NOT = { exists = scope:tm_bae2_no_client }
			NOT = { exists = scope:tm_bae2_relapse }
		}
		custom_tooltip = tm_bae2.0100.a.tt
		scope:tm_bae2_outlaw = { trigger_event = tm_bae2.0101 }
		ai_chance = {
			base = 60
			modifier = { add = 30 has_trait = zealous }
			modifier = { add = 20 has_trait = just }
		}
	}
	option = {	# A relapse: the secular arm (Mitchell pp. 123, 169-71). The event ends at the cell door.
		name = tm_bae2.0100.arm
		trigger = { exists = scope:tm_bae2_relapse }
		custom_tooltip = tm_bae2.0100.arm.tt
		tm_bae2_secular_arm_effect = yes
		ai_chance = {
			base = 60
			modifier = { add = 30 has_trait = zealous }
			modifier = { add = 20 has_trait = just }
		}
	}
	option = {	# Quash the suit
		name = tm_bae2.0100.b
		trigger = { NOT = { exists = scope:tm_bae2_no_client } }
		scope:tm_bae2_accuser = {
			if = {
				limit = { NOT = { this = root } }
				add_opinion = { target = root modifier = tm_bae2_suit_quashed_opinion }
			}
		}
		add_piety = minor_piety_loss
		ai_chance = {
			base = 25
			opinion_modifier = { opinion_target = scope:tm_bae2_outlaw multiplier = 0.5 }
			modifier = { add = 30 has_trait = compassionate }
			modifier = { add = 20 has_trait = cynical }
			modifier = { factor = 0 this = scope:tm_bae2_accuser }
		}
	}
	option = {	# Iceland: no one is named as her client, so there is no one to sue
		name = tm_bae2.0100.c
		trigger = { exists = scope:tm_bae2_no_client }
		tm_feud_tally_effect = { WHAT = out_iceland_no_client }
	}
}

# The sentence. Root = the one sued (always AI: she is no ruler). Under Icelandic law this is the seeress's client
# (scope:tm_bae2_seeress exists), who may be cleared by a jury of twelve (Grágás, Mitchell p. 159).

	type = character_event
	title = tm_bae2.0101.t
	desc = {
		first_valid = {
			triggered_desc = {
				trigger = { exists = scope:tm_bae2_seeress }
				desc = tm_bae2.0101.desc.client
			}
			triggered_desc = {
				trigger = { scope:tm_bae2_charge = flag:seidr }
				desc = tm_bae2.0101.desc.seidr
			}
			desc = tm_bae2.0101.desc.dress
		}
		triggered_desc = {
			trigger = {
				scope:tm_bae2_charge = flag:seidr
				current_year >= 1300
			}
			desc = tm_bae2.0101.desc.penance
		}
	}
	theme = faith
	override_background = { reference = terrain }
	left_portrait = { character = root animation = personality_rational }
	right_portrait = { character = scope:tm_bae2_accuser animation = disapproval }
	lower_right_portrait = scope:tm_bae2_seeress

	trigger = { is_alive = yes }

	immediate = {
		save_scope_as = tm_bae2_outlaw

# The confiscation court (féránsdómr), a fortnight after the sentence (Grágás K §53; Ahola p. 97). Root = the accuser,
# who must hold it; scope:tm_bae2_outlaw = the one sentenced; scope:tm_bae2_sentence = flag:lesser or flag:full. Text
# only: the court is not a battle. If her friends stop it, no court is held lawfully and the sentence lapses.
tm_bae2.0105 = {
	type = character_event
	title = tm_bae2.0105.t
	desc = {
		first_valid = {
			triggered_desc = {
				trigger = { exists = scope:tm_bae2_court_failed }
				desc = tm_bae2.0105.desc.failed
			}
			triggered_desc = {
				trigger = { scope:tm_bae2_sentence = flag:full }
				desc = tm_bae2.0105.desc.full
			}
			desc = tm_bae2.0105.desc.lesser
		}
		triggered_desc = {
			trigger = {
				NOT = { exists = scope:tm_bae2_court_failed }
				exists = scope:tm_bae2_magnus
			}
			desc = tm_bae2.0105.desc.magnus
		}
	}
	theme = faith
	override_background = { reference = terrain }
	left_portrait = { character = root animation = personality_rational }
	right_portrait = { character = scope:tm_bae2_outlaw animation = worry }

	trigger = {
		scope:tm_bae2_outlaw = {
			is_alive = yes
			has_variable = tm_bae2_pending_court
		}
	}

	immediate = {
		save_scope_as = tm_bae2_accuser
		scope:tm_bae2_outlaw = {
			remove_variable = tm_bae2_pending_court
			random = {
				chance = tm_bae2_court_fails_value
				save_scope_value_as = { name = tm_bae2_court_failed value = yes }
			}
		}
		if = {
			limit = {
				exists = scope:tm_bae2_lord
				scope:tm_bae2_lord = {
					is_alive = yes
					tm_bae3_lord_magnus_law_trigger = yes
				}
			}
			save_scope_value_as = { name = tm_bae2_magnus value = yes }
		}
	}

	option = {
		name = tm_bae2.0105.a
		trigger = { NOT = { exists = scope:tm_bae2_court_failed } }
		custom_tooltip = tm_bae2.0105.a.tt
		hidden_effect = {
			scope:tm_bae2_outlaw = {
				if = {
					limit = { scope:tm_bae2_sentence = flag:full }
					save_scope_value_as = { name = tm_bae2_forfeit value = gold }
					tm_bae2_forfeit_effect = yes
					tm_bae2_full_outlawry_effect = yes
				}
				else = {	# the life-ring: one mark
					save_scope_value_as = {
						name = tm_bae2_forfeit
						value = {
							value = gold
							max = 10
						}
					}
					tm_bae2_forfeit_effect = yes
					tm_bae2_lesser_outlawry_effect = yes
				}
			}
			tm_feud_tally_effect = { WHAT = out_court_held }
		}
	}
	option = {
		name = tm_bae2.0105.b
		trigger = { exists = scope:tm_bae2_court_failed }
		add_prestige = minor_prestige_loss
		scope:tm_bae2_outlaw = { tm_bae2_lift_outlawry_effect = yes }
		hidden_effect = { tm_feud_tally_effect = { WHAT = out_court_failed } }
	}
}

# A full outlaw offers compensation for a settlement (Miller pp. 262-77; Ahola p. 100). Root = the accuser;
# scope:tm_bae2_outlaw = the outlaw; scope:tm_bae2_price = the sum. On the ride for sympathy (wave 7, §3b;
# tm_bae2_ride_effect) scope:tm_bae2_ride is set and scope:tm_bae2_pledgers counts her well-wishers: the pledged
# silver is the price, she needs no purse, and refusing so many costs him honour.
tm_bae2.0130 = {
	type = character_event
	title = tm_bae2.0130.t
	desc = {
		first_valid = {
			triggered_desc = {
				trigger = { exists = scope:tm_bae2_ride }
				desc = tm_bae2.0130.desc.ride
			}
			desc = tm_bae2.0130.desc
		}
	}
	theme = stewardship
	override_background = { reference = study }
	left_portrait = { character = scope:tm_bae2_outlaw animation = personality_rational }

	trigger = {
		scope:tm_bae2_outlaw = {
			is_alive = yes
			has_character_modifier = tm_bae2_full_outlaw_modifier
			trigger_if = {
				limit = { NOT = { exists = scope:tm_bae2_ride } }
				gold >= scope:tm_bae2_price
			}
		}
	}

	option = {	# Take the price
		name = tm_bae2.0130.a
		if = {
			limit = { exists = scope:tm_bae2_ride }
			add_gold = scope:tm_bae2_price	# the well-wishers' pledges
			hidden_effect = { tm_feud_tally_effect = { WHAT = out_ride_settled } }
		}
		else = {
			scope:tm_bae2_outlaw = { pay_short_term_gold = { target = root gold = scope:tm_bae2_price } }
		}
		scope:tm_bae2_outlaw = {
			tm_bae2_lift_outlawry_effect = yes
			add_opinion = { target = root modifier = tm_bae2_settled_with_me_opinion }
		}
		add_opinion = { target = scope:tm_bae2_outlaw modifier = tm_bae2_settled_with_me_opinion }
		ai_chance = {
			base = 50
			modifier = { add = 40 has_trait = greedy }
			modifier = { add = 30 has_trait = forgiving }
			modifier = { add = -40 has_trait = vengeful }
			modifier = { add = -40 has_trait = zealous }
		}
	}
	option = {	# Refuse
		name = tm_bae2.0130.b
		scope:tm_bae2_outlaw = { set_variable = { name = tm_bae2_bot_refused value = yes years = 3 } }
		if = {	# "pardoned against the chieftains' will" (Ahola p. 100): a district that rode for her sees him refuse
			limit = { exists = scope:tm_bae2_pledgers }
			add_prestige = {
				value = minor_prestige_loss
				multiply = scope:tm_bae2_pledgers
			}
			hidden_effect = { tm_feud_tally_effect = { WHAT = out_ride_refused } }
		}
		ai_chance = {
			base = 50
			modifier = { add = 30 has_trait = stubborn }
			modifier = { add = -40 exists = scope:tm_bae2_ride }	# -20 a pledger: two at least
			modifier = { add = -20 exists = scope:tm_bae2_pledgers scope:tm_bae2_pledgers >= 3 }
			modifier = { add = -20 exists = scope:tm_bae2_pledgers scope:tm_bae2_pledgers >= 4 }
			modifier = { add = -20 exists = scope:tm_bae2_pledgers scope:tm_bae2_pledgers >= 5 }
		}
	}
}
