namespace = tm_refuge

# The Mother House answers the petition. Root = the petitioner.
tm_refuge.0001 = {
	type = character_event
	title = tm_refuge.0001.t
	desc = tm_refuge.0001.desc
	theme = faith
	left_portrait = { character = root animation = personality_zealous }
	right_portrait = {
		trigger = {
			exists = title:d_tm_mother_house.holder
			NOT = { title:d_tm_mother_house.holder = root }
		}
		character = title:d_tm_mother_house.holder
		animation = happiness
	}

	option = {
		name = tm_refuge.0001.a
		custom_tooltip = tm_refuge.0001.a.tt
	}
}

# The Houses send word of three who need a hearth. Any of them may decline (an invitation, not a
# recruitment). Root = the ruler.
tm_refuge.0010 = {
	type = character_event
	title = tm_refuge.0010.t
	desc = {
		desc = tm_refuge.0010.desc
		first_valid = {
			triggered_desc = {
				trigger = { scope:tm_cand_young = { has_variable = tm_declines } }
				desc = tm_refuge.0010.desc_young_declines
			}
			triggered_desc = {
				trigger = { scope:tm_cand_elder = { has_variable = tm_declines } }
				desc = tm_refuge.0010.desc_elder_declines
			}
		}
	}
	theme = faith
	left_portrait = { character = root animation = personality_rational }
	lower_left_portrait = scope:tm_cand_child
	lower_center_portrait = scope:tm_cand_young
	lower_right_portrait = scope:tm_cand_elder

	immediate = {
		# A Marked girl, still a child. She can be taken as a daughter now; the gate waits until 16.
		create_character = {
			gender = male
			age = { 6 11 }
			culture = root.capital_county.culture
			faith = root.faith
			dynasty = none
			location = root.capital_province
			trait = tm_transgender
			save_scope_as = tm_cand_child
		}
		scope:tm_cand_child = {
			tm_give_daughter_name_effect = yes
			set_variable = tm_declared
		}
		# A young daughter and an elder sister. With re-embodiment on, they already have the female body
		# (and are infertile, as re-embodied daughters are); otherwise the body is male.
		if = {
			limit = { has_game_rule = tm_reembodiment_on }
			create_character = {
				gender = female
				age = { 17 24 }
				culture = root.capital_county.culture
				faith = root.faith
				dynasty = none
				location = root.capital_province
				fertility = 0
				save_scope_as = tm_cand_young
			}
			create_character = {
				gender = female
				age = { 26 40 }
				culture = root.capital_county.culture
				faith = root.faith
				dynasty = none
				location = root.capital_province
				fertility = 0
				save_scope_as = tm_cand_elder
			}
		}
		else = {
			create_character = {
				gender = male
				age = { 17 24 }
				culture = root.capital_county.culture
				faith = root.faith
				dynasty = none
				location = root.capital_province
				save_scope_as = tm_cand_young
			}
			create_character = {
				gender = male
				age = { 26 40 }
				culture = root.capital_county.culture
				faith = root.faith
				dynasty = none
				location = root.capital_province
				save_scope_as = tm_cand_elder
			}
		}
		scope:tm_cand_young = {
			tm_add_rite_daughter_trait_effect = { ROOT_RULER = root }
			tm_give_daughter_name_effect = yes
		}
		scope:tm_cand_elder = {
			tm_add_rite_daughter_trait_effect = { ROOT_RULER = root }
			tm_give_daughter_name_effect = yes
			# The elder has had years among the Elder Sisters: a little more learning and stewardship.
			add_learning_skill = 3
			add_stewardship_skill = 2
		}
		# Sometimes one of them would rather stay where she is.
		random_list = {
			70 = {}
			15 = { scope:tm_cand_young = { set_variable = tm_declines } }
			15 = { scope:tm_cand_elder = { set_variable = tm_declines } }
		}
	}

	option = {
		name = tm_refuge.0010.a
		add_courtier = scope:tm_cand_child
		tm_refuge_offer_adoption_effect = { CANDIDATE = scope:tm_cand_child }
	}
	option = {
		name = tm_refuge.0010.b
		trigger = { scope:tm_cand_young = { NOT = { has_variable = tm_declines } } }
		add_courtier = scope:tm_cand_young
		tm_refuge_offer_adoption_effect = { CANDIDATE = scope:tm_cand_young }
	}
	option = {
		name = tm_refuge.0010.c
		trigger = { scope:tm_cand_elder = { NOT = { has_variable = tm_declines } } }
		add_courtier = scope:tm_cand_elder
		tm_refuge_offer_adoption_effect = { CANDIDATE = scope:tm_cand_elder }
	}
	option = {	# all who will come, at a price in piety
		name = tm_refuge.0010.d
		add_piety = -100
		add_courtier = scope:tm_cand_child
		if = {
			limit = { scope:tm_cand_young = { NOT = { has_variable = tm_declines } } }
			add_courtier = scope:tm_cand_young
		}
		if = {
			limit = { scope:tm_cand_elder = { NOT = { has_variable = tm_declines } } }
			add_courtier = scope:tm_cand_elder
		}
		ai_chance = { base = 0 }
	}
	option = {
		name = tm_refuge.0010.e
		ai_chance = { base = 0 }
	}
	# Whoever is not chosen stays in the world as a wanderer; nobody is killed off.
}

# Optional second page: take the newcomer as a daughter at once. Root = the ruler.
tm_refuge.0011 = {
	type = character_event
	title = tm_refuge.0011.t
	desc = tm_refuge.0011.desc
	theme = family
	left_portrait = { character = root animation = happiness }
	right_portrait = { character = scope:tm_refuge_candidate animation = happiness }

	trigger = {
		exists = scope:tm_refuge_candidate
		scope:tm_refuge_candidate = { is_alive = yes is_courtier_of = root }
	}

	option = {
		name = tm_refuge.0011.a
		adopt_effect = {
			CHILD = scope:tm_refuge_candidate
			ADOPTER = root
		}
		scope:tm_refuge_candidate = {
			add_opinion = { target = root modifier = grateful_opinion opinion = 30 }
		}
	}
	option = {
		name = tm_refuge.0011.b
	}
}
