# Year one: the faith chooses its House of the Knife. Fires once per game, for the first human
# player of the faith; the choice rewrites the main rite's tm_knife doctrine.
tm_knife.0100 = {
	type = character_event
	title = tm_knife.0100.t
	desc = tm_knife.0100.desc
	theme = faith
	left_portrait = {
		character = root
		animation = personality_rational
	}

	trigger = {
		is_ai = no
		faith = faith:transbian_maximalism
		NOT = { has_global_variable = tm_knife_house_chosen }
	}

	immediate = {
		set_global_variable = { name = tm_knife_house_chosen value = yes }
	}

	option = {	# Aunt Bea's House (default)
		name = tm_knife.0100.b
		custom_tooltip = tm_knife.0100.b.tt
	}
	option = {	# Grandmother's House
		name = tm_knife.0100.a
		rite = { change_rite_doctrine = tm_knife_grandmother }
		add_dread = 20
	}
	option = {	# The Gift Alone
		name = tm_knife.0100.c
		rite = { change_rite_doctrine = tm_knife_gift_alone }
		add_piety = 100
	}
}
