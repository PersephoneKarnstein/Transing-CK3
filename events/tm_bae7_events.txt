# Hidden. The watch's year. Root = the band's leader; scope:employer, scope:task_contract. Checks the band still
# keeps the road, pays the watch its share of the tax, and ends the term or brings one of the year's events.
tm_bae7.0002 = {
	type = character_event
	hidden = yes

	trigger = {
		exists = var:tm_bae7_vigla_county
	}

	immediate = {
		if = {
			limit = { tm_bae7_vigla_holding_trigger = no }
			if = {
				limit = {
					OR = {
						NOT = { exists = var:tm_bae7_vigla_lord }
						var:tm_bae7_vigla_lord = { is_alive = no }
						NOT = { var:tm_bae7_vigla_county.holder ?= var:tm_bae7_vigla_lord }
						is_at_war_with = var:tm_bae7_vigla_lord
					}
				}
				save_scope_value_as = { name = tm_bae7_end value = flag:lord }
			}
			else = {
				save_scope_value_as = { name = tm_bae7_end value = flag:left }
			}
			trigger_event = tm_bae7.0021
		}
		else = {
			change_variable = { name = tm_bae7_vigla_served add = 1 }
			var:tm_bae7_vigla_county = { save_scope_as = tm_bae7_vigla_county }
			# The watch's share of the road tax.
			scope:employer = {
				if = {
					limit = { gold >= 15 }
					pay_short_term_gold = { target = root gold = 15 }
				}
			}
			if = {
				limit = { var:tm_bae7_vigla_served >= var:tm_bae7_vigla_term }
				# A band that robbed the road may be found out when the lord's men count the caravans.
				if = {
					limit = { has_variable = tm_bae7_vigla_robbed }
					random_list = {
						25 = {
							save_scope_value_as = { name = tm_bae7_end value = flag:found }
							trigger_event = tm_bae7.0021
						}
						75 = { trigger_event = tm_bae7.0020 }
					}
				}
				else = {
					trigger_event = tm_bae7.0020
				}
			}
			else = {
				random_list = {
					30 = { trigger_event = { id = tm_bae7.0010 days = { 10 120 } } }
					25 = { trigger_event = { id = tm_bae7.0011 days = { 10 120 } } }
					20 = {
						trigger = {
							current_year >= 1327
							NOT = { has_variable = tm_bae7_vigla_habit_asked }
						}
						trigger_event = { id = tm_bae7.0012 days = { 10 120 } }
					}
					40 = {}
				}
				trigger_event = { id = tm_bae7.0002 days = 365 }
			}
		}
	}
}

# Hidden. Root = a landed Baedling band losing a war over her land (quarterly pulse). Finds the nearest petty coastal
# lord who would take her in, and brings his word (0031).
tm_bae7.0030 = {
	type = character_event
	hidden = yes

	trigger = {
		tm_bae7_landed_band_trigger = yes
		tm_bae7_losing_her_land_trigger = yes
		NOT = { has_variable = tm_bae7_harbour }
	}

	immediate = {
		ordered_ruler = {
			limit = { tm_bae7_harbour_lord_trigger = { BAND = root } }
			order_by = {
				value = 0
				subtract = "capital_province.squared_distance(root.capital_province)"
			}
			save_scope_as = tm_bae7_harbour_lord
		}
		if = {
			limit = { exists = scope:tm_bae7_harbour_lord }
			set_variable = { name = tm_bae7_harbour_asked value = yes years = 10 }
			if = {
				limit = { is_ai = no }
				tm_flavour_spent_effect = yes
			}
			tm_bae7_tally_effect = { WHAT = harbour_offered }
			trigger_event = tm_bae7.0031
		}
	}
}

# The last letter. Root = whoever holds the county; scope:tm_bae7_band (the wronged band), scope:tm_bae7_sword_county.
# Yaqub, asked for a caliph's patent, showed a sword (Tor p. 150); Tor reads it as a claim to justice, not "might
# makes right" (pp. 150-151). Here it is her line, never the narrator's.
tm_bae7.0040 = {
	type = character_event
	title = tm_bae7.0040.t
	desc = tm_bae7.0040.desc
	theme = martial
	override_background = { reference = throne_room }
	left_portrait = { character = scope:tm_bae7_band animation = anger }

	trigger = {
		scope:tm_bae7_sword_county.holder ?= root
		scope:tm_bae7_band = {
			is_alive = yes
			tm_bae2_faith_band_trigger = yes
			is_at_war = no
		}
	}

	option = {	# Give her the county now
		name = tm_bae7.0040.a
		trigger = {
			tm_band_patron_trigger = { LORD = root }
			scope:tm_bae7_sword_county = { tm_band_grantable_county_trigger = { LORD = root } }
		}
		custom_tooltip = tm_bae7.0040.a.tt
		scope:tm_bae7_band = {
			save_scope_as = tm_band_band
			remove_variable = tm_bae7_wronged_by
			remove_variable = tm_bae7_claim_county
		}
		scope:tm_bae7_sword_county = { save_scope_as = tm_band_county }
		tm_band_grant_patent_effect = yes
		tm_bae7_tally_effect = { WHAT = sword_granted }
		ai_chance = {
			base = 20
			modifier = { add = 30 scope:tm_bae7_band.current_military_strength > current_military_strength }
			modifier = { add = 20 has_trait = craven }
			modifier = { add = 10 has_trait = just }
			modifier = { factor = 0.5 has_trait = arrogant }
			modifier = { factor = 0.5 has_trait = stubborn }
		}
	}
	option = {	# Then let it be settled in the field
		name = tm_bae7.0040.b
		custom_tooltip = tm_bae7.0040.b.tt
		scope:tm_bae7_band = {
			remove_variable = tm_bae7_wronged_by
			remove_variable = tm_bae7_claim_county
			set_variable = { name = tm_bae2_was_band value = yes years = 3 }
			start_war = {
				cb = claim_cb
				target = root
				claimant = scope:tm_bae7_band
				target_title = scope:tm_bae7_sword_county
			}
		}
		tm_bae7_tally_effect = { WHAT = sword_war }
		ai_chance = {
			base = 60
			modifier = { add = 20 has_trait = arrogant }
			modifier = { add = 20 has_trait = wrathful }
		}
	}
}
