# Crusader Dolls — the Worn Garland, wave 10 (docs/wave10/garland.md).

# Waiting for Tai. Root = the daughter.
tm_ml4.0001 = {
	type = character_event
	hidden = yes

	trigger = {
		has_variable = tm_ml4_tai_waiting
		faith = faith:tm_garland
		is_imprisoned = no
	}

	immediate = {
		tm_ml4_await_tai_effect = yes
	}
}

# The Tai bath. Root = the daughter.
tm_ml4.0002 = {
	type = character_event
	title = tm_ml4.0002.t
	desc = tm_ml4.0002.desc
	theme = faith
	override_background = { reference = terrain }
	left_portrait = { character = root animation = personality_rational }

	trigger = { tm_ml4_tai_keeper_trigger = yes }

	option = {	# Keep it alone
		name = tm_ml4.0002.a
		add_character_modifier = { modifier = tm_ml4_tai_keeping_modifier days = 30 }
		set_variable = { name = tm_ml4_tai_kept value = yes years = 3 }
		if = {
			limit = { location.county ?= title:c_kanchipuram }
			add_piety = 15
		}
		trigger_event = { id = tm_ml4.0003 days = 30 }
		tm_ml4_tally_effect = { WHAT = tai_kept }
		ai_chance = {
			base = 60
			modifier = { add = 20 has_trait = zealous }
			modifier = { add = 10 has_trait = patient }
		}
	}
	option = {	# Not this Tai
		name = tm_ml4.0002.b
		add_stress = -5
		set_variable = { name = tm_ml4_tai_kept value = yes years = 1 }
		tm_ml4_tally_effect = { WHAT = tai_not_now }
		ai_chance = {
			base = 30
			modifier = { add = 20 has_trait = lazy }
			modifier = { add = 10 has_trait = impatient }
		}
	}
}
