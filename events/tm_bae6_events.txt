# A night for it. Root = the raiser.
tm_bae6.0001 = {
	type = character_event
	title = tm_bae6.0001.t
	desc = tm_bae6.0001.desc
	theme = intrigue
	override_background = { reference = terrain }
	left_portrait = { character = root animation = scheme }
	right_portrait = { character = scope:tm_bae6_carver animation = personality_rational }

	trigger = {
		exists = scope:tm_bae6_scheme
		scope:tm_bae6_scheme = {
			scheme_target_character = { is_alive = yes }
		}
	}

	immediate = {
		save_scope_as = tm_bae6_raiser
		scope:tm_bae6_scheme = {
			scheme_target_character = { save_scope_as = tm_bae6_target }
		}
		tm_bae6_find_carver_effect = yes
	}

	option = {	# Set it up tonight
		name = tm_bae6.0001.a
		custom_tooltip = tm_bae6.0001.a.tt
		hidden_effect = {
			random = {
				chance = scope:tm_bae6_scheme.scheme_success_chance
				save_scope_value_as = { name = tm_bae6_success value = yes }
			}
			random = {
				chance = {
					value = 100
					subtract = scope:tm_bae6_scheme.scheme_secrecy
				}
				save_scope_value_as = { name = tm_bae6_discovered value = yes }
			}
			scope:tm_bae6_target ?= { top_liege ?= { save_scope_as = tm_bae6_law_lord } }	# the target may be gone (run 28)
			if = {
				limit = { exists = scope:tm_bae6_success }
				trigger_event = tm_bae6.0010
			}
			else = {
				trigger_event = tm_bae6.0020
			}
		}
		ai_chance = {
			base = 50
			modifier = {
				add = 100
				scope:tm_bae6_scheme = { scheme_success_chance >= 60 }
			}
			modifier = {
				add = -40
				scope:tm_bae6_scheme = { scheme_success_chance < 30 }
			}
		}
	}
	option = {	# Wait for a darker night
		name = tm_bae6.0001.b
		custom_tooltip = tm_bae6.0001.b.tt
		ai_chance = {
			base = 30
			modifier = { add = 30 has_trait = patient }
			modifier = { add = -20 has_trait = impatient }
		}
	}
}
