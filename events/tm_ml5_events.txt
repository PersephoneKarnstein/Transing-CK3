# Crusader Dolls — the Worn Garland, wave 11: the world after 1178, as era events in every start (docs/wave11/garland.md).
# The seat, the faith and the gate are the same; only the world changes. Every gate is a date in a source, given at
# each event; a date that is inferred says so. The narrator does not claim the temple women of the record, the
# reciting families, the commentators or any living community for the faith. "Guise" is the commentary's word, said
# by a hostile reciter, never the narrator's word for her.
#   0001  to her and her descendants: the temple sells an honour, a verse or the flywhisk, 1178–1300 (Orr 2000)
#   0002  the honour passes to her daughter, born or taken
#   0010  the commentator's word: a reciter reads the "guise" against her, from 1150 (Carman and Narayanan 1989)
#   0011  the girl's part: the men's performance of the girl at the festival of recitation, from 1150 (Narayanan 1994)
#   0020  both temples: the men's sectarian titles and a keeper of registers who wants one house, from 1178 (Orr 2000)
#   0030  the first house in the quarter: where the king's hand is light, 1178–1300 (Orr 2000)
#   0031  the council's guard: a band of the Garland walks beside the temple's paddy, from 1200 (Hall 1980)
# Saved scopes: tm_ml5_temple_county (0001, 0030); tm_ml5_mother (0002); tm_ml5_host (0010, 0011, 0020);
# tm_ml5_council_county (0031).

# To her and her descendants. Root = a daughter of the Garland living on the street of her share.
# All 21 of Orr's "deals" fall in 1070–1300: a temple woman's gift exchanged for honours, a place near the god in
# procession, the flywhisk, consecrated food, the right to sing a set verse; ten are hereditary, "to her
# descendants" (Orr 2000, pp. 74–75, 151). The verse is the Tiruvempavai's, bought at Nallur in the 12th c. and at
# Tirupampuram in 1213 (pp. 17–18, 56, 105–08). The 1178 gate is Orr's period break (pp. 46, 67–73, 161–63); after
# 1300 temple women stop giving (pp. 174–77). Stage 2's share (tm_ml2.0001) already sells from 1070; this is the
# honour on top of it.
tm_ml5.0001 = {
	type = character_event
	title = tm_ml5.0001.t
	desc = tm_ml5.0001.desc
	theme = faith
	override_background = { reference = temple }
	left_portrait = { character = root animation = personality_rational }

	trigger = {
		tm_ml5_street_daughter_trigger = yes
		gold >= 30
	}

	immediate = {
		var:tm_ml2_share = { save_scope_as = tm_ml5_temple_county }

		name = tm_ml5.0001.a

# The commentator's word. Root = a daughter of the Garland at court; scope:tm_ml5_host = her lord.
# Pillan's Six Thousand, the first commentary on the Tiruvaymoli, was "written between 1100 and 1150" (Carman and
# Narayanan 1989, p. xi): the 1150 gate is the date by which it exists. Pillan says the poet speaks "in the guise of
# another" (p. 96), yet tells decad 6.2 with the poet as "she" (p. 99), and the friend knows the girl "by her exceeding
# love" (6.5.10; p. 76). The reciter is invented and hostile; the word is his. The faith's answer is its own
# inversion (the guise was the truth), said plainly as the faith's.
tm_ml5.0010 = {
	type = character_event
	title = tm_ml5.0010.t
	desc = tm_ml5.0010.desc
	theme = faith
	override_background = { reference = throne_room }
	left_portrait = { character = root animation = personality_rational }
	right_portrait = { character = scope:tm_ml5_host animation = debating }

	trigger = {
		exists = scope:tm_ml5_host
		faith = faith:tm_garland
		employer ?= scope:tm_ml5_host
		is_imprisoned = no
	}

	option = {	# Read on: your teacher calls her "she"
		name = tm_ml5.0010.a
		random_list = {
			45 = {
				modifier = { add = 20 learning >= 12 }
				modifier = { add = 15 learning >= 18 }
				add_prestige = 40
				add_character_modifier = { modifier = tm_ml5_answered_modifier years = 10 }
				scope:tm_ml5_host = { add_opinion = { target = root modifier = respect_opinion opinion = 10 } }
				send_interface_toast = {
					type = event_toast_effect_good
					title = tm_ml5.0010.a.won
				}
				tm_ml5_tally_effect = { WHAT = guise_won }
			}
			40 = {
				modifier = { add = 15 learning < 8 }
				add_piety = 10
				send_interface_toast = {
					type = event_toast_effect_neutral
					title = tm_ml5.0010.a.lost
				}
				tm_ml5_tally_effect = { WHAT = guise_lost }
			}
		}
		ai_chance = {
			base = 30
			modifier = { add = 25 learning >= 12 }
			modifier = { add = 10 has_trait = arrogant }
		}
	}
	option = {	# A guise is put on and taken off; she has nothing to take off
		name = tm_ml5.0010.b
		add_piety = 30
		if = {
			limit = { scope:tm_ml5_host = { faith = faith:vaishnava_faith } }
			scope:tm_ml5_host = { add_opinion = { target = root modifier = disappointed_opinion opinion = -10 } }
		}
		tm_ml5_tally_effect = { WHAT = guise_face }
		ai_chance = {
			base = 30
			modifier = { add = 25 has_trait = zealous }
			modifier = { add = 10 has_trait = brave }
		}
	}
	option = {	# Let him finish
		name = tm_ml5.0010.c
		add_stress = -10
		tm_ml5_tally_effect = { WHAT = guise_let_pass }
		ai_chance = {
			base = 30
			modifier = { add = 20 has_trait = shy }
			modifier = { add = 20 has_trait = calm }
		}
	}
}

# The girl's part. Root = a daughter of the Garland at court; scope:tm_ml5_host = her lord.
# At the festival of recitation a man of the hereditary reciting families sings and dances the poet's girl, her mother
# and her friend (Narayanan 1994, p. 130). In-period evidence is 12th-c. anecdote told in 13th-c. commentary: Empar
# corrects a performer, who accepts (p. 126); Nanjiyar learns from his teacher's weeping (p. 132). The 1150 gate is
# inferred from those anecdotes; no inscription read here dates the service. Women were not allowed to perform (ch. 8
# n. 14), and only three temples have the service (p. 123), so she goes to "the island temple" with the pilgrims.
# The office is not named in the text: the word names a living hereditary office (research note, design implications).
