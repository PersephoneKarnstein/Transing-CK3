			modifier = { add = -30 has_trait = craven }

		name = tm_got.0100.b
		ai_chance = {
			base = 40
			modifier = { add = 20 has_trait = honest }

# Gotland has passed to a lord of another faith. Root = the new holder. Finds the time, the voice of the new court, and
# the one of the faith on the island who will choose; if no one can, the island chooses quietly.
tm_got.0200 = {
	hidden = yes
	immediate = {
		save_scope_as = tm_got_lord
		tm_got_find_era_effect = yes
		# A hostile voice, attributed: the new lord's Christian chaplain, else the lord himself if Christian.
		if = {
			limit = {
				cp:councillor_court_chaplain ?= { religion = religion:christianity_religion }
			}
			cp:councillor_court_chaplain = { save_scope_as = tm_got_voice }
		}
		else_if = {
			limit = { religion = religion:christianity_religion }
			save_scope_as = tm_got_voice
		}
		# Who speaks for the faith's people: a player first, then a ruler, then the ablest.
		if = {
			limit = { any_living_character = { tm_got_islander_trigger = yes } }
			ordered_living_character = {
				limit = { tm_got_islander_trigger = yes }
				order_by = {
					value = martial
					add = diplomacy
					if = {
						limit = { is_ai = no }
						add = 1000
					}
					if = {
						limit = { is_ruler = yes }
						add = 100
					}
				}
				save_scope_as = tm_got_leader
			}
		}
		if = {
			limit = {
				exists = scope:tm_got_leader
				scope:tm_got_leader = { tm_flavour_ready_trigger = yes }
			}
			scope:tm_got_leader = {
				tm_flavour_spent_effect = yes
				trigger_event = tm_got.0201
			}
		}
		else = {
			tm_got_resolve_quietly_effect = yes
		}
	}
}
