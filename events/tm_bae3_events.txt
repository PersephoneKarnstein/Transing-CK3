# The King's Example. Root = a Bædling in a realm whose top liege is a Christian, in its conversion years
# (tm_bae3_lore_conversion_band_trigger); scope:tm_bae3_teller = the courtier who tells it. Mitchell row 1(a): at
# conversion kings make an example of sorcerers, and Þorgeirr's rule leaves secret rites alone (p. 38). Rule 13: the
# teller says the pagan kings burned too (Eiríkr and the Eighty, Price pp. 80–81). Óláfr is never named as the villain.
# The "open" choice is flavour only: it does not open the suit early (the suit is tm_bae2's, and it reads no variable
# set here).
tm_bae3.0001 = {
	type = character_event
	title = tm_bae3.0001.t
	desc = {
		desc = tm_bae3.0001.desc
		first_valid = {
			triggered_desc = {
				trigger = { location.county ?= { tm_bae3_county_icelandic_trigger = yes } }
				desc = tm_bae3.0001.desc.iceland
			}
			triggered_desc = {
				trigger = { location.county ?= { tm_bae3_county_norwegian_trigger = yes } }
				desc = tm_bae3.0001.desc.norway
			}
			triggered_desc = {
				trigger = { location.county ?= { target_is_de_jure_liege_or_above = title:k_denmark } }
				desc = tm_bae3.0001.desc.denmark
			}
			desc = tm_bae3.0001.desc.sweden
		}
		desc = tm_bae3.0001.desc.end
	}
	theme = faith
	override_background = { reference = feast }
	left_portrait = { character = root animation = worry }
	right_portrait = { character = scope:tm_bae3_teller animation = personality_rational }

	trigger = {
		scope:tm_bae3_teller ?= { is_alive = yes }
	}

	option = {	# Behind closed doors
		name = tm_bae3.0001.a
		custom_tooltip = tm_bae3.0001.a.tt
		add_stress = 15
		ai_chance = {
			base = 50
			modifier = { add = 30 has_trait = craven }
			modifier = { add = 20 has_trait = deceitful }
		}
	}
	option = {	# In the hall, where they have always been
		name = tm_bae3.0001.b
		add_piety = medium_piety_gain
		add_stress = -10
		if = {
			limit = {
				liege ?= {
					NOT = { this = root }
					NOT = { rite ?= { rite_has_parameter = tm_sisters_faith } }
				}
			}
			liege = {
				add_opinion = { target = root modifier = tm_bae3_open_rites_opinion }
			}
		}
		ai_chance = {
			base = 25
			modifier = { add = 30 has_trait = zealous }
			modifier = { add = 20 has_trait = brave }
			modifier = { add = -20 has_trait = craven }
		}
	}
	option = {	# To a court of the six faiths
		name = tm_bae3.0001.c
		trigger = { is_ruler = no }
		custom_tooltip = tm_bae3.0001.c.tt
		add_prestige = minor_prestige_loss
		tm_bae3_leave_realm_effect = yes
		ai_chance = {
			base = 15
			modifier = { add = 20 has_trait = paranoid }
		}
	}
}

# The Bishop's Letter. Root = a Bædling who holds a county with a House, in a Christian realm, from 1300;
# scope:tm_bae3_bishop = who brings the letter (the realm's chaplain, else the king himself); scope:tm_bae3_house_county.
# In 1319 Bishop Auðfinnr of Bergen forbade a cult "through offerings, pilgrimages… fasting or prayer": an edict, not a
# trial (contesting_orthodoxy §3b; Mitchell pp. 123, 169–71). Before c. 1430 the clergy call it pagan illusion and sin,
# never a pact (rule 25). The cult and the woman of 1319 are never named. No trial follows from any choice.
tm_bae3.0010 = {
	type = character_event
	title = tm_bae3.0010.t
	desc = tm_bae3.0010.desc
	theme = faith
	override_background = { reference = temple_church }
	left_portrait = { character = root animation = worry }
	right_portrait = { character = scope:tm_bae3_bishop animation = disapproval }

	trigger = {
		scope:tm_bae3_bishop ?= { is_alive = yes }
		scope:tm_bae3_house_county ?= {
			has_variable = tm_house
			holder ?= root
		}
	}

	option = {	# Obey the letter
		name = tm_bae3.0010.a
		custom_tooltip = tm_bae3.0010.a.tt
		scope:tm_bae3_house_county = { change_variable = { name = tm_house_strain add = 1 } }
		scope:tm_bae3_bishop = { add_opinion = { target = root modifier = respect_opinion opinion = 10 } }
		ai_chance = {
			base = 35
			modifier = { add = 30 has_trait = craven }
			modifier = { add = 10 has_trait = content }
		}
	}
	option = {	# Close the doors and keep the rites inside
		name = tm_bae3.0010.b
		add_stress = 10
		ai_chance = {
			base = 40
			modifier = { add = 20 has_trait = deceitful }
			modifier = { add = 10 has_trait = patient }
		}
	}
	option = {	# Answer the bishop
		name = tm_bae3.0010.c
		duel = {
			skill = learning
			target = scope:tm_bae3_bishop
			50 = {
				desc = tm_bae3.0010.c.success
				compare_modifier = {
					value = scope:duel_value
					multiplier = 3.5
					min = -49
				}
				add_prestige = minor_prestige_gain
			}
			50 = {
				desc = tm_bae3.0010.c.failure
				compare_modifier = {
					value = scope:duel_value
					multiplier = -3.5
					min = -49
				}
				scope:tm_bae3_house_county = { change_variable = { name = tm_house_strain add = 1 } }
				scope:tm_bae3_bishop = { add_opinion = { target = root modifier = tm_bae3_answered_bishop_opinion } }
			}
		}
		ai_chance = {
			base = 25
			modifier = { add = 20 learning >= 12 }
			modifier = { add = 20 has_trait = stubborn }
		}
	}
}

# They Say He Keeps Seiðmenn. Root = the lord (any faith, north Germanic or Anglo-Saxon); scope:tm_bae3_rival = who
# spreads the charge; scope:tm_bae3_seidr_one = the daughter of seiðr or friend of the House at his court. "May be no
# more than accusation spread by his enemies"; male seiðr was distasteful enough for "political discreditation"
# (Tolley p. 154). The charge is the rival's voice only. The narrator never confirms that anyone keeps seiðmenn,
# never calls anyone in it sisters or a third kind, and never ties it to Sámi descent (Tolley p. 61, guardrail 5).

	type = character_event

	desc = tm_bae3.0100.desc

		scope:tm_bae3_rival ?= { is_alive = yes }

	immediate = {
		# Whatever the truth, vassals of another faith hear it.

		duel = {
			skill = prowess

# Calling Herself. Root = a daughter, Marked or declared woman of any of the six faiths in an English county,
# 1350–1453. An echo of the record only (v2 ground rule; rykener_and_mobilities.md): she worked as a tapster and an
# embroideress (Henningsen pp. 251–52), a woman "first dressed" her and called her by name (p. 252), and in the record
# she calls herself (p. 250). The older woman is generated (tm_bae3_make_broiderer_effect), never Eleanor, Anne or
# Elizabeth, and she is a friend, not a procuress (rule: Anna and Elizabeth are not villains). No clients, no reveal,
# and sex work is not her fate here. If she has no woman's name yet she may take the broiderer's, choose her own
# (the declaration, tm_house.0001), or say not yet (rule 23).

	type = character_event
	title = tm_bae3.0300.t
	desc = {
		desc = tm_bae3.0300.desc
		first_valid = {
			triggered_desc = {
				trigger = { var:tm_bae3_trade ?= flag:tapster }
				desc = tm_bae3.0300.desc.tapster
			}
			desc = tm_bae3.0300.desc.broiderer
		}
		first_valid = {
			triggered_desc = {
				trigger = { tm_bae3_lore_has_her_name_trigger = yes }
				desc = tm_bae3.0300.desc.named
			}
			desc = tm_bae3.0300.desc.unnamed
		}
	}
	theme = friendly
	override_background = { reference = market }

	immediate = {

			50 = { set_variable = { name = tm_bae3_trade value = flag:broiderer } }
