# A prodigy in the prefecture: a man reported to have become a woman. The omen-readers blamed the court, never the
# person, and mostly only recorded it (research/cn_wuxingzhi_sex_change_records.md). Root = a ruler of the Way.
tm_yutai.0001 = {
	type = character_event
	title = tm_yutai.0001.t
	desc = tm_yutai.0001.desc
	theme = learning
	left_portrait = { character = scope:tm_prodigy animation = worry }

	trigger = { exists = capital_province }

	immediate = {
		create_character = {
			gender = female
			age = { 18 40 }
			culture = root.capital_county.culture
			faith = root.capital_county.faith
			location = root.capital_province
			fertility = 0
			save_scope_as = tm_prodigy
		}
	}

	option = {	# Report it up the chain, as the law requires
		name = tm_yutai.0001.a
		add_prestige = 25
	}
	option = {	# Hold the report
		name = tm_yutai.0001.b
		random_list = {
			70 = { desc = tm_yutai.0001.b.quiet }
			30 = {
				desc = tm_yutai.0001.b.found
				add_prestige = -50
			}
		}
	}
	option = {	# Ordain her in the Way
		name = tm_yutai.0001.c
		add_courtier = scope:tm_prodigy
		scope:tm_prodigy = {
			set_variable = { name = tm_conv_ours value = yes days = 1 }
			set_character_faith = root.faith
			add_trait = tm_daughter_of_the_embryo
			tm_give_daughter_name_effect = yes
			set_variable = tm_declared
			add_opinion = { target = root modifier = grateful_opinion opinion = 30 }
		}
		add_piety = 50
	}
	option = {	# Send her past the gate, to the nearest House
		name = tm_yutai.0001.d
		scope:tm_prodigy = { tm_send_on_effect = { EXCLUDE = root } }
	}
}

# The magistrate's inventory: a wave of confiscation against unnamed temple houses (the Song took everything above
# the "permanent" store in 1150; heirless estates went to the state from 1264). Named Houses are exempt.
# Root = a ruler of the Way holding an unnamed House.
tm_yutai.0010 = {
	type = character_event
	title = tm_yutai.0010.t
	desc = tm_yutai.0010.desc
	theme = stewardship
	left_portrait = { character = root animation = worry }

	# Scapegoat pressure (Transpsychoanalytics row 1): a hostile faith that is secure lets this pass some years
	# (tm_mech_scapegoat_effect, common/scripted_effects/tm_mech_effects.txt).
	trigger = { NOT = { has_variable = tm_scapegoat_calm } }

	option = {	# Open the store to him
		name = tm_yutai.0010.a
		remove_short_term_gold = { value = gold multiply = 0.2 max = 100 min = 0 }
	}
	option = {	# Pay him to count slowly
		name = tm_yutai.0010.b
		trigger = { gold >= 25 }
		remove_short_term_gold = 25
		add_prestige = -25
	}
	option = {	# Hide the store
		name = tm_yutai.0010.c
		random_list = {
			60 = {
				modifier = { add = { value = intrigue multiply = 2 } }
				desc = tm_yutai.0010.c.hidden
			}
			40 = {
				desc = tm_yutai.0010.c.found
				every_held_title = {
					limit = { tier = tier_county has_variable = tm_house }
					change_variable = { name = tm_house_strain add = 2 }
				}
				liege ?= { add_opinion = { target = root modifier = angry_opinion opinion = -20 } }
			}
		}
	}
}
