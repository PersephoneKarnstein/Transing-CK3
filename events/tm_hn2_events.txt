# Ha-Nehpakhot, stage 2 (docs/stage2/ha-nehpakhot.md). Hooks: common/on_action/tm_hn2_on_actions.txt.
#   0001  by moonlight: the spinning circle (Even Bohan p. 16, research/jud_kalonymus_even_bohan.md §§2, 8). In it she
#         may tell of Sarah, whom the sages call an aylonit and for whom, Rashi says, "a female organ was made"
#         (bYevamot 64b with Rashi; research/jud_rabbinic_primary_texts.md §§10-11). Faith's voice only.
#   0002  the spindle talks: what is said at the spindle can reach a court (Mishnah Sotah 6:1, after Rosen 2003,
#         p. 249 n. 26; research/jud_rosen_circumcised_cinderella.md). The worthies answer with Rashi; so does she
#         (Rashi on Deut. 22:5; research/jud_rabbinic_primary_texts.md §15 and summary table).
#   0010  the adornment: at her wedding the spouse may give her an ornament of gold, as Kalonymus imagined a husband
#         would ("he would adorn me with ornaments of gold", Even Bohan p. 17). The choice is the spouse's own, and
#         a refusal costs her nothing.
#   0011  she wears it
#   0020  the bailiff: a rival takes her case to the Christian lord's court. In the Crown of Aragon such charges were
#         tried by the king's officers and usually ended in a fine or a pardon (Assis 2001, pp. 34-36, 44;
#         research/jud_assis_catalonia_sexuality.md §§2.1-2.2, 4); in Provence the fines of the lord's court were his
#         income (Shatzmiller 2023, pp. 321-324). Protection is earned by service, never bought.
#   0029  hidden: a member waits for Carnival (February)
#   0030  Carnival before Purim: the abbot of youth demands the pelote; the seneschal's lieutenant ordered the pledges
#         returned, "a novelty never seen" (Arles, 24 Feb. 1433; Hildenfinger, REJ 41: 73 n. 1, 93-94;
#         Stouff, "Chretiens et juifs" §59). Christian harassment and a lord's protection; never Jewish wrongdoing.
#   0050  a seat below: a widow and her daughter sell their seats, two in the men's school above and one below in
#         the women's (Arles 1451/52, Hildenfinger, REJ 48: 75-76; Stouff n. 24 for 1409). From 1409.
#   0040  the lord's licence: a ruler of the sect is asked to confirm the ban on a daughter. A Provencal community could
#         ban only by its lord's consent (Abba Mari, c. 1305, Shatzmiller 2023, p. 86); the Arles licence of 1405
#         ran for one year and reserved absolution to the lord's vicar (Hildenfinger, REJ 41: 67 n. 5, 85-86).
# Dates: every text is gated by what its source allows (docs/stage2/ha-nehpakhot.md, "Dates"; tm_hn2_triggers.txt):
#   Rashi from 1105, Maimonides' Mishneh Torah from 1180, Sefer ha-Hinukh from 1300, Even Bohan from 1323; the
#   bailiff from 1260 (Assis's first case, Lleida 1263); Carnival from 1400.

# By moonlight. Root = a daughter of the Turning. scope:tm_hn2_court = whose court; scope:tm_hn2_spinner = a younger
# woman or seeker of the faith at that court, if any.
tm_hn2.0001 = {
	type = character_event
	title = tm_hn2.0001.t
	desc = {
		first_valid = {
			triggered_desc = {						# the moonlit circle is Kalonymus's image (Even Bohan p. 16, 1322-23)
				trigger = { tm_hn2_touchstone_trigger = yes }
				desc = tm_hn2.0001.desc
			}
			desc = tm_hn2.0001.desc_early
		}
	}
	theme = faith
	left_portrait = { character = root animation = happiness }
	lower_right_portrait = scope:tm_hn2_spinner

	trigger = {
		is_alive = yes
		is_imprisoned = no
		has_trait = tm_daughter_of_the_turning
		OR = {
			is_ruler = yes
			exists = employer
		}
	}

	immediate = {
		if = {
			limit = { is_ruler = yes }
			save_scope_as = tm_hn2_court
		}
		else = {
			employer = { save_scope_as = tm_hn2_court }
		}
		scope:tm_hn2_court = {
			random_courtier = {
				limit = { tm_hn2_spinner_trigger = yes }
				save_scope_as = tm_hn2_spinner
			}
		}
	}

	option = {	# Tell them of Sarah, from the Talmud alone (bYevamot 64b): before Rashi
		name = tm_hn2.0001.a_early
		trigger = { tm_hn2_rashi_trigger = no }
		tm_hn2_tell_sarah_effect = yes
		ai_chance = {
			base = 40
			modifier = { add = 20 has_trait = zealous }
			modifier = { add = 10 has_trait = gregarious }
		}
	}
	option = {	# Tell them of Sarah: the sages say she was an aylonit, and Rashi says a woman's body was made for her
		name = tm_hn2.0001.a
		trigger = { tm_hn2_rashi_trigger = yes }
		tm_hn2_tell_sarah_effect = yes
		ai_chance = {
			base = 40
			modifier = { add = 20 has_trait = zealous }
			modifier = { add = 10 has_trait = gregarious }
		}
	}
	option = {	# Teach her to comb the flax and spin
		name = tm_hn2.0001.b
		trigger = { exists = scope:tm_hn2_spinner }
		add_stress = -10
		add_stewardship_lifestyle_xp = 50
		scope:tm_hn2_spinner = {
			add_opinion = { target = root modifier = respect_opinion opinion = 15 }
			if = {			# the prayer goes with the spinning, as the elders teach it (tm_hipukh.0020)
				limit = { tm_turning_seeker_trigger = yes }
				set_variable = tm_hevrah_taught
			}
		}
		ai_chance = {
			base = 30
			modifier = { add = 20 has_trait = patient }
			modifier = { add = 10 has_trait = compassionate }
		}
	}
	option = {	# Only listen to their stories
		name = tm_hn2.0001.c
		add_stress = -20
		add_character_modifier = { modifier = tm_spinning_circle_modifier years = 1 }
		random = {
			chance = 5
			set_variable = { name = tm_hn2_spindle_talk value = yes years = 1 }
			trigger_event = { id = tm_hn2.0002 days = { 60 180 } }
		}
		ai_chance = {
			base = 30
			modifier = { add = 20 has_trait = shy }
			modifier = { add = 20 stress >= 50 }
		}
	}
}

	type = character_event
	title = tm_hn2.0020.t
	desc = {
		desc = tm_hn2.0020.desc
		first_valid = {
			triggered_desc = {
				trigger = { tm_hn2_served_lord_trigger = yes }

		trigger = { tm_hn2_served_lord_trigger = yes }
		add_prestige = 25
		scope:tm_hn2_lord = {
			add_opinion = { target = root modifier = respect_opinion opinion = 5 }
		}
		ai_chance = { base = 80 }
	}
	option = {	# Answer the charge before the bailiff myself
		name = tm_hn2.0020.c

			40 = {
				modifier = { add = { value = learning multiply = 2 } }

				add_prestige = 40
			}
			50 = {

						exists = scope:tm_hn2_lord

					add_prestige = -50
				}
			}
		}
		ai_chance = {
			base = 30
			modifier = { add = 20 learning >= 12 }
			modifier = { add = 20 has_trait = brave }
		}
	}
}

# Hidden. Root = a member of the faith at a Christian court. Waits for Carnival (February), a year at most.
tm_hn2.0029 = {
	type = character_event
	hidden = yes

	immediate = {
		if = {
			limit = { current_month = 2 }
			remove_variable = tm_hn2_carnival_wait
			trigger_event = { id = tm_hn2.0030 days = { 1 20 } }
		}
		else_if = {
			limit = { NOT = { has_variable = tm_hn2_carnival_wait } }		# change_variable fails on an unset variable
			set_variable = { name = tm_hn2_carnival_wait value = 1 }
			trigger_event = { id = tm_hn2.0029 days = 30 }
		}
		else_if = {
			limit = { var:tm_hn2_carnival_wait < 12 }
			change_variable = { name = tm_hn2_carnival_wait add = 1 }
			trigger_event = { id = tm_hn2.0029 days = 30 }
		}
		else = {
			remove_variable = tm_hn2_carnival_wait
		}
	}
}
