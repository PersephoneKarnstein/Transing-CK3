# 0010 Ignatios returns. Root = the Mother of Mothers, or the greatest ruler of the faith, and every player of the
# faith. Ignatios, son of Michael I, was castrated at about fourteen when his father fell and made a monk; patriarch
# 847-58 and again from November 867 (Ringrose 2003, pp. 62, 118; Szabo in Spencer-Hall and Gutt 2021, pp. 117-19;
# research/byz_spencer_hall_gutt_hagiography.md row 3, research/byz_ringrose_perfect_servant.md). The only beardless
# patriarch in the Hagia Sophia mosaics (Szabo p. 119). His Life makes him manly and calls him "father" (pp. 121-25).
# The faith does not claim him; the Sisters speak of what the Church does with him.
######################################################################
tm_dl2.0010 = {
	type = character_event
	title = tm_dl2.0010.t
	desc = tm_dl2.0010.desc

		ai_chance = { base = 60 }

		name = tm_dl2.0010.b

		add_stress = -5
		ai_chance = { base = 40 }
	}
}

######################################################################
# 0020 The train to Tarsus. Root = a Dorleyaian ruler in Asia Minor while Cilicia is held by a Muslim ruler.
# Rum boys were carried into Syria and northern Mesopotamia "by means of purchase, or by means of raids" until the
# frontier fortresses fell (al-Muqaddasi, d. c. 990, in Ayalon 1999, pp. 107, 218-19; research/hist_ayalon_eunuchs_caliphs.md
# §4.2). Arab writers named Byzantium as a castration centre (El-Azhari 2019, pp. 143-45; al-Jahiz in Ayalon p. 105).
# Event shape from research/isl_el_azhari_queens_eunuchs.md row 2 ("Boys for Tarsus"). The cutting is never shown,
# and no child is cut, given a rite or a name. Grandmother's House is the one condemned way: it buys the cut children
# from the market and lets the rest go on. Aunt Bea's House cells the merchant; whether he can be brought to the knife is
# the House's ordinary rule (tm_intake_eligible_trigger: violent men only), not this event's.
######################################################################
tm_dl2.0020 = {
	type = character_event
	title = tm_dl2.0020.t
	desc = tm_dl2.0020.desc
	theme = faith
	left_portrait = { character = root animation = worry }

	trigger = { tm_flavour_ready_trigger = yes }

	immediate = {
		tm_flavour_spent_effect = yes
		set_variable = { name = tm_dl2_tarsus_seen value = yes years = 20 }
	}

	option = {	# Gift Alone: buy them all free
		name = tm_dl2.0020.a
		trigger = {
			rite ?= { rite_has_doctrine = tm_knife_gift_alone }
			gold >= 100
		}
		remove_short_term_gold = 100
		tm_dl2_take_in_children_effect = { CUT = 2 UNCUT = 1 }
		add_piety = 100
		ai_chance = { base = 100 }
	}
	option = {	# Aunt Bea's House: free them, and a cell for the merchant
		name = tm_dl2.0020.b
		trigger = { rite ?= { rite_has_doctrine = tm_knife_aunt_bea } }
		create_character = {
			age = { 30 50 }
			gender = male
			trait = greedy
			faith = faith:sunni
			culture = culture:levantine
			dynasty = none
			employer = root
			save_scope_as = tm_dl2_slaver
		}
		imprison = { target = scope:tm_dl2_slaver type = dungeon }
		tm_dl2_take_in_children_effect = { CUT = 2 UNCUT = 1 }
		add_prestige = 50
		title:c_cilicia.holder ?= {
			add_opinion = { target = root modifier = angry_opinion opinion = -15 }
		}
		ai_chance = { base = 100 }
	}
	option = {	# Grandmother's House: pay for the cut ones
		name = tm_dl2.0020.c
		trigger = {
			rite ?= { rite_has_doctrine = tm_knife_grandmother }
			gold >= 100
		}
		remove_short_term_gold = 100
		tm_dl2_take_cut_children_effect = yes
		add_dread = 10
		ai_chance = { base = 100 }
	}
	option = {	# Send word to the strategos
		name = tm_dl2.0020.d
		trigger = { top_liege ?= { faith = { religion_tag = christianity_religion } } }
		random_list = {
			50 = {
				desc = tm_dl2.0020.d.stopped
				add_piety = 25
			}
			50 = {
				desc = tm_dl2.0020.d.paid
				add_stress = 10
			}
		}
		ai_chance = { base = 20 }
	}
	option = {	# Past the Gates
		name = tm_dl2.0020.e
		add_stress = 15
		ai_chance = { base = 5 }
	}
}

# 0030 The Galenic wing's book. Root = a Dorleyaian ruler, from 1107. Theophylact of Ochrid wrote his logos "In
# Defence of Eunuchs" c. 1100-07 for his eunuch brother Demetrios (d. c. 1107): a monk attacks eunuchs and a eunuch
# answers him; the laws against castration are outmoded; a boy cut by his family for chastity is a vine pruned;
# ascetics alter their bodies too; adults who cut themselves are their own murderers (Tougher 2008, pp. 75, 98, 108;
# Mullett in Tougher (ed.) 2002, pp. 177-86; Ringrose 2003, pp. 195-97). The dialogue is spoken over the head of the
# eunuch's newly cut nephew (Ringrose p. 195). Mullett: a personal gift, about asceticism, not proof of a shift.
# Aristotelian answer after research/byz_tougher_eunuchs_antiquity_beyond.md row 12 and quote list.
