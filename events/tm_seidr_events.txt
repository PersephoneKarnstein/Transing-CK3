namespace = tm_seidr

# The high seat. Root = the one going under.
tm_seidr.0001 = {
	type = character_event
	title = tm_seidr.0001.t
	desc = tm_seidr.0001.desc
	theme = witchcraft
	left_portrait = { character = root animation = worry }

	option = {	# Go under
		name = tm_seidr.0001.a
		random_list = {
			60 = {
				modifier = { add = 30 has_trait = tm_transgender }
				modifier = { add = 20 has_personal_tenet_flag = tm_personal_seidr_easier }
				modifier = { add = { value = learning multiply = 2 } }
				desc = tm_seidr.0001.a.success
				send_interface_toast = {
					type = event_toast_effect_good
					title = tm_seidr.0001.a.success
				}
				tm_rise_daughter_effect = { TRAIT = tm_daughter_of_seidr RENAME = tm_seidr.0002 }
				# The declaration goes to whichever body survives the rite.
				if = {
					limit = { exists = scope:tm_new_body }
					scope:tm_new_body = { trigger_event = { id = tm_seidr.0003 days = 3 } }
				}
				else = {
					trigger_event = { id = tm_seidr.0003 days = 3 }
				}
			}
			40 = {
				modifier = { add = 20 has_trait = craven }
				desc = tm_seidr.0001.a.failure
				send_interface_toast = {
					type = event_toast_effect_bad
					title = tm_seidr.0001.a.failure
					add_stress = 30
				}
			}
		}
	}
	option = {	# Not this winter
		name = tm_seidr.0001.b
		add_stress = 10
	}
}
