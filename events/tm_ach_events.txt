# Crusader Dolls — flavour "achievements": the Deeds of the House. Design: docs/faith_design.md §11, kept as dark comedy
# by the user (§12 revision 8). Fired by tm_ach_after_knife_effect (the knife's paths) and tm_ach_grandmother_pulse.
# Root: the player who brought the deed about (or, for a deed by consent, a player of the new daughter's faith).
# Saved scopes: tm_ach_target (the old body), tm_ach_daughter (the body that survives the rite), tm_ach_consent,
# tm_ach_target_faith, tm_ach_master, tm_ach_khan, tm_ach_khan_is_self.
# Writing: the comedy is at the courts' panic, never at the cut or the body (§16 rule 5); the knife makes the body the
# world reads, not the woman (rule 8). Hostile lines belong to a named court.

# Ioanna Secunda: the Pope. Aunt Bea's House (the Dorley path) reads as uneasy rehabilitation; the Gift Alone version
# (he asked for the knife himself) is the rarer consent deed. Grandmother's House: tm_ach.0002.
tm_ach.0001 = {
	type = character_event
	title = tm_ach.0001.t
	desc = {
		first_valid = {
			triggered_desc = {
				trigger = { exists = scope:tm_ach_consent }
				desc = tm_ach.0001.desc_consent
			}
			desc = tm_ach.0001.desc
		}
		first_valid = {
			triggered_desc = {
				trigger = { current_year >= 1250 }
				desc = tm_ach.0001.desc_joan
			}
			desc = tm_ach.0001.desc_nojoan
		}
	}
	theme = faith
	left_portrait = { character = scope:tm_ach_daughter animation = personality_zealous }
	right_portrait = { character = root animation = happiness }

	trigger = {
		is_ai = no
		exists = scope:tm_ach_daughter
		NOT = { has_character_flag = tm_ach_pope_done }
	}

	immediate = {
		if = {
			limit = { exists = scope:tm_ach_consent }
			tm_ach_record_effect = { DEED = pope_consent FLAG = pope }
			add_piety = 300
		}
		else = {
			tm_ach_record_effect = { DEED = pope FLAG = pope }
			tm_ach_nickname_effect = { NICK = tm_ach_nick_the_unmaker }
			add_prestige = 500
			add_piety = 100
			tm_ach_memories_effect = yes
			# The See's faith remembers for a generation (§11).
			every_ruler = {
				limit = {
					faith = scope:tm_ach_target_faith
					NOT = { this = scope:tm_ach_daughter }
				}
				custom = tm_ach.0001.rulers_tt
				add_opinion = { target = root modifier = tm_ach_unmade_the_pope_opinion }
			}
		}
	}

	option = {	# Ioanna Secunda.
		name = tm_ach.0001.a
	}
	option = {	# Take her as my daughter.
		name = tm_ach.0001.b
		trigger = {
			NOT = { exists = scope:tm_ach_consent }
			rite ?= { rite_has_parameter = tm_adoptive_matriline }
			scope:tm_ach_daughter = {
				is_alive = yes
				NOT = { is_child_of = root }
				age < root.age
			}
		}
		adopt_effect = { CHILD = scope:tm_ach_daughter ADOPTER = root }
		add_prestige = 100
	}
}

# Gelded Like My Camels: a khan or a khan's son. The steppe kept no knife for men and met cut men first at the courts
# of the south (research/east_mongol_sex_genghis.md E26, Terbish p. 24); the khan's camp answers (tm_ach.0031).
tm_ach.0030 = {
	type = character_event
	title = tm_ach.0030.t
	desc = {
		first_valid = {
			triggered_desc = {
				trigger = { exists = scope:tm_ach_khan_is_self }
				desc = tm_ach.0030.desc_khan
			}
			desc = tm_ach.0030.desc_son
		}
	}
	theme = nomads
	left_portrait = { character = scope:tm_ach_daughter animation = personality_bold }
	right_portrait = { character = root animation = happiness }

	trigger = {
		is_ai = no
		exists = scope:tm_ach_daughter
		NOT = { has_character_flag = tm_ach_steppe_done }
	}

	immediate = {
		tm_ach_record_effect = { DEED = steppe FLAG = steppe }
		tm_ach_nickname_effect = { NICK = tm_ach_nick_gelder_of_khans }
		add_prestige = 250
		tm_ach_memories_effect = yes
		# Who speaks for the camp: the khan, for his son; for a khan, her heir.
		if = {
			limit = {
				NOT = { exists = scope:tm_ach_khan_is_self }
				scope:tm_ach_khan = { is_alive = yes }
			}
			scope:tm_ach_khan = { save_scope_as = tm_ach_speaker }
		}
		else = {
			scope:tm_ach_daughter = {
				primary_heir ?= {
					if = {
						limit = {
							is_alive = yes
							NOT = { this = root }
						}
						save_scope_as = tm_ach_speaker
					}
				}
			}
		}
		if = {
			limit = { exists = scope:tm_ach_speaker }
			trigger_event = { id = tm_ach.0031 months = 2 }
		}
	}

	option = {
		name = tm_ach.0030.a
	}
}
