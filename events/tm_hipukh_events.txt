# The Turning (ha-Hipukh), the gate of Ha-Nehpakhot. It is kept at Purim, as weddings and the völva rite are
# kept in parts, but it plays differently (faith_design.md §13):
#   - it can start only in Adar (tm_is_adar_trigger);
#   - it turns on learning and on patience: each Adar's unanswered prayer makes the next likelier, as Leah
#     prayed (tm_turning_prayers never resets);
#   - its stakes are the community's: the baylons and council grant a licence to the county, or shun her
#     (only by the lord's licence: tm_turning_niddui_effect).
# Sources: Even Boḥan (Lemberg 1865) pp. 15–19 and 29, research/jud_kalonymus_even_bohan.md; Berakhot 60a;
# Isserles on OḤ 696:8, research/jud_rabbinic_primary_texts.md.
#   0010  the scroll: the Megillah read in the women's synagogue, the room below the men's (Arles 1452)
#   0011  the revel: from 1323 the young men put on the woman's dress for the joke; she keeps it on
#   0001  the prayer: "turn me from male to female", said aloud
#   0002  the name, on Shushan Purim
#   0003  before the street's judges (worthies, rectors or baylons, by date): the dress has outlived Purim
# Root = the seeker. scope:tm_zekenah = the elder among the women of the lamp; scope:tm_host = whose court it is.

# Hidden: a courtier offered the Turning waits for Adar (tm_offer_courtier_rite_effect). Re-fires monthly, a year at most.
tm_hipukh.0009 = {
	type = character_event
	hidden = yes

	immediate = {
		if = {
			limit = { tm_is_adar_trigger = yes }
			remove_variable = tm_adar_wait
			tm_turning_begin_effect = yes
		}
		else_if = {
			limit = { NOT = { has_variable = tm_adar_wait } }		# change_variable fails on an unset variable
			set_variable = { name = tm_adar_wait value = 1 }
			trigger_event = { id = tm_hipukh.0009 days = 30 }
		}
		else_if = {
			limit = { var:tm_adar_wait < 12 }
			change_variable = { name = tm_adar_wait add = 1 }
			trigger_event = { id = tm_hipukh.0009 days = 30 }
		}
		else = {
			remove_variable = tm_adar_wait
		}
	}
}

# The prayer. Root = the seeker.
tm_hipukh.0001 = {
	type = character_event
	title = tm_hipukh.0001.t
	desc = {
		first_valid = {
			triggered_desc = {
				trigger = { current_date >= 1323.1.1 }
				desc = tm_hipukh.0001.desc_touchstone
			}
			desc = tm_hipukh.0001.desc
		}
	}
	theme = faith
	left_portrait = { character = scope:tm_zekenah animation = personality_zealous }
	right_portrait = { character = root animation = worry }

	trigger = { exists = scope:tm_zekenah }

	option = {	# Say it aloud
		name = tm_hipukh.0001.a
		random_list = {
			30 = {
				modifier = { add = 30 OR = { has_trait = tm_transgender has_variable = tm_declared } }	# Marked, or named before witnesses (§16 rule 8)
				modifier = { add = learning }
				modifier = { add = scope:tm_zekenah.learning }
				modifier = { add = 20 has_personal_tenet_flag = tm_personal_turning_easier }
				modifier = { add = 10 has_variable = tm_turning_women_synagogue }
				modifier = { add = 20 has_variable = tm_turning_kept_dress }
				modifier = { add = 15 has_variable = tm_hevrah_taught }
				modifier = {							# the worthies of this town have licensed it before
					add = 15
					location ?= { county ?= { has_variable = tm_worthies_licence } }
				}
				modifier = {							# the women of the lamp's own street (Arles at the start)
					add = 10
					location ?= { county ?= { has_variable = tm_kahal } }
				}
				modifier = {							# Leah prayed: every Adar's prayer counts
					has_variable = tm_turning_prayers
					add = { value = var:tm_turning_prayers multiply = 10 }
				}
				desc = tm_hipukh.0001.a.success
				send_interface_toast = {
					type = event_toast_effect_good
					title = tm_hipukh.0001.a.success
				}
				tm_turning_clear_effect = yes
				remove_variable = tm_turning_prayers
				remove_variable = tm_hevrah_taught
				# A seeker of another Jewish faith is turned into the sect.
				if = {
					limit = { NOT = { faith = faith:tm_nehpakhot } }
					set_character_faith = faith:tm_nehpakhot
				}
				tm_rise_daughter_effect = { TRAIT = tm_daughter_of_the_turning RENAME = tm_hipukh.0002 }
				# The elders answer whichever body survives the rite.
				if = {
					limit = { exists = scope:tm_new_body }
					scope:tm_new_body = { trigger_event = { id = tm_hipukh.0003 days = 7 } }
				}
				else = {
					trigger_event = { id = tm_hipukh.0003 days = 7 }
				}
			}
			60 = {								# no fault in the seeker: Leah's prayer was answered in its time
				desc = tm_hipukh.0001.a.failure
				send_interface_toast = {
					type = event_toast_effect_bad
					title = tm_hipukh.0001.a.failure
					add_stress = 10
				}
				tm_turning_clear_effect = yes
				if = {
					limit = { has_variable = tm_turning_prayers }
					change_variable = { name = tm_turning_prayers add = 1 }
				}
				else = {
					set_variable = { name = tm_turning_prayers value = 1 }
				}
			}
		}
	}
}

# Before the street's judges. The rabbis only grumble at the Purim dress (permission is Minz and Isserles, 16th c.); now
# she wears it after Purim, and the street must answer. Who judges depends on the date (tm_kahal_judges in
# common/customizable_localization/tm_custom_loc.txt): the worthies before 1215, the archbishop's three rectors from
# 1215, the baylons from 1386 (earliest dated in our sources), with the council from 1401. Root = the daughter.
tm_hipukh.0003 = {
	type = character_event
	title = tm_hipukh.0003.t
	desc = {
		first_valid = {
			triggered_desc = {						# Arles, Oct. 1401: the council called five women (Stouff; research/jud_stouff_gross_arles.md)
				trigger = { scope:tm_council_women = yes }
				desc = tm_hipukh.0003.desc_council
			}
			triggered_desc = {						# the young men's Purim dress is first attested in Even Bohan (1322-23)
				trigger = { current_date >= 1323.1.1 }
				desc = tm_hipukh.0003.desc
			}
			desc = tm_hipukh.0003.desc_early		# before it, she alone wore one (as in tm_hipukh.0011.desc_early)
		}
	}

	immediate = {
		if = {
			limit = {
				current_date >= 1401.1.1
				location ?= { county ?= { has_variable = tm_kahal } }
			}
			save_scope_value_as = { name = tm_council_women value = yes }
		}
		else = {
			save_scope_value_as = { name = tm_council_women value = no }
		}
	}
	theme = faith
	left_portrait = { character = root animation = worry }

	option = {	# Answer with Dinah: a learned dispute
		name = tm_hipukh.0003.a
		random_list = {
			40 = {
				modifier = { add = { value = learning multiply = 2 } }
				modifier = { add = 10 scope:tm_council_women = yes }	# the women of the street sit in the council
				modifier = { add = 20 has_character_modifier = tm_taught_answers_modifier }	# Teach Her the Answers (tm_mech_interactions.txt)
				desc = tm_hipukh.0003.a.win
				add_prestige = 75
				location ?= { county ?= { set_variable = { name = tm_worthies_licence value = yes years = 10 } } }
			}
			50 = {
				desc = tm_hipukh.0003.a.lose
				tm_turning_niddui_effect = yes
				add_stress = 15
			}
		}
	}
	option = {	# Answer with the Touchstone: the young men put on the dress every Adar
		name = tm_hipukh.0003.b
		trigger = { current_date >= 1323.1.1 }
		add_piety = 50
		add_prestige = 25
	}
	option = {	# "I am the lady of the house"
		name = tm_hipukh.0003.c
		add_piety = 75
		add_stress = -20
		tm_turning_niddui_effect = yes
	}
	option = {	# Put the dress away until next Adar
		name = tm_hipukh.0003.d
		add_stress = 25
	}
}

# The three lamps: what she does, every week (Even Boḥan, p. 17; Mishnah Shabbat 2:6). Root = the daughter.
tm_hipukh.0020 = {
	type = character_event
	title = tm_hipukh.0020.t
	desc = tm_hipukh.0020.desc
	theme = faith
	left_portrait = { character = root animation = happiness }
	lower_right_portrait = scope:tm_hevrah_pupil

	immediate = {
		random_courtier = {
			limit = {
				is_adult = yes
				tm_turning_seeker_trigger = yes
				NOT = { has_variable = tm_hevrah_taught }
			}
			save_scope_as = tm_hevrah_pupil

	}

		name = tm_hipukh.0020.a
		add_character_modifier = { modifier = tm_three_lamps_modifier years = 1 }
	}
	option = {	# Spin with the women by moonlight
		name = tm_hipukh.0020.b
		add_character_modifier = { modifier = tm_spinning_circle_modifier years = 1 }
		every_courtier = {
			limit = { is_female = yes is_adult = yes }

		trigger = { exists = scope:tm_hevrah_pupil }
		scope:tm_hevrah_pupil = {
			set_variable = tm_hevrah_taught
			add_opinion = { target = root modifier = respect_opinion opinion = 15 }
		}
		add_piety = 25
	}
}

# A fisica at court. Women physicians are recorded in dozens of Provençal documents: Sara of Saint-Gilles took a
# pupil at Marseille in 1326; Mayrona was fined at Bayons for practising unlicensed (Shatzmiller 1994, pp. 20–25,
# 109–111; research/jud_shatzmiller_medicine.md). Root = the lord. scope:tm_fisica = the daughter.
tm_hipukh.0031 = {
	type = character_event
	title = tm_hipukh.0031.t
	desc = tm_hipukh.0031.desc
	theme = learning
	left_portrait = { character = scope:tm_fisica animation = personality_rational }

	trigger = { exists = scope:tm_fisica }

	option = {	# Licence her to practise
		name = tm_hipukh.0031.a
		trigger = { gold >= 50 }
		remove_short_term_gold = 50
		scope:tm_fisica = {
			add_character_modifier = { modifier = tm_fisica_licensed_modifier years = 10 }
			add_opinion = { target = root modifier = respect_opinion opinion = 20 }
		}
	}
	option = {	# Licence her to treat women only: "they are better suited to treat sick women than are men"
		name = tm_hipukh.0031.b
		trigger = { gold >= 25 }
		remove_short_term_gold = 25
		scope:tm_fisica = {
			add_character_modifier = { modifier = tm_fisica_women_modifier years = 10 }
			add_opinion = { target = root modifier = respect_opinion opinion = 10 }
		}
	}
	option = {	# No licence
		name = tm_hipukh.0031.c
		scope:tm_fisica = { add_opinion = { target = root modifier = disappointed_opinion opinion = -10 } }
	}
}
