# The Turning (ha-Hipukh), the gate of Ha-Nehpakhot. It is kept at Purim, as weddings and the völva rite are
# kept in parts, but it plays differently (faith_design.md §13):
#   - it can start only in Adar (tm_is_adar_trigger);
#   - it turns on learning and on patience: each Adar's unanswered prayer makes the next likelier, as Leah
#     prayed (tm_turning_prayers never resets);
#   - its stakes are the community's: the baylons and council grant a licence to the county, or shun her
#     (only by the lord's licence: tm_turning_niddui_effect).
# Sources: Even Boḥan (Lemberg 1865) pp. 15–19 and 29, research/jud_kalonymus_even_bohan.md; Berakhot 60a;
# Isserles on OḤ 696:8, research/jud_rabbinic_primary_texts.md.
#   0010  the scroll: the Megillah read in the women's synagogue, the room below the men's (Arles 1452)
#   0011  the revel: from 1323 the young men put on the woman's dress for the joke; she keeps it on
#   0001  the prayer: "turn me from male to female", said aloud
#   0002  the name, on Shushan Purim
#   0003  before the street's judges (worthies, rectors or baylons, by date): the dress has outlived Purim
# Root = the seeker. scope:tm_zekenah = the elder among the women of the lamp; scope:tm_host = whose court it is.

# Hidden: a courtier offered the Turning waits for Adar (tm_offer_courtier_rite_effect). Re-fires monthly, a year at most.
tm_hipukh.0009 = {
	type = character_event
	hidden = yes

	immediate = {
		if = {
			limit = { tm_is_adar_trigger = yes }
			remove_variable = tm_adar_wait
			tm_turning_begin_effect = yes
		}
		else_if = {
			limit = { NOT = { has_variable = tm_adar_wait } }		# change_variable fails on an unset variable
			set_variable = { name = tm_adar_wait value = 1 }
			trigger_event = { id = tm_hipukh.0009 days = 30 }
		}
		else_if = {
			limit = { var:tm_adar_wait < 12 }
			change_variable = { name = tm_adar_wait add = 1 }
			trigger_event = { id = tm_hipukh.0009 days = 30 }
		}
		else = {
			remove_variable = tm_adar_wait
		}
	}
}

		name = tm_hipukh.0001.a

			30 = {
				modifier = { add = 15 any_relation = { type = tm_sister tm_is_daughter_trigger = yes } }	# her sworn sister stands as her sponsor (round 2, T21)
				modifier = { add = 30 OR = { has_trait = tm_transgender has_variable = tm_declared } }	# Marked, or named before witnesses (§16 rule 8)
				modifier = { add = learning }
				modifier = { add = scope:tm_zekenah.learning }
				modifier = { add = 10 scope:tm_zekenah = { has_court_position = tm_voice_lamp_elder_court_position } }	# the court's Elder of the Lamp reads (tm_voice_court_positions.txt)
				modifier = { add = 20 has_personal_tenet_flag = tm_personal_turning_easier }
				modifier = { add = 10 has_variable = tm_turning_women_synagogue }
				modifier = { add = 10 has_variable = tm_hn2_women_seat }	# a seat of her own below (Arles 1409/1452; tm_hn2.0050)
				modifier = { add = 20 has_variable = tm_turning_kept_dress }
				modifier = { add = 10 has_variable = tm_hn3_purim_fire }	# she leapt the Purim fire (wave 3; Hirschberg p. 170)
				modifier = { add = 15 has_variable = tm_hevrah_taught }
				modifier = {							# the worthies of this town have licensed it before
					add = 15
					location ?= { county ?= { has_variable = tm_worthies_licence } }
				}
				modifier = {							# the women of the lamp's own street (Arles at the start)
					add = 10
					location ?= { county ?= { has_variable = tm_kahal } }
				}
				modifier = {							# Leah prayed: every Adar's prayer counts
					has_variable = tm_turning_prayers
					add = { value = var:tm_turning_prayers multiply = 10 }
				}
				desc = tm_hipukh.0001.a.success
				send_interface_toast = {
					type = event_toast_effect_good
					title = tm_hipukh.0001.a.success
				}
				tm_turning_clear_effect = yes
				remove_variable = tm_hn3_purim_fire
				remove_variable = tm_turning_prayers
				remove_variable = tm_hevrah_taught
				# A seeker of another Jewish faith is turned into the sect.
				if = {
					limit = { NOT = { faith = faith:tm_nehpakhot } }
					set_character_faith = faith:tm_nehpakhot
				}
				tm_rise_daughter_effect = { TRAIT = tm_daughter_of_the_turning RENAME = tm_hipukh.0002 }
				# The elders answer whichever body survives the rite.
				if = {
					limit = { exists = scope:tm_new_body }
					scope:tm_new_body = { trigger_event = { id = tm_hipukh.0003 days = 7 } }
				}
				else = {
					trigger_event = { id = tm_hipukh.0003 days = 7 }
				}
			}
			60 = {								# no fault in the seeker: Leah's prayer was answered in its time
				desc = tm_hipukh.0001.a.failure
				send_interface_toast = {
					type = event_toast_effect_bad
					title = tm_hipukh.0001.a.failure
					add_stress = 5
				}
				tm_turning_clear_effect = yes
				remove_variable = tm_hn3_purim_fire
				if = {
					limit = { has_variable = tm_turning_prayers }
					change_variable = { name = tm_turning_prayers add = 1 }
				}
				else = {
					set_variable = { name = tm_turning_prayers value = 1 }
				}
			}
		}
	}
}

# 1215, the baylons from 1386 (earliest dated in our sources), with the council from 1401. Root = the daughter.

				add_stress = 15
			}
		}
	}
	option = {	# A learned answer: the prophets called every man of Israel a bride (round 2, T33;
