			add_piety = medium_piety_gain
		}
		scope:tm_bd3_asker = { add_opinion = { target = root modifier = tm_bd3_kept_from_me_opinion } }
		scope:tm_bd3_runaway = { add_opinion = { target = root modifier = tm_bd3_kept_me_opinion } }
		tm_bd3_tally_effect = { WHAT = rock_kept }
		ai_chance = {
			base = 60
			modifier = { add = 1000 faith = faith:tm_badenya }
			modifier = { add = 30 has_trait = just }
			modifier = { add = 20 has_trait = compassionate }
			modifier = { add = 20 has_trait = brave }
		}
	}
	option = {	# Give her up
		name = tm_bd3.0012.b
		custom_tooltip = tm_bd3.0012.b.tt
		scope:tm_bd3_runaway = {
			add_opinion = { target = root modifier = tm_bd3_gave_me_up_opinion }
			tm_bd3_leave_rock_effect = yes
		}
		if = {
			limit = { scope:tm_bd3_asker = { is_ruler = yes } }
			scope:tm_bd3_asker = { add_courtier = scope:tm_bd3_runaway }
		}
		else_if = {
			limit = { scope:tm_bd3_asker = { exists = court_owner } }
			scope:tm_bd3_asker.court_owner = { add_courtier = scope:tm_bd3_runaway }
		}
		add_prestige = minor_prestige_loss
		tm_bd3_tally_effect = { WHAT = rock_given }
		ai_chance = {
			base = 40
			modifier = { add = 30 has_trait = greedy }
			modifier = { add = 30 has_trait = craven }
			modifier = { add = 20 opinion = { target = scope:tm_bd3_asker value >= 50 } }
		}
	}
}

# A quarrel with the associate (Ibn Battuta IV, p. 417). Root = the ruler.
tm_bd3.0020 = {
	type = character_event
	title = tm_bd3.0020.t
	desc = tm_bd3.0020.desc
	theme = court
	override_background = { reference = throne_room }
	left_portrait = { character = root animation = anger }
	right_portrait = { character = scope:tm_bd3_fallen animation = disapproval }
	lower_right_portrait = { character = scope:tm_bd3_raised trigger = { exists = scope:tm_bd3_raised } }

	trigger = {
		has_variable = tm_bd2_associate
		var:tm_bd2_associate ?= { is_alive = yes }
	}

	immediate = {
		save_scope_as = tm_bd3_ruler
		var:tm_bd2_associate = { save_scope_as = tm_bd3_fallen }
		if = {
			limit = {
				OR = {
					any_spouse = { tm_bd3_rival_ok_trigger = yes }
					any_concubine = { tm_bd3_rival_ok_trigger = yes }
				}
			}
			every_spouse = {
				limit = { tm_bd3_rival_ok_trigger = yes }
				add_to_temporary_list = tm_bd3_rivals
			}
			every_concubine = {
				limit = { tm_bd3_rival_ok_trigger = yes }
				add_to_temporary_list = tm_bd3_rivals
			}
			ordered_in_list = {
				list = tm_bd3_rivals
				order_by = "reverse_opinion(root)"			# whom the ruler favours most
				save_scope_as = tm_bd3_raised
			}
		}
		tm_bd3_tally_effect = { WHAT = quarrels }
	}

	option = {	# Raise the other in her place
		name = tm_bd3.0020.a
		trigger = { exists = scope:tm_bd3_raised }
		custom_tooltip = tm_bd3.0020.a.tt
		tm_bd3_raise_other_effect = yes
		trigger_event = { id = tm_bd3.0021 days = { 30 90 } }
		scope:tm_bd3_fallen = { trigger_event = { id = tm_bd3.0022 days = { 60 180 } } }
		ai_chance = {
			base = 30
			modifier = { add = 40 opinion = { target = scope:tm_bd3_fallen value < 0 } }
			modifier = { add = 20 has_trait = arrogant }
			modifier = { add = 20 has_trait = fickle }
		}
	}
	option = {	# Make peace with her
		name = tm_bd3.0020.b
		add_stress = 10
		scope:tm_bd3_fallen = { add_opinion = { target = root modifier = tm_bd3_reconciled_opinion } }
		add_opinion = { target = scope:tm_bd3_fallen modifier = tm_bd3_reconciled_opinion }
		ai_chance = {
			base = 40
			modifier = { add = 30 has_trait = forgiving }
			modifier = { add = 20 has_trait = calm }
			modifier = { add = 20 opinion = { target = scope:tm_bd3_fallen value >= 20 } }
		}
	}
	option = {	# Let it pass
		name = tm_bd3.0020.c
		ai_chance = { base = 20 }
	}
}

# Ashes and dust (p. 417): the women of the dynasty choose whom to honour. Root = the ruler; scope:tm_bd3_fallen,
# scope:tm_bd3_raised.
tm_bd3.0021 = {
	type = character_event
	title = tm_bd3.0021.t
	desc = {
		desc = tm_bd3.0021.desc
		first_valid = {
			triggered_desc = {
				trigger = { scope:tm_bd3_dust_for ?= flag:fallen }
				desc = tm_bd3.0021.desc.dust
			}
			desc = tm_bd3.0021.desc.ashes
		}
	}
	theme = court
	override_background = { reference = courtyard }
	left_portrait = { character = scope:tm_bd3_fallen animation = sadness }
	lower_left_portrait = scope:tm_bd3_raised
	right_portrait = { character = root animation = thinking }

	trigger = {
		scope:tm_bd3_fallen ?= { is_alive = yes }
		scope:tm_bd3_raised ?= { is_alive = yes }
	}

	immediate = {
		# The grown women of the dynasty, at court or in the realm, by whom they think better of.
		every_courtier_or_guest = {
			limit = { tm_bd3_dynasty_woman_trigger = yes }
			add_to_temporary_list = tm_bd3_cousins
		}
		every_vassal = {
			limit = { tm_bd3_dynasty_woman_trigger = yes }
			add_to_temporary_list = tm_bd3_cousins
		}
		if = {
			limit = {
				any_in_list = {
					list = tm_bd3_cousins
					count >= 1
				}
			}
			if = {
				limit = {
					any_in_list = {
						list = tm_bd3_cousins
						percent >= 0.5
						opinion = { target = scope:tm_bd3_fallen value > 0 }
						NOT = { opinion = { target = scope:tm_bd3_raised value > 50 } }
					}
				}
				save_scope_value_as = { name = tm_bd3_dust_for value = flag:fallen }
			}
		}
	}

	option = {
		name = tm_bd3.0021.a
		if = {
			limit = { scope:tm_bd3_dust_for ?= flag:fallen }
			add_prestige = medium_prestige_loss
			scope:tm_bd3_fallen = {
				add_prestige = medium_prestige_gain
				add_character_modifier = { modifier = tm_bd3_dust_modifier years = 5 }
			}
			tm_bd3_tally_effect = { WHAT = dust }
		}
		else = {
			add_prestige = minor_prestige_gain
			tm_bd3_tally_effect = { WHAT = ashes }
		}
	}
	option = {	# Free her from the commander
		name = tm_bd3.0021.b
		trigger = { scope:tm_bd3_fallen = { has_character_modifier = tm_bd3_kept_by_commander_modifier } }
		scope:tm_bd3_fallen = { remove_character_modifier = tm_bd3_kept_by_commander_modifier }
		add_prestige = minor_prestige_loss
		ai_chance = {
			base = 10
			modifier = { add = 40 scope:tm_bd3_dust_for ?= flag:fallen }
		}
	}
}
