# The pāvai vow, the gate of the Worn Garland (faith_design.md §15). Kept in Mārkaḻi (mid-December to mid-January)
# over a month, with companions. What decides it is whether the circle holds, not one person's skill.
#   0010  the waking: before dawn she wakes the girls of the street (Tiruppāvai; Hardy pp. 416, 436)
#   0011  the song: the Śaiva Tiruvempāvai (Orr pp. 17–18, 107–08) or Āṇṭāḷ's Tiruppāvai
#   0012  the garland worn before it is offered (Āṇṭāḷ's transgression; Venkatesan p. 223)
#   0013  the dream of the wedding: a sign, not the success (Nācciyār Tirumoḻi)
#   0001  the staff on the head and a name: a daughter of God (Goodall pp. 110–112; Orr pp. 52–59)
# Root = the seeker. scope:tm_garland_elder = an elder daughter of the Garland; scope:tm_host = whose court.

# The waking. Root = the seeker.
tm_garland.0010 = {
	type = character_event
	title = tm_garland.0010.t
	desc = tm_garland.0010.desc
	theme = faith
	left_portrait = { character = root animation = happiness }
	right_portrait = { character = scope:tm_garland_elder animation = personality_rational }

	trigger = { exists = scope:tm_garland_elder }

	option = {	# Go from door to door and wake them all
		name = tm_garland.0010.a
		trigger = { gold >= 10 }
		remove_short_term_gold = 10			# sweets for the girls who come
		change_variable = { name = tm_pavai_companions add = 1 }
		trigger_event = { id = tm_garland.0011 days = 10 }
	}
	option = {	# Wake those who will come
		name = tm_garland.0010.b
		trigger_event = { id = tm_garland.0011 days = 10 }
	}
	option = {	# Not this Mārkaḻi
		name = tm_garland.0010.c
		tm_pavai_clear_effect = yes
	}
}

# The song. Root = the seeker.

	}
}

# The staff and the name. Root = the seeker.
tm_garland.0001 = {
	type = character_event
	title = tm_garland.0001.t
	desc = tm_garland.0001.desc
	theme = faith
	left_portrait = { character = scope:tm_garland_elder animation = personality_zealous }
	right_portrait = { character = root animation = worry }

	trigger = { exists = scope:tm_garland_elder }

	option = {	# Bow the head to the staff
		name = tm_garland.0001.a
		random_list = {
			25 = {
				modifier = { add = 15 any_relation = { type = tm_sister tm_is_daughter_trigger = yes } }	# her sworn sister stands as her sponsor (round 2, T21)
				modifier = { add = 30 OR = { has_trait = tm_transgender has_variable = tm_declared } }	# Marked, or named before witnesses (§16 rule 8)
				modifier = { add = 15 has_character_modifier = tm_taught_answers_modifier }	# Teach Her the Answers (tm_mech_interactions.txt)
				modifier = {							# the circle that held
					has_variable = tm_pavai_companions
					add = { value = var:tm_pavai_companions multiply = 8 }
				}
				modifier = { add = 20 has_variable = tm_pavai_garland_worn }
				modifier = { add = 15 location ?= { county ?= { this = title:c_kanchipuram } } }
				desc = tm_garland.0001.a.success
				send_interface_toast = {
					type = event_toast_effect_good
					title = tm_garland.0001.a.success
				}
				tm_pavai_clear_effect = yes
				if = {
					limit = { NOT = { faith = faith:tm_garland } }
					set_character_faith = faith:tm_garland
				}
				tm_rise_daughter_effect = { TRAIT = tm_daughter_of_the_garland RENAME = tm_garland.0002 }
			}
			60 = {								# the circle broke: the vow waits for next Mārkaḻi
				desc = tm_garland.0001.a.failure
				send_interface_toast = {
					type = event_toast_effect_bad
					title = tm_garland.0001.a.failure
					add_stress = 5
				}
				tm_pavai_clear_effect = yes
			}
		}
	}
}

# Hidden: a courtier offered the vow waits for Mārkaḻi (tm_offer_courtier_rite_effect). Re-fires monthly, a year at most.
tm_garland.0009 = {
	type = character_event
	hidden = yes

	immediate = {
		if = {
			limit = { tm_is_markali_trigger = yes }
			remove_variable = tm_markali_wait
			tm_pavai_begin_effect = yes
		}
		else_if = {
			limit = { NOT = { var:tm_markali_wait ?= 12 } }
			change_variable = { name = tm_markali_wait add = 1 }
			trigger_event = { id = tm_garland.0009 days = 30 }
		}
		else = {
			remove_variable = tm_markali_wait
		}
	}
}

# A pandit's objection. A Hindu lord not of the Garland is asked to rule on a daughter at court.
# Root = the lord. scope:tm_garland_daughter = the daughter (any gate).

	type = character_event
	title = tm_garland.0030.t
	desc = tm_garland.0030.desc
	theme = faith

	trigger = { exists = scope:tm_garland_daughter }

	option = {	# "To hell with birth as a woman."
		name = tm_garland.0030.a
		add_piety = 20
		scope:tm_garland_daughter = { add_opinion = { target = root modifier = disappointed_opinion opinion = -20 } }
	}

	}
