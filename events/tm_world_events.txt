# Daughters in a hostile world: items 33, 34, 36, 37, 38 and 42 of the October-reading revisions (faith_design.md §16).
# Hooked in common/on_action/tm_world_on_actions.txt. Every hostile line belongs to a named speaker of a vanilla faith or
# court, and is tagged here by family (inner: the seed's heat; birth: "I baptised a boy"; trace: "born to take") and
# by shape (unpersoning: "a thing"; exposing: "a deceiver"), as §16 rule 3 asks. The narrator never confirms a rumour,
# and never says a child was taken. No hostile line calls a daughter "he": where the speaker would, the text has him
# say "you" or "this one".
#
#   0001  Whispers: a tale about a daughter of the House, drawn from what the teller fears of her faith (33)
#   0002  the child-lifter rumour: a child is missing near the House (33)
#   0003  the child is found, never with the House (33)
#   0010  Many Ledgers: the church's register, the lord's roll and the House book disagree (34)
#   0011  the physician's record, and the House's counter-record (34)
#   0012  the copyist writes her in the masculine (Ha-Nehpakhot, the Jade Embryo) (34)
#   0020  questions in hall: the double bind; a sister answers for her; she may refuse a question about her body (36, 38)
#   0030  the Court Wonders: marvel, miracle or deceit; none of them friendly (37)
#   0031  the abbess keeps a monk-saint's feast and bars the gate; the Sisters answer "we imitate" (37)
#   0040  the qadi's ruling: the Sunni and Shi'i voices; the threatened re-dressing set against the chosen rite (38)
#   0041  Those Without Desire: a place in the women's rooms as one "without desire" (38)
#   0042  she insists she is a woman: the household answers (38)
#   0050  the Women's Hour at the baths (42). Never for Ha-Nehpakhot: no mikveh scene without a source.
#
# Scopes. scope:tm_world_daughter = the daughter (root in 0012 and 0041). scope:tm_world_court = the world's court
# (her liege of another faith, or a neighbour). scope:tm_world_speaker = the one who speaks against her.
# scope:tm_world_sister = the daughter of the House in 0001–0003; scope:tm_world_county = the House's county.

# Whispers. Root = a lord who keeps a Sisters' House. scope:tm_world_rumour picks the tale (Elliot and Lyons's list of
# imputed harms, via tsq_transpsychoanalytics_2017.md row 2 and §3.1):
#   seidr    Bædling: spells that unman (inner; exposing)
#   palace   Yutai Dao: the palace's secrets carried out under a veil (trace; exposing)
#   feast    Ha-Nehpakhot: the women of the lamp pray the young men of the Purim play out of their manhood. Told by
#            gossips of the lord's court, not by Jews; never a child, never Christ, never gold: the blood-libel and
#            host-desecration shapes are barred (inner; exposing)
#   garland  the Worn Garland: she wears the garland before the god does, a profanation (birth; unpersoning)
#   draught  Badenya: a draught that unmans boys (contagion is the teller's word; inner; exposing)
#   baths    Adelphai Dorleyaiou: she goes to the baths and the convent gate to look at women (trace; exposing)
#   ear      any other: she has the lord's ear and turns his mind (birth; exposing)
tm_world.0001 = {
	type = character_event
	title = tm_world.0001.t
	desc = {
		first_valid = {
			triggered_desc = { trigger = { scope:tm_world_rumour = flag:seidr } desc = tm_world.0001.desc_seidr }
			triggered_desc = { trigger = { scope:tm_world_rumour = flag:palace } desc = tm_world.0001.desc_palace }
			triggered_desc = { trigger = { scope:tm_world_rumour = flag:feast } desc = tm_world.0001.desc_feast }
			triggered_desc = { trigger = { scope:tm_world_rumour = flag:garland } desc = tm_world.0001.desc_garland }
			triggered_desc = { trigger = { scope:tm_world_rumour = flag:draught } desc = tm_world.0001.desc_draught }
			triggered_desc = { trigger = { scope:tm_world_rumour = flag:baths } desc = tm_world.0001.desc_baths }
			desc = tm_world.0001.desc_ear
		}
		desc = tm_world.0001.desc_end
	}
	theme = intrigue
	left_portrait = { character = scope:tm_world_sister animation = worry }
	right_portrait = { character = scope:tm_world_speaker animation = scheme }

	trigger = {
		exists = scope:tm_world_sister
		exists = scope:tm_world_speaker
		scope:tm_world_sister = { is_alive = yes is_courtier_of = root }
		scope:tm_world_speaker = { is_alive = yes }
	}

	option = {	# Silence it
		name = tm_world.0001.a
		add_prestige = -30
		scope:tm_world_speaker = { add_opinion = { target = root modifier = tm_world_silenced_opinion } }
		scope:tm_world_sister = { add_opinion = { target = root modifier = grateful_opinion opinion = 10 } }
		ai_chance = {
			base = 30
			modifier = { add = 30 rite ?= { rite_has_parameter = tm_sisters_faith } }
			modifier = { add = 10 has_trait = just }
		}
	}
	option = {	# Let it run
		name = tm_world.0001.b
		scope:tm_world_county ?= { tm_world_strain_house_effect = yes }
		scope:tm_world_sister = {
			add_opinion = { target = root modifier = tm_world_let_run_opinion }
			add_stress = 10
		}
		ai_chance = {
			base = 20
			modifier = { add = 30 has_trait = zealous }
			modifier = { add = 15 has_trait = callous }
			modifier = { factor = 0.1 rite ?= { rite_has_parameter = tm_sisters_faith } }
		}
	}
	option = {	# Disprove it
		name = tm_world.0001.c
		duel = {
			skill = learning
			target = scope:tm_world_speaker
			50 = {
				desc = tm_world.0001.c.success
				compare_modifier = {
					value = scope:duel_value
					multiplier = 3.5
					min = -49
				}
				add_prestige = 30
				scope:tm_world_speaker = { add_opinion = { target = root modifier = tm_world_silenced_opinion } }
				scope:tm_world_sister = { add_opinion = { target = root modifier = respect_opinion opinion = 15 } }
			}
			50 = {
				desc = tm_world.0001.c.failure
				compare_modifier = {
					value = scope:duel_value
					multiplier = -3.5
					min = -49
				}
				add_prestige = -15
				scope:tm_world_county ?= { tm_world_strain_house_effect = yes }
			}
		}
		ai_chance = {
			base = 25
			modifier = { add = 20 learning >= 12 }
		}
	}
}
